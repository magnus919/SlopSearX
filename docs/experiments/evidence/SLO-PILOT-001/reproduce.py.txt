import runpy
from fastapi.testclient import TestClient
from slopsearx import server, metrics
module=runpy.run_path('tests/test_server.py')
with TestClient(server.app) as client:
 server._active_engines={'mocktest':module['_MockEngine']()}
 server._router=None
 for params in ({'q':'ok','format':'json'}, {'q':'error','format':'json'}, {'q':'ok','format':'json','pageno':'bad'}):
  before=sum(metrics.server_requests._values.values())
  result=client.get('/search',params=params)
  print({'params':params,'status':result.status_code,'request_counter_delta':sum(metrics.server_requests._values.values())-before})
print('completed_outcome_metric_exists',hasattr(metrics,'http_search_completed'))
