    def test_transport_timeout_with_unknown_usage_is_a_single_attempt_and_stops(self):
        inputs={k:(N.REPO/k).read_bytes() for k in P.PINS};acq,queries=N.source_data(inputs);out=self.root/'timeout'
        class Timeout:
            def __init__(self,*_): self.last_raw=None;self.last_transport=None
            def __call__(self,body,timeout_s=1):
                sha=hashlib.sha256(body).hexdigest();empty=hashlib.sha256(b'').hexdigest()
                self.last_transport={'status':'timeout','request_sha256':sha,'response_sha256':empty,'stdout_sha256':empty,'stderr_sha256':empty,'stderr_hex':'','remote_http_ms':None}
                raise TimeoutError('transport-timeout')
        with patch.object(N,'Owned',Timeout): N.run(acq,queries,out,offline=True)
        self._bind_analysis(out);r=M.verify(out)
        self.assertEqual((r['physical_attempts'],r['unknown_usage'],r['raw_calls_checked']),(1,True,1))

        def mono(): clock[0]+=.001;return clock[0]
        original_sync=N.sync_directory
