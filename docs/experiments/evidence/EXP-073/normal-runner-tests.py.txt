def load(name,path):
    l=importlib.machinery.SourceFileLoader(name,str(path));s=importlib.util.spec_from_loader(l.name,l);m=importlib.util.module_from_spec(s);l.exec_module(m);return m
P=load('exp073_tests_prepare',HERE/'prepare.py.txt');N=load('exp073_tests_runner',HERE/'normal-runner.py.txt')

def data():
    acq=P.strict_json((N.REPO/'docs/experiments/evidence/EXP-060/fresh-acquisition.json').read_bytes())
    queries=P.strict_json((N.REPO/'docs/experiments/evidence/EXP-060/queries.json').read_bytes())
    return acq,queries
def inputs(): return {k:(N.REPO/k).read_bytes() for k in P.PINS}
def fake_response(req,usage=None,malformed=False):
    answers={k:{'type':'score','score':2,'confidence':.7,'legend':'synthetic'} for k in req['questions']}
    if malformed and answers: answers.pop(next(iter(answers)))
    result={'model':P.MODEL,'answers':answers,'usage':usage or {'input_tokens':10,'output_tokens':2}}
    return json.dumps(result,separators=(',',':')).encode()

class PreparationTests(unittest.TestCase):
    def test_registered_schedule_requests_are_byte_identical_to_pinned_exp072_builders(self):
        acq,queries=data();byid={q['id']:q for q in queries}
        for item in P.PROTOCOL['schedule']:
            old=P.BASE.bodies(byid[item['case']],acq['pools'][item['case']]['cards'],item['variant'])
            new=P.bodies(byid[item['case']],acq['pools'][item['case']]['cards'],item['variant'])
            self.assertEqual(P.wire(new['w0']),P.wire(old['w0']))
            self.assertEqual(P.wire(new['purpose_component']),P.wire(old['candidate']))
            self.assertEqual(new['provenance_order_ids'],old['provenance_order_ids'])
    def test_rotation_preserves_canonical_purpose_request_and_w0_native_tail(self):
        acq,queries=data();q=next(x for x in queries if x['id']=='q1');cards=acq['pools']['q1']['cards']
        b=P.bodies(q,cards,'base');r=P.bodies(q,cards,'rotate')
        self.assertEqual(P.wire(b['purpose_component']),P.wire(r['purpose_component']))
        self.assertEqual(r['provenance_order_ids'],b['provenance_order_ids'][20:]+b['provenance_order_ids'][:20])
        self.assertEqual(r['tail_ids'],r['provenance_order_ids'][40:])
    def test_fusion_requires_exact_full_permutations(self):
        cards=[{'id':str(i),'url':'u'+str(i),'title':'t','snippet':'s'} for i in range(3)]
        for bad in (['0','1'],['0','1','1'],['0','1','x']):
            with self.subTest(order=bad),self.assertRaises(ValueError): N.FUSE.fuse(cards,['0','1','2'],bad)
        result=N.FUSE.fuse(cards,['0','1','2'],['2','1','0'])
        self.assertEqual(set(result['order_ids']),{'0','1','2'});self.assertEqual(set(result['rational_scores']),{'0','1','2'})
