loader.exec_module(A)


def pack(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()


def answer(value):
    return {"type": "score", "score": value, "confidence": .7,
            "legend": {"0": "no fit", "1": "adjacent", "2": "partial", "3": "direct"}}


def scored_arm(cards, scores, order, elapsed=100):
    ids = list(scores)
    return {"status": "valid", "scores": scores, "order_ids": order,
            "raw_answers": {cid: answer(scores[cid]) for cid in ids},
            "rational_scores": {},
            "usage": {"input_tokens": 100, "output_tokens": 100}, "elapsed_ms": elapsed}

    neutral = {"status": "fixture_only", "synthetic_input": True, "quality_credit": False,
               "case": {"id": "neutral", "query": "neutral query", "purpose": "neutral purpose",
                        "facets": [{"id": f"f{i}", "description": f"facet {i}"} for i in range(4)]},

                                      "scores": {}, "raw_answers": {}, "rational_scores": neutral_rational,
                                      "usage": {"input_tokens": 200, "output_tokens": 200}, "elapsed_ms": 5}},
               "candidate_phase_ms": 205}
    operations = []
    for plan in protocol["schedule"]:
        q, variant = plan["case"], plan["variant"]
        native_cards = acquisition["pools"][q]["cards"]
        native = [c["id"] for c in native_cards]
        provenance = native

            provenance = native[20:] + native[:20]
        by_id = {c["id"]: c for c in native_cards}
        cards = [by_id[cid] for cid in A.canonical_ids(native_cards)]
        canonical = A.canonical_ids(cards)
        w0_scores = {cid: float(len(provenance[:40]) - i) / len(provenance[:40]) * 9
                     for i, cid in enumerate(provenance[:40])}
        w0_prefix = sorted(provenance[:40], key=lambda cid: (-w0_scores[cid], provenance[:40].index(cid)))
        w0_order = w0_prefix + provenance[40:]
        purpose_scores = {cid: float(len(canonical) - i) / len(canonical) * 3 for i, cid in enumerate(canonical)}
        purpose_order = sorted(canonical, key=lambda cid: (-purpose_scores[cid], canonical.index(cid)))
        wrow = scored_arm(cards, w0_scores, w0_order)
        prow = scored_arm(cards, purpose_scores, purpose_order)
        fused_order, rational = A._expected_fusion(cards, w0_order, purpose_order)
        usage = {"input_tokens": 200, "output_tokens": 200}
        operations.append({"id": f"{q}-{variant}", "case": q, "variant": variant,
                           "provenance_order_ids": provenance, "cards": cards,

    revision = "e90534bcad9aff3daa34b34399293bfbbec76088"

    run = {"schema": "exp073-run/1", "status": "complete", "synthetic": True,
           "input_pins": copy.deepcopy(protocol["inputs"]), "input_revision": revision,
           "source_integrity": source_integrity,
           "budget": {"attempts": 44, "input_tokens": 4_400, "output_tokens": 4_400, "unknown_usage": False},

class AnalyzerTests(unittest.TestCase):
    def test_neutral_first_failure_with_null_timing_is_incomplete_without_crash(self):
        run, acquisition, references, queries, protocol, input_bytes, protocol_bytes, registration = fixture()
        run["status"] = "incomplete"
        run["neutral"]["arms"]["w0"] = {
            "status": "failed", "elapsed_ms": None, "usage": None, "error": "fixture timeout"}
        run["neutral"]["arms"]["purpose_component"] = None
        run["neutral"]["arms"]["candidate"] = None
        run["neutral"]["candidate_phase_ms"] = None
        result = A.analyze(run, acquisition, references, queries, protocol,
                           input_bytes, protocol_bytes, registration)
        self.assertEqual(result["outcome"], "incomplete")
        self.assertIn("neutral-w0-schema", result["structural_failures"])
        self.assertFalse(result["quality_analysis_performed"])

    def test_first_real_operation_failure_with_null_timing_is_incomplete_without_crash(self):
        run, acquisition, references, queries, protocol, input_bytes, protocol_bytes, registration = fixture()
        run["status"] = "incomplete"
        run["operations"][0]["arms"]["w0"] = {
            "status": "failed", "elapsed_ms": None, "usage": None, "error": "fixture timeout"}
        run["operations"][0]["arms"]["purpose_component"] = None
        run["operations"][0]["arms"]["candidate"] = None
        run["operations"][0]["candidate_phase_ms"] = None
        result = A.analyze(run, acquisition, references, queries, protocol,
                           input_bytes, protocol_bytes, registration)
        self.assertEqual(result["outcome"], "incomplete")
        self.assertIn("incomplete-run", result["structural_failures"])
        self.assertFalse(result["quality_analysis_performed"])

    def test_actual_frozen_inputs_run_offline_and_stop_before_quality(self):
        runner_loader = importlib.machinery.SourceFileLoader("runner073_frozen", str(ROOT / "normal-runner.py.txt"))
        runner_spec = importlib.util.spec_from_loader(runner_loader.name, runner_loader)
        runner = importlib.util.module_from_spec(runner_spec)
        runner_loader.exec_module(runner)
        repo = ROOT.parents[3]
        protocol_bytes = (ROOT / "protocol.json").read_bytes()
        registration_bytes = (ROOT / "registration-integrity.json").read_bytes()
        protocol = A._strict_json(protocol_bytes)
        input_bytes = {path: (repo / path).read_bytes() for path in protocol["inputs"]}
        acquisition, queries = runner.source_data(input_bytes)
        references = A._strict_json(input_bytes[
            "docs/experiments/evidence/EXP-064/completed/metric-reference-view.json"])
        revision = runner.REGISTRATION_REVISION
        source_integrity = {"input_pins": copy.deepcopy(protocol["inputs"]), "input_revision": revision,
                            "protocol_sha256": hashlib.sha256(protocol_bytes).hexdigest(),
                            "registration_sha256": hashlib.sha256(registration_bytes).hexdigest()}
        with tempfile.TemporaryDirectory(prefix="exp073-frozen-offline-") as temp:
            run = runner.run(acquisition, queries, pathlib.Path(temp) / "run", offline=True,
                             source_integrity=source_integrity)
        result = A.analyze(run, acquisition, references, queries, protocol,
                           input_bytes, protocol_bytes, registration_bytes)
        self.assertEqual(run["status"], "complete")
        self.assertEqual(run["budget"]["attempts"], 44)
        self.assertEqual(len(run["operations"]), 21)
        self.assertEqual(result["structural_failures"], [])
        self.assertEqual(result["outcome"], "synthetic_fixture_only")
        self.assertFalse(result["quality_analysis_performed"])
        self.assertNotIn("assessors", result)

    def test_offline_runner_output_matches_analyzer_schema_without_quality_metrics(self):

        runner_spec = importlib.util.spec_from_loader(runner_loader.name, runner_loader)
        runner = importlib.util.module_from_spec(runner_spec)
        runner_loader.exec_module(runner)
        acquisition, references, queries = {"pools": {}}, {"A": {}, "B": {}}, []
        for n in range(1, 14):
            q = f"q{n}"
            cards = [{"id": f"c{i:02}", "url": f"https://synthetic.example/{i:02}",
                      "title": f"Synthetic card {i:02}", "snippet": f"Synthetic snippet {i:02}"}
                     for i in range(80)]
            acquisition["pools"][q] = {"cards": cards}
            query = {"id": q, "kind": "navigation" if n >= 9 else "research",
                            "query": f"synthetic query {q}",
                            "purpose": f"synthetic purpose {q}",
                            "facets": [{"id": f"f{i}", "description": f"Synthetic facet {i}"} for i in range(4)]}
            if n >= 9:
                query["target_url"] = "https://missing.synthetic/target"
            queries.append(query)
            refs = {c["id"]: {"lead": 0, "facets": []} for c in cards}
            references["A"][q] = copy.deepcopy(refs)
            references["B"][q] = copy.deepcopy(refs)
        input_bytes = {
            "docs/experiments/evidence/EXP-060/fresh-acquisition.json": pack(acquisition),
            "docs/experiments/evidence/EXP-060/queries.json": pack(queries),
            "docs/experiments/evidence/EXP-039/source-selection-contract.json": b'{"fixture":"contract"}',
            "docs/experiments/evidence/EXP-064/completed/metric-reference-view.json": pack(references),
            "docs/experiments/evidence/EXP-064/completed/source-bound-references.json": b'{"fixture":"source-bound"}',
            "docs/experiments/evidence/EXP-064/completed/semantic-correction-audit-final.json": b'{"fixture":"audit"}',
            "docs/experiments/evidence/EXP-072/protocol.json": b'{"fixture":"exp072 protocol pin"}',
        }
        protocol = json.loads((ROOT / "protocol.json").read_text())
        protocol["inputs"] = {path: hashlib.sha256(raw).hexdigest() for path, raw in input_bytes.items()}
        protocol_bytes = pack(protocol)
        registration = pack({"schema": "exp073-registration/1", "provider_calls": 0, "search_calls": 0,
                             "pins": {"docs/experiments/EXP-073-fixed-query-purpose-rank-fusion.md": "a" * 64,
                                      "docs/experiments/evidence/EXP-073/protocol.json": hashlib.sha256(protocol_bytes).hexdigest()}})

        source_integrity = {"input_pins": copy.deepcopy(protocol["inputs"]), "input_revision": revision,
                            "protocol_sha256": hashlib.sha256(protocol_bytes).hexdigest(),
                            "registration_sha256": hashlib.sha256(registration).hexdigest()}
        with tempfile.TemporaryDirectory(prefix="exp073-analyzer-integration-") as temp:
            out = pathlib.Path(temp) / "run"
            run = runner.run(acquisition, queries, out, offline=True, source_integrity=source_integrity)
            run.setdefault("input_revision", revision)  # older runner shape; current runner supplies this field
        runner_module_owned = runner.Owned
        owned_bodies = []
        class RecordingOwned(runner_module_owned):
            def __call__(self, body, timeout_s=1):
                owned_bodies.append(json.loads(body))
                return super().__call__(body, timeout_s)
        runner.Owned = RecordingOwned
        try:

                inspected = runner.run(acquisition, queries, pathlib.Path(temp) / "run",

        neutral_purpose = [body for body in owned_bodies

    def test_both_independent_reference_gates_are_required(self):
        x = fixture()
        x[0]["synthetic"] = False
        # Both assessor views independently reject this deliberately unfavorable synthetic cohort.
        for assessor in ("A", "B"):
            for q in x[2][assessor]:
                for row in x[2][assessor][q].values():
                    row["lead"] = 0
                    row["facets"] = []
        # Rebind only the synthetic reference fixture in its pinned bytes.
        path = "docs/experiments/evidence/EXP-064/completed/metric-reference-view.json"
        x[5][path] = pack(x[2])
        x[4]["inputs"][path] = hashlib.sha256(x[5][path]).hexdigest()
        x[0]["input_pins"] = copy.deepcopy(x[4]["inputs"])
        x[6] = pack(x[4])
        registration = json.loads(x[7])
        registration["pins"]["docs/experiments/evidence/EXP-073/protocol.json"] = hashlib.sha256(x[6]).hexdigest()
        x[7] = pack(registration)
        result = A.analyze(*x)
        self.assertEqual(result["structural_failures"], [])
        self.assertFalse(result["assessors"]["A"]["quality_gate_passed"])
        self.assertFalse(result["assessors"]["B"]["quality_gate_passed"])
        self.assertTrue(result["quality_analysis_performed"])
