class FixtureResponse:
    status=200
    def __init__(self,body): self.body=body
    def __enter__(self): return self

    def read(self,n=-1):

class FixtureOpener:
    def __init__(self,mode='neutral'): self.mode=mode
    def open(self,request,timeout):
        if request.method!='POST' or timeout!=1: raise ValueError('fixture-envelope')
        parsed=P.BASE.validate_request(request.data)
        if self.mode=='http-error': raise ValueError('fixture-http-error')
        raw=P.response_fixture(parsed,score=2,usage=None if self.mode=='missing-usage' else {'input_tokens':11,'output_tokens':3})
        return FixtureResponse(raw)
