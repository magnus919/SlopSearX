def atomic(path,value):
    path=Path(path);temp=path.with_name(path.name+'.tmp')
    raw=value if isinstance(value,bytes) else json.dumps(value,ensure_ascii=False,separators=(',',':'),allow_nan=False).encode()
    with temp.open('xb') as f:f.write(raw);f.flush();os.fsync(f.fileno())
    os.replace(temp,path);sync_directory(path.parent)
def sync_directory(path):
    fd=os.open(path,os.O_RDONLY)
    try:os.fsync(fd)
    finally:os.close(fd)
def sync_pending(path,value):
    raw=json.dumps(value,ensure_ascii=False,separators=(',',':'),allow_nan=False).encode()
    with Path(path).open('xb') as f:f.write(raw);f.flush();os.fsync(f.fileno())
    sync_directory(Path(path).parent)
def digest(raw):return hashlib.sha256(raw).hexdigest()

def study_identity(protocol_raw,registration_raw,input_bytes):
    if type(protocol_raw)is not bytes or type(registration_raw)is not bytes or type(input_bytes)is not dict or set(input_bytes)!=set(P.PINS):raise ValueError('study-identity-inputs')

def live_gate(input_bytes,input_revision):

    proof=validate_inputs(input_bytes,input_revision)
    qpath=HERE/'qualification-integrity.json'

    if type(q)is not dict or set(q)!=keys or q['schema']!='exp074-qualification-integrity/1' or q['qualified'] is not True:
        raise ValueError('runner-not-qualified')
    revision=q['qualified_source_revision']
    if type(revision)is not str or len(revision)!=40 or any(c not in '0123456789abcdef' for c in revision):raise ValueError('qualification-revision')
    if type(q['files'])is not dict or set(q['files'])!=OWNED_FILES:raise ValueError('qualification-files')
    if type(q['source_pins'])is not dict or set(q['source_pins'])!=SOURCE_FILES:raise ValueError('qualification-source-files')

            raise ValueError('qualification-source-pin')
    if type(q['dependencies'])is not dict or set(q['dependencies'])!=set(DEPENDENCIES):raise ValueError('qualification-dependencies')

        raise ValueError('qualification-python-pin')

class AsyncOwned:
    def __init__(self,offline=True,client_factory=None,api_key=None):
        self.offline=offline;self.api_key=('offline-fixture-non-secret' if offline else api_key)
        if type(self.api_key)is not str or (not offline and not self.api_key):raise ValueError('missing-credential')
        self.transport=T.AsyncHTTPTransport(client_factory=client_factory)
        self.last_attempt=None
    async def __call__(self,body,timeout_s=1.0,on_dispatch=None):
        try:
            raw,status,complete,timings=await self.transport.exchange(body,self.api_key,timeout_s,on_dispatch)
            self.last_attempt={**self.transport.last_attempt,'status':'returned'}
            return raw,status,complete,timings
        except T.TransportFailure as error:
            self.last_attempt={**self.transport.last_attempt,'status':'failed'}
            raise
        except T.DispatchDenied:
            self.last_attempt={**self.transport.last_attempt,'status':'not_dispatched'}
            raise
        except asyncio.CancelledError:
            self.last_attempt={**self.transport.last_attempt,'status':'cancelled'}
            raise


def fixture_client_factory(mode='neutral',stats=None):
    """Return a fresh MockTransport client per call; handler never reaches the network."""
    stats=stats if stats is not None else {'clients':0,'requests':0,'closed':0}
    class CountingAsyncClient(httpx.AsyncClient):
        async def aclose(self):
            if not getattr(self,'_exp074_counted_closed',False):
                self._exp074_counted_closed=True;stats['closed']+=1
            await super().aclose()
    async def handler(request):
        stats['requests']+=1
        parsed=P.BASE.BASE.validate_request(request.content)
        if mode=='redirect':return httpx.Response(302,headers={'Location':'https://fixture.invalid/redirect'},content=b'')
        if mode=='http-error':
            body=P.response_fixture(parsed,score=2,usage={'input_tokens':11,'output_tokens':3})
            return httpx.Response(503,content=body)
        if mode=='missing-usage':
            body=P.response_fixture(parsed,score=2,usage={'input_tokens':11,'output_tokens':3})
            payload=P.strict_json(body);payload.pop('usage',None)
            return httpx.Response(200,content=P.wire(payload))
        if mode=='overflow':return httpx.Response(200,content=b'x'*(T.MAX_RESPONSE_BYTES+10))
        if mode=='compressed-overflow':
            import gzip
            return httpx.Response(200,headers={'Content-Encoding':'gzip'},content=gzip.compress(b'x'*(T.MAX_RESPONSE_BYTES+10)))
        if mode=='partial-timeout':
            class DelayedStream(httpx.AsyncByteStream):
                async def __aiter__(self):
                    yield b'{"model":"jev-1.13.0","answers":'
                    await asyncio.sleep(2)
                    yield b'{}'
                async def aclose(self):stats['stream_closed']=True
            return httpx.Response(200,stream=DelayedStream())
        if mode=='credential-echo':return httpx.Response(200,content=request.headers['authorization'].encode())
        return httpx.Response(200,content=P.response_fixture(parsed,score=2,usage={'input_tokens':11,'output_tokens':3}))
    def factory(**kwargs):
        stats['clients']+=1
        return CountingAsyncClient(transport=httpx.MockTransport(handler),**kwargs)
    factory.stats=stats
    return factory

                raise ValueError('candidate-phase-deadline' if op['candidate_phase_ms']>1000 else 'wall-budget')
            op['arms']['candidate']['status']='valid';successful=True
        except BaseException as error:
            if isinstance(error,CallFailure) and error.record not in op['calls']:
                op['calls'].append(error.record)
            elif isinstance(error,asyncio.CancelledError) and owned.last_attempt is not None:
                # Call() persists the cancelled receipt before propagating cancellation.
                pending=list(receipt_dir.glob('call-*.json'))
                if pending:
                    last=max(pending,key=lambda p:p.stat().st_mtime_ns)
                    record=P.strict_json(last.read_bytes())
                    if record.get('operation')==opid:op['calls'].append(record)
            # Preserve the valid W0 as complete incumbent fallback whenever available.
            failed_record=error.record if isinstance(error,CallFailure) else None
            if 'w0' not in op['arms']:
                op['arms']['w0']={'status':'failed','order_ids':[],'scores':{},'raw_answers':{},'rational_scores':{},'usage':None,'elapsed_ms':None}
            if 'purpose_component' not in op['arms']:
                failed_purpose=failed_record is not None and failed_record.get('arm')=='purpose_component'
                op['arms']['purpose_component']={'status':'failed' if failed_purpose else 'uninvoked',
                    'order_ids':[],'scores':{},'raw_answers':{},'rational_scores':{},
                    'usage':failed_record.get('usage') if failed_purpose else None,
                    'elapsed_ms':failed_record.get('elapsed_ms') if failed_purpose else None}
            if 'candidate' not in op['arms']:
                known_calls=[x for x in op['calls'] if x.get('dispatched')]
                candidate_usage=None if any(x.get('usage') is None for x in known_calls) else {
                    'input_tokens':sum(x['usage']['input_tokens'] for x in known_calls),
                    'output_tokens':sum(x['usage']['output_tokens'] for x in known_calls)}
                op['arms']['candidate']={'status':'failed_fallback' if incumbent else 'failed','order_ids':list(incumbent),
                    'scores':{},'raw_answers':{},'rational_scores':{},'usage':candidate_usage,
                    'elapsed_ms':None}
            elif op['arms']['candidate']['status']=='candidate_body_durable':
                op['arms']['candidate']['status']='failed_fallback';op['arms']['candidate']['order_ids']=list(incumbent)
                op['arms']['candidate']['rational_scores']={}
            if failed_record is not None:
                name='w0' if failed_record.get('arm')=='w0' else 'purpose_component'
                if name=='w0' and op['arms']['w0']['status']!='valid':
                    op['arms']['w0'].update(usage=failed_record.get('usage'),elapsed_ms=failed_record.get('elapsed_ms'))
                elif name=='purpose_component' and op['arms']['purpose_component']['status']!='valid':
                    op['arms']['purpose_component'].update(usage=failed_record.get('usage'),elapsed_ms=failed_record.get('elapsed_ms'))
            failure_code=error.code if isinstance(error,CallFailure) else ('cancelled' if isinstance(error,asyncio.CancelledError) else str(error) if str(error).replace('-','').isalnum() else type(error).__name__.lower())
            op['failure']={'type':type(error).__name__,'code':failure_code}
            failure_path=receipt_dir/(opid.replace('/','-')+'-failure.json')
            sync0=time.perf_counter_ns();atomic(failure_path,{'receipt_status':'failed','operation':op});failure_fsync_ms=(time.perf_counter_ns()-sync0)/1e6
            endpoint=time.perf_counter();op['candidate_phase_ms']=(endpoint-phase_start)*1000
            op['phase_endpoint_after_fsync']=True
            op['final_receipt']={'path':failure_path.name,'sha256':digest(failure_path.read_bytes()),'fsync_ms':failure_fsync_ms}
            successful=False
        # Candidate phase endpoint is after body or failure sidecar and parent-dir fsync.
        return op,successful

    neutral_case,neutral_cards=P.synthetic_case();neutral_built=P.build_requests(neutral_case,neutral_cards)
    neutral,neutral_ok=await run_operation({'case':'neutral','variant':'neutral'},neutral_built,True)
    neutral.update({'status':'fixture_only' if neutral_ok else 'failed','synthetic_input':True,'quality_credit':False})
    result['neutral']=neutral;checkpoint()
    if not neutral_ok:stopped=True;result['status']='incomplete'
    for item,built in prepared:
        if stopped:
            result['operations'].append(_uninvoked(item,built));continue
        op,ok=await run_operation(item,built,False);result['operations'].append(op)
        if not ok:stopped=True;result['status']='incomplete'
        checkpoint()
    if not stopped:result['status']='complete'
    checkpoint()
    return result
