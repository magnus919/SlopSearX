def load(name,path):
 l=importlib.machinery.SourceFileLoader(name,str(path));s=importlib.util.spec_from_loader(l.name,l);m=importlib.util.module_from_spec(s);l.exec_module(m);return m

def card(i):return {"id":"x%02d"%i,"url":"https://example.test/%02d"%i,"title":"Example source %02d"%i,"snippet":"A bounded public metadata excerpt %02d."%i}
def case(q="ordinary query"):return {"id":"q1","query":q,"purpose":"Find strong public reading leads for this task.","facets":[{"id":"f1","description":"implementation"},{"id":"f2","description":"validation"}]}
def response(request,score=None,usage=True,metadata=True,model=None):
 answers={}
 for cid,q in request["questions"].items():
  maximum=3 if len(q["criteria"])==4 else 9
  value=maximum if score is None else score
  answers[cid]={"type":"score","score":value,**({"confidence":.8,"legend":"test metadata","probabilities":{"low":.1,"high":.9},"ignored":"retained"} if metadata else {})}
 obj={"model":P.MODEL if model is None else model,"answers":answers}
 if usage:obj["usage"]={"input_tokens":11,"output_tokens":3}
 return json.dumps(obj,allow_nan=False,separators=(",",":")).encode()

 def test_score_response_bounds_metadata_and_strict_json_membership(self):
  b=P.bodies(case(),[card(0),card(1)])
  for arm,maximum in (("candidate",3),("absolute_control",9),("w0",9)):
   body=b[arm];ids=list(body["questions"]);parsed=P.validate_score_response(response(body,maximum),body,ids,maximum)
   self.assertEqual(parsed["scores"][ids[0]],maximum);self.assertEqual(parsed["raw_answers"][ids[0]]["legend"],"test metadata")
   for invalid in (maximum+.1,True,-.1):
    raw=response(body,invalid)
    with self.subTest(arm=arm,score=invalid),self.assertRaises(ValueError):P.validate_score_response(raw,body,ids,maximum)
   with self.assertRaises(ValueError):P.validate_score_response(response(body,maximum,model="wrong"),body,ids,maximum)
   changed=json.loads(response(body,maximum));changed["answers"].pop(ids[0])
   with self.assertRaises(ValueError):P.validate_score_response(json.dumps(changed).encode(),body,ids,maximum)
   changed=json.loads(response(body,maximum));changed["answers"]["unexpected-id"]=next(iter(changed["answers"].values()))
   with self.assertRaises(ValueError):P.validate_score_response(json.dumps(changed).encode(),body,ids,maximum)
   with self.assertRaisesRegex(ValueError,"duplicate-json-key"):P.validate_score_response(b'{"model":"jev-1.13.0","model":"jev-1.13.0","answers":{}}',body,ids,maximum)
   with self.assertRaisesRegex(ValueError,"nonfinite-json"):P.validate_score_response(response(body,maximum).replace((b'"score":'+str(maximum).encode()),b'"score":NaN',1),body,ids,maximum)
   with self.assertRaisesRegex(ValueError,"nonfinite-json"):P.validate_score_response(response(body,maximum).replace(b'"ignored":"retained"',b'"ignored":1e999',1),body,ids,maximum)
