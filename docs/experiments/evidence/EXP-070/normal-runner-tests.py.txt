def choice_raw(body,probs=None,selected=None,usage=True):
 answers={}
 options=list(P.PROTOCOL["criteria"])
 for qid,q in body["questions"].items():
  if q["type"]=="score": answers[qid]={"type":"score","score":5}
  else:
   weights=probs or {"unsupported":0.2,"supported_1":0.2,"supported_2":0.2,"supported_3":0.2,"supported_4":0.2}
   answers[qid]={"type":"choice","choice":selected or max(options,key=lambda k:weights[k]),"probabilities":{k:weights[k] for k in options},"confidence":0.8}
 d={"model":P.MODEL,"answers":answers}
 if usage:d["usage"]={"input_tokens":11,"output_tokens":3}
 return json.dumps(d,allow_nan=False,separators=(",",":")).encode()
