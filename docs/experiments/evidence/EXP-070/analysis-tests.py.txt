def answer(support_band=4,support_mass=Decimal('.8')):
    probs={k:Decimal(0) for k in A.OPTIONS}
    probs['unsupported']=Decimal(1) if support_band is None else Decimal(1)-support_mass
    if support_band is not None:probs[f'supported_{support_band}']=support_mass
    raw={k:float(v) for k,v in probs.items()}
    choice=max(A.OPTIONS,key=lambda k:raw[k])
    return {'type':'choice','choice':choice,'probabilities':raw,'confidence':.7}

def add_policy(raw_choices,ids):
    normalized={};mass={};conditional={}
    for cid in ids:
        raw=raw_choices[cid]['probabilities'];d={k:Decimal(str(raw[k])) for k in A.OPTIONS};total=sum(d.values(),Decimal(0));norm={k:d[k]/total for k in A.OPTIONS}
        support=sum((norm[f'supported_{i}'] for i in range(1,5)),Decimal(0));band=(sum((Decimal(i)*norm[f'supported_{i}'] for i in range(1,5)),Decimal(0))/support) if support>Decimal('.5') else None
        normalized[cid]={k:str(v) for k,v in norm.items()};mass[cid]=str(support);conditional[cid]=None if band is None else str(band)
    return normalized,mass,conditional

    def test_raw_choice_distribution_schema_range_mass_and_argmax(self):
        mutations=('bool','nan','range','huge-integer','mass','argmax','confidence','unknown-option','unknown-answer-key','unknown-distribution-key','missing','wrong-id')
        for mutation in mutations:
            x=fixture();a=x[0]['operations'][0]['arms']['relative']['choices']['c0']
            if mutation=='bool':a['probabilities']['unsupported']=True
            elif mutation=='nan':a['probabilities']['unsupported']=float('nan')
            elif mutation=='range':a['probabilities']['unsupported']=1.1
            elif mutation=='huge-integer':a['probabilities']['unsupported']=10**1000
            elif mutation=='mass':a['probabilities']['supported_1']=.3
            elif mutation=='argmax':a['choice']='unsupported'
            elif mutation=='confidence':a['confidence']=float('nan')
            elif mutation=='unknown-option':a['choice']='other'
            elif mutation=='unknown-answer-key':a['metadata']='extra'
            elif mutation=='unknown-distribution-key':a['probabilities']['other']=0
            elif mutation=='missing':del x[0]['operations'][0]['arms']['relative']['choices']['c0']
            elif mutation=='wrong-id':
                row=x[0]['operations'][0]['arms']['relative'];row['choices']['not-a-card']=row['choices'].pop('c0')
            self.assertEqual(A.analyze(*x)['outcome'],'incomplete',mutation)
