"""Mechanical EXP-099 design consistency checks; never grant authority.

The contract deliberately accepts a prospective status so that preparation can
be checked offline. Admission independently requires registered, qualified
source and a fresh runtime health gate.
"""
from __future__ import annotations

import hashlib
import json
from pathlib import Path


def validate(protocol, cohorts, registration, qualification, *, source_prompt_raw):
    expected = (
        (protocol, "exp099-protocol/1"),
        (cohorts, "exp099-cohorts/1"),
        (registration, "exp099-registration/1"),
        (qualification, "exp099-qualification/1"),
    )
    if any(type(obj) is not dict or obj.get("schema") != schema
           for obj, schema in expected):
        raise ValueError("current-design-schema")
    if protocol.get("status") not in {"prospective-not-qualified", "registered", "qualified"}:
        raise ValueError("current-protocol-status")
    if cohorts.get("status") not in {"prospective-not-qualified", "registered", "qualified"}:
        raise ValueError("current-cohorts-status")
    if (registration.get("study_id") != "EXP-099"
            or qualification.get("study_id") != "EXP-099"
            or cohorts.get("study_id") != "EXP-099"):
        raise ValueError("current-study-identity")
    if (protocol.get("predecessor") != "EXP-098"
            or registration.get("predecessor") != "EXP-098"):
        raise ValueError("current-lineage")

    # These public descriptions define the registered execution sequence and
    # scope. Hash resealing cannot make inherited EXP-098/EXP-097 prose current.
    expected_descriptions = (
        (protocol, "fresh_stage_scope", "starts_before",
         "The single health check and first selector call after new admission; all 48 original source references are replayed before the clock.",
         "current-description-fresh-stage-scope"),
        (protocol, "input_provenance", "source",
         "Reuse the same 513 frozen input files and 16 carried card assignments. Retain all 48 accepted source references through original provenance replay: 23 nested and 25 accepted EXP-098 assignments. Exclude rejected responses; perform no repair, regrading or quality selection.",
         "current-description-input-provenance"),
        (protocol, "", "protocol_intervention",
         "Prospectively increase selector per-call input allowance and input reserve from 64000 to 128000, and aggregate reported output allowance from 160000 to 256000. Carry all 48 original source references without repair or regrading. Preserve representation, model, instructions, rubrics, timing, fallback, all numerical quality gates and untouched confirmation.",
         "current-description-protocol-intervention"),
        (protocol, "", "purpose",
         "Continue the terminal EXP-098 development study with all 48 accepted source assignments unchanged: 23 nested and 25 accepted EXP-098 assignments. Dispatch zero source graders and 32 fresh answer graders. Preserve all scientific gates and untouched confirmation requirements.",
         "current-description-purpose"),
        (protocol, "recovery_runtime", "purpose",
         "Replay the frozen original source contexts, task identities and citation mapping. Carry all 48 source references and eight separate R1/R2 closures unchanged; dispatch no source graders. Change only the registered selector token allowances and carry handoff mechanics. Preserve answerer, answer rubrics, context projection and quality gates; change no deployment configuration.",
         "current-description-recovery-runtime"),
        (protocol, "models", "purpose",
         "Resume development after terminal EXP-098 by mechanically replaying all 48 accepted source assignments unchanged. Count 72 historical grader submissions and 32 fresh answer assessments, totaling 104 graders; add 18 free answerer calls for 122 grading/answerer submissions. No source repair, regrading or partial quality analysis is permitted. Delivery, fresh admission and successful health remain mandatory before scientific dispatch.",
         "current-description-model-purpose"),
        (cohorts, "", "request_count_basis",
         "48 carried source assignments plus zero fresh source assignments; 72 historical grader submissions plus 32 fresh answer assessments equals 104 graders; 18 answerer API calls gives 122 grading/answerer submissions; at most 44 Jev calls, one health check and zero searches or source fetches.",
         "current-description-request-count-basis"),
    )
    for obj, section, key, expected_text, error_code in expected_descriptions:
        parent = obj.get(section) if section else obj
        if type(parent) is not dict or type(parent.get(key)) is not str or parent[key] != expected_text:
            raise ValueError(error_code)
    expected_order = [
        "Verify all 513 frozen input files and replay all 48 original source references before the clock, without quality analysis.",
        "Create a fresh EXP-099 clock and admission, then complete the single successful health check.",
        "Carry all eight original source closures with R1 and R2 separate; create no source input bindings and dispatch no source graders.",
        "Run the unchanged 13-case, 21-operation selector schedule with at most 44 Jev calls inside the registered token bounds.",
        "Generate 16 task/arm answers and two readiness checks using free; freeze all 16 complete answer outputs.",
        "Prepare all 32 answer-assessment packets and verify exact inversions, coverage and complete view delivery under 384000 bytes before any grader dispatch.",
        "Dispatch each of the 32 independent answer assessments once; retain exact raw finals and validate transcripts, citations, coverage and frozen rubrics.",
        "Only after complete valid reference closure compute the unchanged quality and task-use metrics; untouched confirmation remains required before adoption.",
    ]
    if protocol.get("grading_order") != expected_order:
        raise ValueError("current-description-grading-order")

    expected_counts = {
        "source_assessments_carried_unchanged": 48,
        "fresh_source_assessments": 0,
        "fresh_answer_assessments": 32,
        "historical_grader_submissions_total": 72,
        "fresh_grader_submissions_total": 32,
        "planned_grader_submissions_total": 104,
        "planned_all_model_submissions_total": 122,
    }
    for key, value in expected_counts.items():
        if cohorts.get(key) != value:
            raise ValueError("cohort-current-accounting")
    acquisition = cohorts.get("acquisition")
    if type(acquisition) is not dict:
        raise ValueError("cohort-acquisition-shape")
    for key, value in {
        "source_assessments_carried_unchanged": 48,
        "fresh_source_assessments": 0,
        "source_assessments_total_required": 48,
        "fresh_answer_assessments": 32,
        "new_searches": 0,
    }.items():
        if acquisition.get(key) != value:
            raise ValueError("cohort-acquisition-current-accounting")

    acq = protocol.get("acquisition")
    if type(acq) is not dict:
        raise ValueError("protocol-acquisition-shape")
    for key, value in {
        "source_assessments_carried_unchanged": 48,
        "source_assessments_fresh": 0,
        "carried_source_assignments": 48,
        "source_assessments_total_required": 48,
        "fresh_answer_assessments": 32,
        "historical_grader_submissions_total": 72,
        "fresh_grader_submissions_total": 32,
        "planned_grader_submissions_total": 104,
        "planned_answer_api_calls": 18,
        "planned_all_model_submissions_total": 122,
        "maximum_all_model_submissions_total": 130,
        "new_searches": 0,
        "new_source_fetches": 0,
        "retries": 0,
    }.items():
        if acq.get(key) != value:
            raise ValueError("protocol-acquisition-current-accounting")

    models = protocol.get("models")
    if type(models) is not dict:
        raise ValueError("protocol-models-shape")
    for key, value in {
        "historical_grader_submissions_counted": 72,
        "source_assessments_carried_unchanged": 48,
        "fresh_source_assessments": 0,
        "fresh_answer_assessments": 32,
        "planned_fresh_grader_submissions": 32,
        "planned_total_logical_model_submissions_including_historical": 122,
    }.items():
        if models.get(key) != value:
            raise ValueError("protocol-model-current-accounting")
    if (models.get("answerer_alias") != "free"
            or models.get("assessment_model", {}).get("model_id") != "gpt-6.1-sol"):
        raise ValueError("current-model-policy")
    if 72 + 32 != 104 or 104 + 18 != 122 or 122 > 130:
        raise ValueError("current-budget-arithmetic")

    correction = protocol.get("budget_correction", {})
    if (correction.get("per_call_input_tokens") != {"before": 64000, "after": 128000}
            or correction.get("reserve_input_tokens") != {"before": 64000, "after": 128000}
            or correction.get("aggregate_output_tokens") != {"before": 160000, "after": 256000}):
        raise ValueError("input-budget-correction")
    invocation = protocol.get("jev_invocation")
    if (type(invocation) is not dict
            or invocation.get("reported_output_tokens_total_maximum") != 256000
            or invocation.get("per_call_reported_output_tokens_maximum") != 32000
            or invocation.get("reserve_before_each_call", {}).get("output") != 32000):
        raise ValueError("output-budget-correction")
    selector_path = Path(__file__).resolve().parent / "selector-runner.py.txt"
    reference_selector_path = Path(__file__).resolve().parent.parent / "EXP-078" / "selector-runner.py.txt"
    if (selector_path.is_symlink() or reference_selector_path.is_symlink()
            or not selector_path.is_file() or not reference_selector_path.is_file()):
        raise ValueError("selector-budget-provenance")
    selector_raw = selector_path.read_bytes()
    reference_selector_raw = reference_selector_path.read_bytes()
    expected_selector_raw = reference_selector_raw
    for old, new in ((b"RESERVE_INPUT = 64_000", b"RESERVE_INPUT = 128_000"),
                     (b"CALL_INPUT_MAX = 64_000", b"CALL_INPUT_MAX = 128_000"),
                     (b"MAX_OUTPUT_TOTAL = 160_000", b"MAX_OUTPUT_TOTAL = 256_000")):
        if expected_selector_raw.count(old) != 1:
            raise ValueError("selector-budget-reference-shape")
        expected_selector_raw = expected_selector_raw.replace(old, new, 1)
    selector_pin = protocol.get("selector_provenance")
    if (selector_raw != expected_selector_raw or type(selector_pin) is not dict
            or selector_pin.get("original_path") != "docs/experiments/evidence/EXP-078/selector-runner.py.txt"
            or selector_pin.get("original_sha256") != hashlib.sha256(reference_selector_raw).hexdigest()
            or selector_pin.get("byte_diff_restricted_to_three_budget_constants") is not True
            or selector_pin.get("prospective_sha256") != hashlib.sha256(selector_raw).hexdigest()):
        raise ValueError("selector-budget-provenance")
    # Numerical quality gates and task definitions are inherited unchanged.
    # Compare to the frozen EXP-098 files instead of trusting a self-declared
    # snapshot embedded in the prospective protocol.
    reference_root = Path(__file__).resolve().parent.parent / "EXP-098"
    reference_protocol_path = reference_root / "protocol.json"
    reference_cohorts_path = reference_root / "cohorts.json"
    if reference_protocol_path.is_symlink() or reference_cohorts_path.is_symlink():
        raise ValueError("frozen-reference-symlink")
    try:
        reference_protocol = json.loads(reference_protocol_path.read_bytes())
        reference_cohorts = json.loads(reference_cohorts_path.read_bytes())
    except (OSError, ValueError) as exc:
        raise ValueError("frozen-reference-unavailable") from exc
    frozen = ("quality", "candidate", "consumer", "confirmation_scope",
              "confirmation_requires_development_pass")
    if any(protocol.get(key) != reference_protocol.get(key) for key in frozen):
        raise ValueError("frozen-semantics-changed")
    if cohorts.get("cohorts") != reference_cohorts.get("cohorts"):
        raise ValueError("frozen-cohort-changed")

    aliases = protocol.get("source_citation_aliases")
    if (type(source_prompt_raw) is not bytes or type(aliases) is not dict
            or aliases.get("source_prompt_sha256") != hashlib.sha256(source_prompt_raw).hexdigest()):
        raise ValueError("current-source-prompt-binding")
    if protocol.get("external_source_pins") != qualification.get("external_source_pins"):
        raise ValueError("current-external-helper-pins")
    pins = qualification.get("external_source_pins")
    checkpoint = pins.get("source_checkpoint") if type(pins) is dict else None
    if (type(checkpoint) is not dict or set(checkpoint) != {"sha256"}
            or type(checkpoint.get("sha256")) is not str
            or len(checkpoint["sha256"]) != 64
            or any(c not in "0123456789abcdef" for c in checkpoint["sha256"])):
        raise ValueError("source-checkpoint-pin-required")
    if (protocol.get("runtime_readiness_baseline", {}).get("schema")
            != "exp099-readiness-baseline/1"
            or protocol.get("input_provenance", {}).get("schema")
            != "exp099-reused-input-pins/1"):
        raise ValueError("current-readiness-or-input-schema")
    return {"study_id": "EXP-099", "internally_consistent": True,
            "authority_granted": False, "quality_claim": False}
