"""EXP-098 one-process controller for the frozen card checkpoint and exact final-text continuation.

Import is inert. No search, source capture, provider, credential-file, or model
I/O occurs unless the caller explicitly invokes a post-admission operation.
"""
from __future__ import annotations
import hashlib
import hmac
import importlib.machinery
import importlib.util
import json
import os
from pathlib import Path
import re
import subprocess
import sys
from dataclasses import dataclass

HERE = Path(__file__).resolve().parent
REPO = HERE.parents[3]


def load(name: str, filename: str):
    path = HERE / filename
    relative = path.relative_to(REPO).as_posix()
    committed = subprocess.run(["git", "show", f"HEAD:{relative}"], cwd=REPO,
        check=True, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL).stdout
    if path.is_symlink() or path.read_bytes() != committed:
        raise ValueError("uncommitted-controller-dependency")
    if name in sys.modules:
        module = sys.modules[name]
        if Path(module.__file__).resolve() != path.resolve():
            raise ValueError("module-identity-path")
        return module
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_loader(name, loader)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    try:
        loader.exec_module(module)
    except BaseException:
        sys.modules.pop(name, None)
        raise
    return module


def preflight(bundle_root, cli_path=None, *, source_checkpoint_root=None,
              source_checkpoint_pins_path=None, readiness_baseline_path=None):
    """Verify EXP-098 authority, private readiness baseline, and frozen inputs first."""
    if source_checkpoint_pins_path is None:
        raise ValueError("source-checkpoint-manifest-required")
    if source_checkpoint_root is None:
        raise ValueError("source-checkpoint-required")
    admission = load("exp098_admission", "admission.py.txt")
    # Validate the real committed authority before reading any qualification fields.
    # Prospective or legacy registration fails here, before a clock, lease, or
    # credential/configuration read can be attempted.
    _registration, rraw, qraw, q = admission._committed_design()
    source_revision = q.get("qualified_source_revision")
    if type(source_revision) is not str or not re.fullmatch(r"[0-9a-f]{40}", source_revision):
        raise admission.AdmissionError("qualified-source-revision")
    integrity = {"study_id": "EXP-098", "synthetic": False,
        "qualification_sha256": admission.sha(qraw), "registration_sha256": admission.sha(rraw),
        "qualified_source_revision": source_revision,
        "protocol_sha256": admission.sha((HERE / "protocol.json").read_bytes()),
        "cohorts_sha256": admission.sha((HERE / "cohorts.json").read_bytes()),
        "stage_manifest_sha256": q.get("stage_manifest_sha256"),
        "input_pins_sha256": admission.sha((HERE / "reuse-input-pins.json").read_bytes()),
        "input_bundle_sha256": q.get("input_bundle_sha256"),
        "runtime_config_receipt_sha256": q.get("runtime_config_receipt_sha256"),
        "environment": q.get("environment"), "dependencies": q.get("dependencies"),
        "external_source_pins": q.get("external_source_pins")}
    baseline_path = admission._normalized_readiness_baseline_path(readiness_baseline_path)
    admission.verify_closure(integrity, qraw, readiness_baseline_path=baseline_path)
    proof = admission.verify_closure(integrity, qraw, Path(bundle_root), baseline_path)
    if proof.get("readiness_baseline_path") != baseline_path:
        raise admission.AdmissionError("readiness-baseline-path-binding")
    pins = q.get("external_source_pins")
    readiness_pin = pins.get("readiness_baseline") if type(pins) is dict else None
    if (type(readiness_pin) is not dict or set(readiness_pin) != {"sha256"}
            or type(readiness_pin.get("sha256")) is not str
            or not re.fullmatch(r"[0-9a-f]{64}", readiness_pin["sha256"])):
        raise ValueError("readiness-baseline-pin-invalid")
    baseline_module = load("exp098_readiness_baseline", "readiness-baseline.py.txt")
    readiness_baseline = baseline_module.load_baseline(baseline_path, readiness_pin["sha256"])
    cli_pins = integrity.get("external_source_pins")
    pin = cli_pins.get("groktocrawl_cli") if isinstance(cli_pins, dict) else None
    if (not isinstance(pin, dict) or set(pin) != {"sha256", "revision"}
            or not isinstance(pin.get("sha256"), str) or not re.fullmatch(r"[0-9a-f]{64}", pin["sha256"])
            or not isinstance(pin.get("revision"), str) or not re.fullmatch(r"[0-9a-f]{40}", pin["revision"])):
        raise ValueError("qualified-cli-pin-invalid")
    cli = Path(cli_path) if cli_path is not None else Path.home() / ".local/bin/groktocrawl"
    if cli.is_symlink() or not cli.is_file() or hashlib.sha256(cli.read_bytes()).hexdigest() != pin["sha256"]:
        raise ValueError("pinned-cli-preflight-failed")
    carry = load("exp098_source_carry", "source-carry.py.txt")
    carry.verify_source_checkpoint(Path(source_checkpoint_root), Path(source_checkpoint_pins_path), carry.CHECKPOINT_PINS_SHA256)
    return admission, integrity, qraw, proof, readiness_baseline


@dataclass(frozen=True)
class State:
    outdir: Path
    bundle_root: Path
    admission: object
    permit: object
    integrity: dict
    clock: tuple
    effective_acquisition: dict
    input_proof: dict
    readiness_baseline: object
    carried_card_references: dict
    carried_source_references: dict | None = None
    carried_source_payloads: dict | None = None


def begin(outdir, bundle_root, cli_path=None, *, source_checkpoint_root=None,
          source_checkpoint_pins_path=None, readiness_baseline_path=None):
    """Verify, mint one new EXP-098 clock/lease, then rebase only clock metadata."""
    admission, integrity, qraw, proof, readiness_baseline = preflight(bundle_root, cli_path,
        source_checkpoint_root=source_checkpoint_root,
        source_checkpoint_pins_path=source_checkpoint_pins_path, readiness_baseline_path=readiness_baseline_path)
    carry = load("exp098_source_carry", "source-carry.py.txt")
    carried_sources = carry.verify_source_checkpoint(Path(source_checkpoint_root),
        Path(source_checkpoint_pins_path), carry.CHECKPOINT_PINS_SHA256)
    # Verify the exact process configuration before any stage directory, clock,
    # or lease is created. Values remain in this frame until admission succeeds.
    capture_config = _read_bound_capture_configuration(bundle_root, integrity, admission.strict_json)
    output = Path(outdir)
    if output.exists() or output.is_symlink():
        raise ValueError("fresh-stage-output-required")
    output.mkdir(mode=0o700, parents=False, exist_ok=False)
    clock = admission.create_stage_clock(output, integrity["qualified_source_revision"])
    capability = admission.admit_live(output, integrity, qraw, Path(bundle_root),
                                      readiness_baseline_path=proof["readiness_baseline_path"])
    permit = admission.consume_capability(capability)
    _configure_capture(capture_config)
    bridge = load("exp098_input_bridge", "input-bridge.py.txt")
    prepared = bridge.prepare_reused_inputs(Path(bundle_root), stage_clock=clock[0],
        stage_clock_digest=clock[2])
    effective = prepared["effective_acquisition"]
    effective_raw = admission.canonical_json(effective)
    admission._write_once(output / "effective-input-envelope.json", effective_raw)
    proof = _opaque_card_proof(prepared["carried_card_references"])
    admission._write_once(output / "carried-card-proof.json", admission.canonical_json(proof))
    admission._write_once(output / "carried-source-proof.json",
        admission.canonical_json(carried_sources["proof"]))
    return State(output, Path(bundle_root), admission, permit, integrity, clock,
                 effective, prepared["input_integrity"], readiness_baseline, proof,
                 carried_sources["proof"], carried_sources["payloads"])


def _opaque_card_proof(carried: dict) -> dict:
    """Preserve the complete bridge-validated, nonsemantic reuse proof."""
    if (type(carried) is not dict
            or carried.get("schema") != "exp095-carried-card-reference-proof/1"
            or carried.get("source_study_id") != "EXP-088"
            or carried.get("exp084_grader_outputs_used") is not False
            or carried.get("card_assessments_carried_unchanged") is not True
            or carried.get("card_job_count") != 16
            or carried.get("record_count") != 884
            or type(carried.get("records")) is not list
            or len(carried["records"]) != 16):
        raise ValueError("carried-card-checkpoint-invalid")
    path = carried.get("assessment_manifest_path")
    digest = carried.get("assessment_manifest_sha256")
    if (path != "exp088-card-reference/provenance/assessment-manifest.json"
            or type(digest) is not str or not re.fullmatch(r"[0-9a-f]{64}", digest)):
        raise ValueError("carried-card-manifest-binding")
    records = carried["records"]
    row_fields = {"logical_submission", "task_id", "assessor_id", "helper_id", "record_count",
                  "input_path", "input_sha256", "assignment_path", "assignment_sha256",
                  "label_binding_path", "label_binding_sha256", "raw_response_path",
                  "raw_response_sha256", "decoded_response_path", "decoded_response_sha256",
                  "verified_response_path", "verified_response_sha256", "dispatch_receipt_path",
                  "dispatch_receipt_sha256", "validation_receipt_path", "validation_receipt_sha256"}
    if any(type(row) is not dict or set(row) != row_fields or type(row.get("record_count")) is not int
           or row["record_count"] < 1 for row in records):
        raise ValueError("carried-card-record-shape")
    hashes = ("input_sha256", "assignment_sha256", "label_binding_sha256", "raw_response_sha256",
              "decoded_response_sha256", "verified_response_sha256", "dispatch_receipt_sha256",
              "validation_receipt_sha256")
    if any(type(row.get(key)) is not str or not re.fullmatch(r"[0-9a-f]{64}", row[key])
           for row in records for key in hashes):
        raise ValueError("carried-card-record-hash")
    if any(type(row.get(key)) is not str or not row[key]
           for row in records for key in ("logical_submission", "task_id", "assessor_id", "helper_id",
                                          "input_path", "assignment_path", "label_binding_path",
                                          "raw_response_path", "decoded_response_path", "verified_response_path",
                                          "dispatch_receipt_path", "validation_receipt_path")):
        raise ValueError("carried-card-record-path")
    if (len({row["logical_submission"] for row in records}) != 16
            or len({(row["task_id"], row["assessor_id"]) for row in records}) != 16
            or sum(row["record_count"] for row in records) != 884):
        raise ValueError("carried-card-record-count")
    # These are bridge-verified opaque paths and hashes, not card grades or
    # response contents. Preserve the validator's field names for the phase guard.
    return json.loads(json.dumps(carried))


def _read_bound_capture_configuration(bundle_root, integrity, strict_json=json.loads):
    """Read current Hermes URL/key in memory and verify the salted frozen binding."""
    binding_path = Path(bundle_root) / "original" / "configuration-binding.json"
    if binding_path.is_symlink() or not binding_path.is_file():
        raise ValueError("runtime-config-receipt-unavailable")
    binding_raw = binding_path.read_bytes()
    if hashlib.sha256(binding_raw).hexdigest() != integrity.get("runtime_config_receipt_sha256"):
        raise ValueError("runtime-config-receipt-binding")
    binding = strict_json(binding_raw)
    config = {}
    for line in (Path.home() / ".hermes" / ".env").read_text().splitlines():
        line = line.strip().removeprefix("export ")
        key, sep, value = line.partition("=")
        if sep and key in {"GROKTOCRAWL_API_URL", "FIRECRAWL_API_URL",
                           "GROKTOCRAWL_API_KEY", "API_KEY", "FIRECRAWL_API_KEY"}:
            config[key] = value.strip().strip("\"'")
    url = config.get("GROKTOCRAWL_API_URL") or config.get("FIRECRAWL_API_URL")
    key = config.get("GROKTOCRAWL_API_KEY") or config.get("API_KEY") or config.get("FIRECRAWL_API_KEY")
    try:
        nonce = bytes.fromhex(binding["nonce"])
        tag = lambda value: hmac.new(nonce, value.encode(), hashlib.sha256).hexdigest()
        valid = (url and key and hmac.compare_digest(tag(url.rstrip("/")), binding["server_tag"])
                 and hmac.compare_digest(tag(key), binding["credential_tag"]))
    except (KeyError, TypeError, ValueError):
        valid = False
    if not valid:
        raise ValueError("existing-candidate-configuration-changed")
    return url.rstrip("/"), key


def _configure_capture(config):
    """Install verified settings in process memory after fresh admission only."""
    url, key = config
    os.environ["GROKTOCRAWL_API_URL"] = url.rstrip("/")
    os.environ["GROKTOCRAWL_API_KEY"] = key
    os.environ["API_KEY"] = key


def health_check(state: State, *, client_factory=None):
    """Perform at most one post-admission health request and retain sanitized outcome.

    Does not load credentials or persist request URL, response body, headers, or
    exception text. Tests inject only a local fake client.
    """
    import httpx
    clock, clock_raw, clock_sha = state.clock
    phase = state.admission.acquire_phase(state.permit, "health", state.outdir, clock_raw)
    state.admission.require_phase(phase, clock_raw, state.outdir)
    baseline = state.readiness_baseline
    expected_revision = getattr(baseline, "expected_revision", None)
    expected_model = getattr(baseline, "model_alias", None)
    if (type(expected_revision) is not str
            or not re.fullmatch(r"[0-9a-f]{40,64}", expected_revision)
            or expected_model != "free"):
        raise ValueError("readiness-baseline-invalid")
    base = os.environ.get("GROKTOCRAWL_API_URL") or os.environ.get("FIRECRAWL_API_URL")
    factory = client_factory or (lambda **kw: httpx.AsyncClient(**kw))
    attempts = 0

    async def perform():
        nonlocal attempts
        safe_status, model, revision = "incomplete", None, None
        failure_code = None
        if not isinstance(base, str) or not base.startswith(("http://", "https://")):
            failure_code = "endpoint_unavailable"
        else:
            endpoint = base.rstrip("/")
            if endpoint.endswith("/v1"):
                endpoint = endpoint[:-3]
            url = endpoint + "/health"
            try:
                async with factory(timeout=httpx.Timeout(8.0), follow_redirects=False,
                                   trust_env=False, transport=None) as client:
                    state.admission.require_phase(phase, clock_raw, state.outdir)
                    attempts += 1
                    response = await client.get(url)
                    response.raise_for_status()
                    body = response.json()
                    runtime = body.get("runtime") if type(body) is dict else None
                    if type(body) is not dict or body.get("status") != "ok" or type(runtime) is not dict:
                        failure_code = "health_status_invalid"
                    else:
                        candidate_model, candidate_revision = runtime.get("model"), runtime.get("revision")
                        if (candidate_model != expected_model
                                or type(candidate_revision) is not str
                                or not re.fullmatch(r"[0-9a-f]{40,64}", candidate_revision)):
                            failure_code = "health_identity_invalid"
                        elif candidate_revision != expected_revision:
                            failure_code = "health_revision_mismatch"
                        else:
                            safe_status, model, revision = "healthy", candidate_model, candidate_revision
            except BaseException as exc:
                if isinstance(exc, (KeyboardInterrupt, SystemExit)):
                    raise
                failure_code = "health_exchange_failed" if attempts else "health_client_failed"
        receipt = {"schema": "exp098-health-receipt/1", "study_id": "EXP-098",
            "status": safe_status, "model_identity": model, "revision": revision,
            "stage_clock_sha256": clock_sha,
            "readiness_baseline_sha256": getattr(baseline, "sha256", None),
            "physical_http_calls": attempts,
            "failure_code": failure_code, "quality_credit": False,
            "endpoint_and_response_body_retained": False}
        state.admission._write_once(state.outdir / "health.json",
                                    state.admission.canonical_json(receipt))
        if safe_status != "healthy":
            raise ValueError(failure_code or "health_check_failed")
        state.admission.complete_health_phase(phase, clock_raw, state.outdir)
        return receipt
    return perform()
