    def test_pinned_receipt_and_raw_history_tampering_rejected(self):
        with tempfile.TemporaryDirectory() as td:
            copied=pathlib.Path(td)/"EXP-043"; shutil.copytree(R.EXP043,copied)
            plan=R.load_json(copied/"prepared-manifest.json")
            verified=R.validate_published_result()
            receipt_path=copied/"receipts/q1-d.json"; blob_path=copied/"receipts/q1-d.response.bin"
            old_receipt=receipt_path.read_bytes(); old_blob=blob_path.read_bytes()
            rec=R.load_json(receipt_path); raw=R.strict_json(old_blob); qid=next(iter(raw["answers"]))
            raw["answers"][qid]["score"]+=0.25
            altered=json.dumps(raw,separators=(",",":"),ensure_ascii=False).encode()
            rec["answers"]=raw["answers"]; rec["response_bytes"]=len(altered); rec["response_sha256"]=R.sha(altered)
            blob_path.write_bytes(altered); receipt_path.write_bytes(R.pretty(rec))
            receipts,errors=R.BASE.read_receipts(copied,plan,True)
            self.assertEqual(errors,[]); self.assertEqual(len(receipts),65)
            with self.assertRaises(ValueError): R.BASE.read_history(copied,plan,verified["check"]["reference_hashes"])
            # Changing only the structured receipt cannot silently pass raw-response validation.
            blob_path.write_bytes(old_blob); receipt_path.write_bytes(old_receipt)
            rec=R.load_json(receipt_path); rec["answers"][qid]["score"]+=0.25; receipt_path.write_bytes(R.pretty(rec))
            _,errors=R.BASE.read_receipts(copied,plan,True)
            self.assertTrue(errors)
            receipt_path.write_bytes(old_receipt)
            # Raw-only tampering must also fail receipt parsing/binding.
            altered_raw=json.loads(old_blob); altered_raw["answers"][qid]["score"]+=0.25
            blob_path.write_bytes(json.dumps(altered_raw,separators=(",",":"),ensure_ascii=False).encode())
            _,errors=R.BASE.read_receipts(copied,plan,True)
            self.assertTrue(errors)
            blob_path.write_bytes(old_blob)
            # The immutable receipt/raw pair still fails if the append-only completion ledger changes.
            attempts_path=copied/"attempts.jsonl"; old_attempts=attempts_path.read_bytes()
            events=[json.loads(line) for line in old_attempts.splitlines()]
            finished=next(i for i,event in enumerate(events) if event.get("event")=="attempt_finished" and event.get("request_id")=="q1-d")
            events[finished]["response_sha256"]="0"*64
            attempts_path.write_bytes(b"\n".join(json.dumps(event,separators=(",",":"),ensure_ascii=False).encode() for event in events)+b"\n")
            with self.assertRaises(ValueError): R.BASE.read_history(copied,plan,verified["check"]["reference_hashes"])
            attempts_path.write_bytes(old_attempts)
            begun,done,_=R.BASE.read_history(copied,plan,verified["check"]["reference_hashes"])
            self.assertEqual(begun,done); self.assertEqual(len(done),65)

    def test_fake_65_receipts_reach_real_analysis_both_refs_and_all_primary_deltas(self):
        receipts={}
        for spec in self.specs:
            if spec["kind"] in {"W0","D"}:
                answers={qid:{"type":"score","score":4.25} for qid in spec["question_ids"]}
            else:
                answers={qid:{"type":"noul","noul":0.0} for qid in spec["question_ids"]}
            response={"model":R.MODEL,"answers":answers,"usage":{"input_tokens":100,"output_tokens":len(answers)},"_elapsed_ms":1.0}
            scores,nouls,errors=R.BASE.validate_answer(response,spec)
            self.assertEqual(errors,[],spec["id"])
            receipts[spec["id"]]={"id":spec["id"],"status":"valid","scores":scores,"nouls":nouls,"usage":response["usage"],"remote_http_ms":1.0}
        self.assertEqual(len(receipts),65)
        ranks,diagnostics,nav,facet_ids=R.build_rankings(self.data,self.grouped,self.specs,receipts)
        ids_by_pool={p:[c["id"] for c in self.grouped[p]] for p in (*R.SOFTWARE,*R.EXTENDED)}
        # Exercise the exact pinned reference adapter used by compute_report, then run the true analysis helper on fake receipts/rankings.
        refs,refhashes=R.analysis_references(self.data,self.grouped)
        self.assertEqual(set(refs),{"A","B"}); self.assertEqual(set(refs["A"]),set((*R.SOFTWARE,*R.NATURAL)))
        self.assertEqual(set(refhashes),{"EXP039","EXP040"})
        neutral_refs={ref:{pool:{cid:{"lead":0,"facets":[],"uncertain":False} for cid in ids_by_pool[pool]} for pool in ids_by_pool} for ref in ("A","B")}
        neutral_report=R.analyze_rankings(ranks,neutral_refs,facet_ids)
        self.assertFalse(neutral_report["development_screen_passed"],"neutral synthetic labels cannot qualify selection")
        report=R.analyze_rankings(ranks,refs,facet_ids)
        self.assertEqual(set(report["primary"]),{"A","B"})
        for ref in ("A","B"):
            self.assertEqual(set(report["primary"][ref]["paired_deltas"]),set(R.PRIMARY))
        self.assertEqual(set(report["membership"]["by_pool"]),set((*R.SOFTWARE,*R.EXTENDED)))
        self.assertTrue(report["membership"]["passed"])
        self.assertFalse(report["development_screen_passed"],"neutral synthetic references must fail selection")
        self.assertEqual({p:len(rows) for p,rows in ids_by_pool.items()}["constructed80"],79)
        self.assertEqual(set(diagnostics),set((*R.SOFTWARE,*R.EXTENDED)))
        self.assertEqual(len(nav),5); self.assertTrue(all(row["facet_policy_bypassed"] for row in nav))
