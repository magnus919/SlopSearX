def norm_url(url: str) -> tuple[str, str]:
    parsed = urlparse(url)
    host = (parsed.hostname or "").lower()
    if host.startswith("www."):
        host = host[4:]
    return host, parsed.path.rstrip("/") or "/"


def is_target(url: str, host: str, mode: str, expected: str) -> bool:
    actual_host, actual = norm_url(url)
    expected = expected.rstrip("/") or "/"
    if actual_host != host:
        return False
    if mode == "exact":
        return actual == expected
    if mode == "prefix":
        return actual == expected or actual.startswith(expected + "/")
    return actual == expected or actual.endswith(expected)


def rr(results: list[Any], host: str, mode: str, path: str) -> tuple[float, int | None]:
    for rank, result in enumerate(results[:20], 1):
        if is_target(result.url, host, mode, path):
            return 1 / rank, rank
    return 0.0, None


def p95(values: list[float]) -> float:
    ordered = sorted(values)
    return ordered[max(0, math.ceil(0.95 * len(ordered)) - 1)]


def confusion(rows: list[dict[str, Any]]) -> dict[str, int]:
    return {
        name: sum(1 for row in rows if row["label"] == label and row["prediction"] == pred)
        for name, label, pred in (("tp", 1, 1), ("tn", 0, 0), ("fp", 0, 1), ("fn", 1, 0))
    }


async def call_jev(client: httpx.AsyncClient, key: str, state: Any, questions: dict[str, Any]) -> dict[str, Any]:
    started = time.monotonic()
    status: int | None = None
    body: dict[str, Any] = {}
    error: str | None = None
    try:
        response = await client.post(
            "https://api.typesafe.ai/v1/systemone",
            headers={"Authorization": f"Bearer {key}"},
            json={"model": MODEL, "state": state, "questions": questions},
        )
        status = response.status_code
        body = response.json()
    except Exception as exc:
        error = f"{type(exc).__name__}: {exc}"
    latency = (time.monotonic() - started) * 1000
    answers = body.get("answers", {}) if isinstance(body, dict) else {}
    probs: dict[str, float] = {}
    for name in questions:
        answer = answers.get(name, {}) if isinstance(answers, dict) else {}
        value = answer.get("noul") if isinstance(answer, dict) else None
        if isinstance(value, (int, float)) and math.isfinite(value) and 0 <= value <= 1:
            probs[name] = float(value)
    usage = body.get("usage", {}) if isinstance(body, dict) else {}
    valid = status == 200 and body.get("model") == MODEL and len(probs) == len(questions) and isinstance(usage, dict) and "input_tokens" in usage
    return {"valid": valid, "status": status, "model": body.get("model"), "probabilities": probs, "usage": usage if isinstance(usage, dict) else {}, "latency_ms": latency, "error": error}


def finish(exp: str, rows: list[dict[str, Any]], metrics: dict[str, Any], calls: list[dict[str, Any]], output: Path, registration: str) -> None:
    input_tokens = sum(int(call["usage"].get("input_tokens", 0) or 0) for call in calls)
    output_tokens = sum(int(call["usage"].get("output_tokens", 0) or 0) for call in calls)
    summary = {"experiment": exp, "registration_commit": registration, "model": MODEL, **metrics, "latency_ms": {"mean": statistics.fmean(call["latency_ms"] for call in calls), "p95_nearest_rank": p95([call["latency_ms"] for call in calls]), "max": max(call["latency_ms"] for call in calls)}, "usage": {"input_tokens": input_tokens, "output_tokens": output_tokens, "estimated_usd": input_tokens / 1_000_000 * PRICE_PER_MILLION_INPUT}}
    output.mkdir(parents=True, exist_ok=True)
    (output / "rows.json").write_text(json.dumps(rows, indent=2, sort_keys=True) + "\n")
    (output / "summary.json").write_text(json.dumps(summary, indent=2, sort_keys=True) + "\n")
    print(json.dumps(summary, indent=2, sort_keys=True))

    rows, calls = [], []
    for cid, query, host, mode, path in HARD_CASES:
        acquired = await engine.search(query, {"categories": ["general"]})
        candidates = acquired.results[:20]
        questions = {f"r{i:03d}": {"type": "noul", "instructions": f"Is candidate {i} a direct, useful answer for this query, preferring the authoritative primary source over adjacent, generic, commercial, or discussion pages?"} for i in range(len(candidates))}
        state = {"query": query, "candidates": [{"index": i, "title": r.title, "url": "".join(norm_url(r.url)), "snippet": (r.content or "")[:500]} for i, r in enumerate(candidates)]}
        call = await call_jev(client, key, state, questions)
        calls.append(call)
        probs = [call["probabilities"].get(f"r{i:03d}", float("-inf")) for i in range(len(candidates))]
        fused = []
        jev = []
        scores: list[float | None] = [None] * len(candidates)
        if call["valid"] and candidates:
            indexed = list(enumerate(candidates))
            jev = [x[1] for x in sorted(indexed, key=lambda x: (-probs[x[0]], x[0], norm_url(x[1].url)))]
            scores = [0.70 / (i + 1) + 0.30 * probs[i] for i in range(len(candidates))]
            fused = [x[1] for x in sorted(indexed, key=lambda x: (-float(scores[x[0]]), x[0], norm_url(x[1].url)))]
        base_rr, base_rank = rr(candidates, host, mode, path)
        jev_rr, jev_rank = rr(jev, host, mode, path) if call["valid"] else (0.0, None)
        fusion_rr, fusion_rank = rr(fused, host, mode, path) if call["valid"] else (0.0, None)
        rows.append({"id": cid, "query": query, "target": {"host": host, "mode": mode, "path": path}, "target_present": base_rank is not None, "brave": {"status": acquired.status.value, "result_count": len(candidates), "latency_ms": acquired.latency_ms, "rank": base_rank, "rr": base_rr}, "jev": {k: call[k] for k in ("valid", "status", "model", "usage", "latency_ms", "error")} | {"rank": jev_rank, "rr": jev_rr}, "fusion": {"rank": fusion_rank, "rr": fusion_rr}, "candidates": [{"brave_rank": i + 1, "url": r.url, "title": r.title, "snippet_chars": len(r.content or ""), "snippet_sha256": hashlib.sha256((r.content or "").encode()).hexdigest(), "jev_probability": probs[i] if call["valid"] else None, "fusion_score": scores[i]} for i, r in enumerate(candidates)]})
        print(cid, base_rank, jev_rank, fusion_rank, flush=True)
    await engine.shutdown()
    coverage = sum(row["target_present"] for row in rows)
    valid = sum(row["jev"]["valid"] for row in rows)
    base_mrr = statistics.fmean(row["brave"]["rr"] for row in rows)
    fusion_mrr = statistics.fmean(row["fusion"]["rr"] for row in rows)
    jev_mrr = statistics.fmean(row["jev"]["rr"] for row in rows)
    drops = [row["fusion"]["rank"] - row["brave"]["rank"] for row in rows if row["target_present"] and row["jev"]["valid"] and row["fusion"]["rank"] > row["brave"]["rank"]]
    prelim_cost = sum(int(c["usage"].get("input_tokens", 0) or 0) for c in calls) / 1_000_000 * PRICE_PER_MILLION_INPUT
    prerequisites = coverage >= 8 and valid >= 11
    supported = prerequisites and fusion_mrr - base_mrr >= 0.05 and len(drops) <= 2 and max(drops, default=0) <= 3 and p95([c["latency_ms"] for c in calls]) <= 1500 and prelim_cost < 0.01
    finish("EXP-006", rows, {"target_present_count": coverage, "valid_calls": valid, "brave_mrr_at_20": base_mrr, "jev_mrr_at_20_diagnostic": jev_mrr, "fusion_mrr_at_20": fusion_mrr, "absolute_gain": fusion_mrr - base_mrr, "regressions": len(drops), "max_rank_drop": max(drops, default=0), "decision": "supported-for-further-research" if supported else ("not-supported" if prerequisites else "inconclusive")}, calls, output, registration)


async def exp007(client: httpx.AsyncClient, key: str, output: Path, registration: str) -> None:
    rows, calls = [], []
    for case in ROBUSTNESS_CASES:

        calls.append(call)
        prob = call["probabilities"].get("relevant")
        pred = int(prob >= 0.5) if prob is not None else None
        rows.append({**case, "probability": prob, "prediction": pred, "call": {k: call[k] for k in ("valid", "status", "model", "usage", "latency_ms", "error")}})
    cm = confusion(rows)
    tpr = cm["tp"] / (cm["tp"] + cm["fn"])
    tnr = cm["tn"] / (cm["tn"] + cm["fp"])
    adversarial = [r for r in rows if "adversarial" in r["tags"] and r["label"] == 0]
    adv_fpr = sum(r["prediction"] == 1 for r in adversarial) / len(adversarial)
    brier = statistics.fmean((r["probability"] - r["label"]) ** 2 for r in rows if r["probability"] is not None)
    valid = sum(r["call"]["valid"] for r in rows)
    cost = sum(int(c["usage"].get("input_tokens", 0) or 0) for c in calls) / 1_000_000 * PRICE_PER_MILLION_INPUT
    supported = valid == 16 and (tpr + tnr) / 2 >= 0.8 and adv_fpr <= 0.25 and 1 - tpr <= 0.25 and p95([c["latency_ms"] for c in calls]) <= 1500 and cost < 0.01
    finish("EXP-007", rows, {"valid_calls": valid, "confusion": cm, "balanced_accuracy": (tpr + tnr) / 2, "adversarial_false_positive_rate": adv_fpr, "positive_false_negative_rate": 1 - tpr, "brier_score_diagnostic": brier, "decision": "supported-for-further-research" if supported else ("not-supported" if valid == 16 else "inconclusive")}, calls, output, registration)

    supported = valid == 16 and cm["fp"] == 0 and recall >= 0.75 and accuracy >= 0.8 and p95([c["latency_ms"] for c in calls]) <= 1500 and cost < 0.01
    finish("EXP-008", rows, {"valid_calls": valid, "confusion": cm, "false_stops": cm["fp"], "sufficient_recall": recall, "accuracy": accuracy, "decision": "supported-for-further-research" if supported else ("not-supported" if valid == 16 else "inconclusive")}, calls, output, registration)


async def exp009(client: httpx.AsyncClient, key: str, output: Path, registration: str) -> None:
    from slopsearx.router import QueryRouter
    router = QueryRouter()
    rows, calls = [], []
    for cid, query, label in ROUTING_CASES:
        questions = {name: {"type": "noul", "instructions": f"Is the {name} plan the single best source family for this query, compared with the other offered plans?"} for name in PLANS}
        call = await call_jev(client, key, {"query": query, "plans": PLANS}, questions)
        calls.append(call)
        selected = max(PLANS, key=lambda name: (call["probabilities"].get(name, float("-inf")), -list(PLANS).index(name))) if call["valid"] else None
        baseline = router.match_topic(query) or "web"
        rows.append({"id": cid, "query": query, "label": label, "baseline": baseline, "selected": selected, "probabilities": call["probabilities"], "call": {k: call[k] for k in ("valid", "status", "model", "usage", "latency_ms", "error")}})
    valid = sum(r["call"]["valid"] for r in rows)
    accuracy = sum(r["selected"] == r["label"] for r in rows) / len(rows)
    baseline_accuracy = sum(r["baseline"] == r["label"] for r in rows) / len(rows)
    outside = sum(r["selected"] not in PLANS for r in rows)
    cost = sum(int(c["usage"].get("input_tokens", 0) or 0) for c in calls) / 1_000_000 * PRICE_PER_MILLION_INPUT
    supported = valid == 15 and accuracy >= 0.8 and accuracy - baseline_accuracy >= 0.1 and outside == 0 and p95([c["latency_ms"] for c in calls]) <= 1500 and cost < 0.01
    finish("EXP-009", rows, {"valid_calls": valid, "accuracy": accuracy, "baseline_accuracy": baseline_accuracy, "absolute_accuracy_gain": accuracy - baseline_accuracy, "outside_plan_outputs": outside, "decision": "supported-for-further-research" if supported else ("not-supported" if valid == 15 else "inconclusive")}, calls, output, registration)


def macro_f1(rows: list[dict[str, Any]]) -> float:
    scores = []
    for label in ANNOTATION_LABELS:
        tp = sum(r["label"] == label and r["prediction"] == label for r in rows)
        fp = sum(r["label"] != label and r["prediction"] == label for r in rows)
        fn = sum(r["label"] == label and r["prediction"] != label for r in rows)
        scores.append(2 * tp / (2 * tp + fp + fn) if 2 * tp + fp + fn else 0.0)
    return statistics.fmean(scores)


async def exp010(client: httpx.AsyncClient, key: str, output: Path, registration: str) -> None:
    rows, calls = [], []
    for cid, query, title, url, snippet, label in ANNOTATION_CASES:
        questions = {name: {"type": "noul", "instructions": f"Does this result primarily belong to the source-type class '{name}'? Choose based on publisher, URL, title, and snippet; do not treat relevance as source type."} for name in ANNOTATION_LABELS}
        call = await call_jev(client, key, {"query": query, "result": {"title": title, "url": url, "snippet": snippet}, "classes": ANNOTATION_LABELS}, questions)
        calls.append(call)
        ordered = sorted(ANNOTATION_LABELS, key=lambda name: (-call["probabilities"].get(name, float("-inf")), ANNOTATION_LABELS.index(name)))
        top = call["probabilities"].get(ordered[0]) if call["valid"] else None
        second = call["probabilities"].get(ordered[1]) if call["valid"] else None
        prediction = ordered[0] if top is not None and second is not None and top >= 0.5 and top - second >= 0.1 else None
        rows.append({"id": cid, "query": query, "title": title, "url": url, "snippet": snippet, "label": label, "prediction": prediction, "top_probability": top, "margin": top - second if top is not None and second is not None else None, "probabilities": call["probabilities"], "call": {k: call[k] for k in ("valid", "status", "model", "usage", "latency_ms", "error")}})
    valid = sum(r["call"]["valid"] for r in rows)
    coverage = sum(r["prediction"] is not None for r in rows) / len(rows)
    accuracy = sum(r["prediction"] == r["label"] for r in rows) / len(rows)
    mf1 = macro_f1(rows)
    cost = sum(int(c["usage"].get("input_tokens", 0) or 0) for c in calls) / 1_000_000 * PRICE_PER_MILLION_INPUT
    supported = valid == 18 and coverage >= 0.75 and accuracy >= 0.8 and mf1 >= 0.75 and p95([c["latency_ms"] for c in calls]) <= 1500 and cost < 0.01
    finish("EXP-010", rows, {"valid_calls": valid, "coverage": coverage, "accuracy": accuracy, "macro_f1": mf1, "decision": "supported-for-further-research" if supported else ("not-supported" if valid == 18 else "inconclusive")}, calls, output, registration)


async def run(args: argparse.Namespace) -> None:
    sys.path.insert(0, str(Path(args.repo).resolve()))
    key = os.environ.get("TYPESAFE_API_KEY", "")
    if not key:
        raise SystemExit("TYPESAFE_API_KEY is required")
    brave_key = os.environ.get("ENGINE_BRAVE_API_KEY", "")
    if args.experiment == "EXP-006" and not brave_key:
        raise SystemExit("ENGINE_BRAVE_API_KEY is required for EXP-006")
    output = Path(args.output)
    async with httpx.AsyncClient(timeout=30.0) as client:
        if args.experiment == "EXP-006":
            await exp006(client, key, brave_key, output, args.registration_commit)
        elif args.experiment == "EXP-007":
            await exp007(client, key, output, args.registration_commit)
        elif args.experiment == "EXP-008":
            await exp008(client, key, output, args.registration_commit)
        elif args.experiment == "EXP-009":
            await exp009(client, key, output, args.registration_commit)
        else:
            await exp010(client, key, output, args.registration_commit)


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("experiment", choices=["EXP-006", "EXP-007", "EXP-008", "EXP-009", "EXP-010"])
    parser.add_argument("--repo", default=".")
    parser.add_argument("--output", required=True)
    parser.add_argument("--registration-commit", required=True)
    asyncio.run(run(parser.parse_args()))


if __name__ == "__main__":
    main()
