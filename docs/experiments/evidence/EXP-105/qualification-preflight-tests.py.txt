def load(name,path):
    loader=importlib.machinery.SourceFileLoader(name,str(path))
    spec=importlib.util.spec_from_loader(name,loader)
    module=importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    return module

def canonical(value):
    return (json.dumps(value,sort_keys=True,separators=(",",":"),ensure_ascii=False,
                        allow_nan=False)+"\n").encode()

def sha(raw):return hashlib.sha256(raw).hexdigest()

class QualificationPreflightTests(unittest.TestCase):
    def setUp(self):self.fx=SyntheticGitFixture()
    def tearDown(self):self.fx.close()

    def test_review_only_source_verification_is_unqualified_and_never_authorizes_execution(self):
        result=self.fx.verify(review_only=True)
        self.assertEqual(result["status"],"verified-unqualified")
        self.assertIs(result["fresh_execution_authorized"],False)
        self.assertEqual(result["study_id"],"EXP-105")
        self.assertTrue(self.fx.capture.is_file(),"explicit prior qualification delegate should run")
        self.assertEqual(json.loads(self.fx.capture.read_bytes()),self.fx.kwargs()["previous_qualification_kwargs"])

    def test_default_flags_false_reject_before_registration_private_pins_or_delegate(self):
        with self.assertRaisesRegex(ValueError,"qualification-not-approved"):
            self.fx.verify(continuation_entrypoint=self.fx.repo/"does-not-exist",
                reader_entrypoint=self.fx.repo/"does-not-exist-reader",
                accepted_snapshot_pins_path=self.fx.repo/"does-not-exist-snapshot")
        self.assertFalse(self.fx.capture.exists())

    def test_flags_true_only_verify_source_seal_and_still_never_claim_execution_authority(self):
        self.fx.close();self.fx=SyntheticGitFixture(qualified=True)
        result=self.fx.verify()
        self.assertEqual(result["status"],"qualified")
        self.assertIs(result["fresh_execution_authorized"],False)

    def test_registration_inventory_must_equal_qualified_source_inventory(self):
        qpath=self.fx.exp105/"qualification.json"
        q=json.loads(qpath.read_bytes())
        q["source_files"].pop("docs/experiments/evidence/EXP-105/protocol.json")
        qpath.write_bytes(canonical(q));self.fx.commit("synthetic trimmed q inventory")
        with self.assertRaisesRegex(ValueError,"qualification-source-inventory"):
            self.fx.verify(review_only=True)

    def test_parent_source_completeness_and_registration_bytes_are_pinned(self):
        self.fx.close();self.fx=SyntheticGitFixture(trim_parent_source=True)
        with self.assertRaisesRegex(ValueError,"qualification-reused-source-changed"):
            self.fx.verify(review_only=True)

        self.fx.close();self.fx=SyntheticGitFixture()
        reg=self.fx.exp105/"registration-integrity.json"
        reg.write_bytes(reg.read_bytes()+b" ")
        with self.assertRaisesRegex(ValueError,"qualification-source-registration-changed"):
            self.fx.verify(review_only=True)

        self.fx.close();self.fx=SyntheticGitFixture()
        files=self.fx.q["source_files"]
        files.pop("docs/experiments/evidence/EXP-104/inherited-only.py")
        regpath=self.fx.exp105/"registration-integrity.json"
        reg=json.loads(regpath.read_bytes());reg["source_files"]=files
        regpath.write_bytes(canonical(reg))
        self.fx.commit("synthetic trimmed carried source inventory")
        self.fx.source_revision=self.fx.git("rev-parse","HEAD").strip()
        qpath=self.fx.exp105/"qualification.json";q=json.loads(qpath.read_bytes());q["source_files"]=files
        q["qualified_source_revision"]=self.fx.source_revision
        qpath.write_bytes(canonical(q));self.fx.commit("synthetic removed EXP-104 inherited source")
        with self.assertRaisesRegex(ValueError,"qualification-exp104-source-changed"):
            self.fx.verify(review_only=True)

    def test_q_must_be_committed_and_source_bytes_must_match_revision_head_and_disk(self):
        qpath=self.fx.exp105/"qualification.json"
        qpath.write_bytes(qpath.read_bytes()+b" ")
        with self.assertRaisesRegex(ValueError,"qualification-uncommitted"):
            self.fx.verify(review_only=True)
        self.fx.close();self.fx=SyntheticGitFixture()
        source=self.fx.exp105/"reader-transport.py.txt"
        source.write_bytes(source.read_bytes()+b"local uncommitted drift")
        with self.assertRaisesRegex(ValueError,"qualification-source-changed"):
            self.fx.verify(review_only=True)

        self.fx.close();self.fx=SyntheticGitFixture()
        source=self.fx.exp105/"reader-transport.py.txt"
        source.write_bytes(source.read_bytes()+b"committed post-registration drift")
        self.fx.commit("synthetic changed HEAD source")
        with self.assertRaisesRegex(ValueError,"qualification-source-changed"):
            self.fx.verify(review_only=True)

    def test_private_scheduler_reader_and_snapshot_pins_must_match(self):
        for name in ("scheduler","reader","snapshot","terminal_snapshot"):
            with self.subTest(pin=name):
                self.fx.close();self.fx=SyntheticGitFixture()
                path=getattr(self.fx,name);path.write_bytes(path.read_bytes()+b"drift")
                with self.assertRaisesRegex(ValueError,"qualification-private-entrypoint-or-carry-changed"):
                    self.fx.verify(review_only=True)

    def test_prior_qualification_is_called_with_explicit_kwargs(self):
        result=self.fx.verify(review_only=True)
        self.assertEqual(result["status"],"verified-unqualified")
        self.assertEqual(json.loads(self.fx.capture.read_bytes()),{
            "fixture_token":"explicit-old-kwargs",
            "previous_failed_run_files":{"only":"synthetic"}})

    def test_protocol_science_and_execution_drift_fail_after_source_is_sealed(self):
        def science(p):p["immutable_science"]["answer_assessment_packets"]["count"]=31
        self.fx.close();self.fx=SyntheticGitFixture(protocol_mutation=science)
        with self.assertRaisesRegex(ValueError,"continuation-science-or-transport-drift"):
            self.fx.verify(review_only=True)
        def budget(p):p["execution"]["maximum_grader_charges"]+=1
        self.fx.close();self.fx=SyntheticGitFixture(protocol_mutation=budget)
        with self.assertRaisesRegex(ValueError,"continuation-execution-budget-drift"):
            self.fx.verify(review_only=True)

if __name__=="__main__":unittest.main(verbosity=2)
