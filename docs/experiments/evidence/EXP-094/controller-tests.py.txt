def load(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_loader(name, loader)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    try:
        loader.exec_module(module)
    except BaseException:
        sys.modules.pop(name, None)
        raise
    return module

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

class OpaqueCardProofTests(unittest.TestCase):
    def test_preserves_complete_bridge_guard_schema_without_grade_payloads(self):
        supplied = valid_proof()
        result = C._opaque_card_proof(supplied)
        for field in ("schema", "source_study_id", "exp084_grader_outputs_used",
                      "card_assessments_carried_unchanged", "card_job_count", "record_count",
                      "assessment_manifest_path", "assessment_manifest_sha256", "records"):
            self.assertEqual(result[field], supplied[field])
        self.assertNotIn("verified_response", result)
        self.assertNotIn("grades", result)
        self.assertEqual((result["card_job_count"], result["record_count"]), (16, 884))

    def test_rejects_changed_lineage_coverage_hashes_and_inventory(self):
        for field, bad in (("schema", "wrong"), ("source_study_id", "EXP-084"),
                           ("exp084_grader_outputs_used", True),
                           ("card_assessments_carried_unchanged", False),
                           ("card_job_count", 15), ("record_count", 883)):
            with self.subTest(field=field):
                proof = valid_proof(); proof[field] = bad
                with self.assertRaises(ValueError): C._opaque_card_proof(proof)
        proof = valid_proof(); proof["records"][0]["input_sha256"] = "not-a-hash"
        with self.assertRaisesRegex(ValueError, "carried-card-record-hash"):
            C._opaque_card_proof(proof)
        proof = valid_proof(); proof["records"].pop()
        with self.assertRaisesRegex(ValueError, "carried-card-checkpoint-invalid"):
            C._opaque_card_proof(proof)
