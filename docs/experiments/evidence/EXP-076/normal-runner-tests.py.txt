 def test_credential_echo_split_across_response_chunks_is_suppressed(self):
  stats={'clients':0,'requests':0,'closed':0};marker=b'Bearer offline-fixture-non-secret'
  class Split(N.httpx.AsyncByteStream):
   async def __aiter__(self):
    yield b'prefix-Bearer offline-fixture-';yield b'non-secret-suffix'
   async def aclose(self):pass
  def factory(**kwargs):
   stats['clients']+=1
   class Counted(N.httpx.AsyncClient):
    async def aclose(self):
     if not getattr(self,'_closed_once',False):self._closed_once=True;stats['closed']+=1
     await super().aclose()
   async def handler(req):
    stats['requests']+=1;body=P.strict_json(req.content)
    if any(k.startswith('query:') for k in body['questions']):return N.httpx.Response(200,stream=Split())
    return N.httpx.Response(200,content=P.response_fixture(body))
   return Counted(transport=N.httpx.MockTransport(handler),**kwargs)
  with tempfile.TemporaryDirectory() as tmp:
   result=asyncio.run(N.run(self.acq,self.queries,pathlib.Path(tmp)/'out',client_factory=factory));rec=result['neutral']['calls'][-1]
   self.assertEqual(result['status'],'incomplete');self.assertEqual(stats['requests'],2);self.assertEqual(stats['clients'],stats['closed'])
   self.assertIsNone(rec['raw_response_hex']);self.assertNotIn(marker.decode(),json.dumps(rec));self.assertTrue(result['budget']['unknown_usage'])
 def test_cancellation_retains_partial_response_and_stops_unknown_usage(self):
  stats={'clients':0,'requests':0,'closed':0};entered=asyncio.Event();block=asyncio.Event()
  class Partial(N.httpx.AsyncByteStream):
   async def __aiter__(self):
    yield b'{"partial":';entered.set();await block.wait();yield b'null}'
   async def aclose(self):pass
  def factory(**kwargs):
   stats['clients']+=1
   class Counted(N.httpx.AsyncClient):
    async def aclose(self):
     if not getattr(self,'_closed_once',False):self._closed_once=True;stats['closed']+=1
     await super().aclose()
   async def handler(req):
    stats['requests']+=1;body=P.strict_json(req.content)
    if any(k.startswith('query:') for k in body['questions']):return N.httpx.Response(200,stream=Partial())
    return N.httpx.Response(200,content=P.response_fixture(body))
   return Counted(transport=N.httpx.MockTransport(handler),**kwargs)
  async def scenario(path):
   task=asyncio.create_task(N.run(self.acq,self.queries,path,client_factory=factory))
   await asyncio.wait_for(entered.wait(),timeout=2);task.cancel();return await task
  with tempfile.TemporaryDirectory() as tmp:
   result=asyncio.run(scenario(pathlib.Path(tmp)/'out'));rec=result['neutral']['calls'][-1]
   self.assertEqual(result['status'],'incomplete');self.assertEqual(stats['requests'],2);self.assertEqual(stats['clients'],stats['closed'])
   self.assertTrue(rec['dispatched']);self.assertEqual(rec['usage_status'],'unknown');self.assertTrue(result['budget']['unknown_usage'])
   self.assertEqual(bytes.fromhex(rec['raw_response_hex']),b'{"partial":')
