 def test_cancellation_retains_partial_response_and_stops_unknown_usage(self):

 def test_gate_validates_then_burns_temporary_lease_before_dns_and_key_sentinels(self):
  qpath=HERE/'qualification-integrity.json';owned={'prepare.py.txt','normal-runner.py.txt','normal-runner-tests.py.txt'}
  source=set(N.SOURCE_FILES);files={n:P.digest((HERE/n).read_bytes()) for n in owned}
  pins={n:P.digest((P.REPO/n).read_bytes()) for n in source}
  proof={'input_pins':dict(P.PINS),'input_revision':N.REGISTRATION_REVISION,
         'protocol_sha256':P.digest((HERE/'protocol.json').read_bytes()),
         'registration_sha256':P.digest((HERE/'registration-integrity.json').read_bytes())}
  q={'schema':'exp076-qualification-integrity/1','qualified':True,'qualified_source_revision':'a'*40,
     'files':files,'source_pins':pins,'dependencies':{n:__import__('importlib').metadata.version(n) for n in N.DEPENDENCIES},
     'environment':{'python_implementation':N.platform.python_implementation(),'python_version':N.platform.python_version()},
     'protocol_sha256':proof['protocol_sha256'],'registration_sha256':proof['registration_sha256']}
  qraw=P.wire(q);original_file=pathlib.Path.is_file;original_read=pathlib.Path.read_bytes
  def is_file(path):return True if path==qpath else original_file(path)
  def read_bytes(path):return qraw if path==qpath else original_read(path)
  old_owned,old_sources=N.OWNED_FILES,N.SOURCE_FILES;N.OWNED_FILES=owned;N.SOURCE_FILES=source
  try:
   with tempfile.TemporaryDirectory() as tmp:
    events=[];hooks={'lease_dir':pathlib.Path(tmp)/'lease','events':events,
      'dns':lambda:None,'credential':lambda:None}
    def committed(path,revision):
     full=path.removeprefix('docs/experiments/evidence/EXP-076/') if path.startswith('docs/experiments/evidence/EXP-076/') else None
     if full in owned:return (HERE/full).read_bytes()
     return (P.REPO/path).read_bytes()
    with mock.patch.object(pathlib.Path,'is_file',is_file),mock.patch.object(pathlib.Path,'read_bytes',read_bytes),\
         mock.patch.object(N,'committed_head',lambda path:(qraw,'a'*40)),mock.patch.object(P,'committed',committed),\
         mock.patch.object(P,'validate_inputs',lambda *args:dict(proof)):
     acq,queries=P.source_data(self.inputs);out=pathlib.Path(tmp)/'out'
     got=N.live_gate(self.inputs,output_path=out,acquisition=acq,queries=queries,_test_hooks=hooks)
    self.assertEqual(events,['lease','dns','credential']);self.assertIsNone(got[2]);self.assertTrue((pathlib.Path(tmp)/'lease'/('study-'+N.study_identity((HERE/'protocol.json').read_bytes(),(HERE/'registration-integrity.json').read_bytes(),self.inputs))).is_file())
    # Existing identity is rejected before either external sentinel; changing
    # qualification revisions does not change this protocol/input identity.
    events.clear();hooks['dns']=lambda:self.fail('DNS reached with existing lease');hooks['credential']=lambda:self.fail('key reached with existing lease')
    with mock.patch.object(pathlib.Path,'is_file',is_file),mock.patch.object(pathlib.Path,'read_bytes',read_bytes),\
         mock.patch.object(N,'committed_head',lambda path:(qraw,'a'*40)),mock.patch.object(P,'committed',committed),\
         mock.patch.object(P,'validate_inputs',lambda *args:dict(proof)),self.assertRaisesRegex(ValueError,'study-already-consumed'):
     N.live_gate(self.inputs,output_path=out,acquisition=acq,queries=queries,_test_hooks=hooks)
    self.assertEqual(events,[])
    # A source pin mutation fails during closure validation, before lease/DNS/key.
    q['source_pins'][next(iter(source))]='0'*64;bad_raw=P.wire(q);events.clear()
    with mock.patch.object(pathlib.Path,'is_file',is_file),mock.patch.object(pathlib.Path,'read_bytes',lambda p:bad_raw if p==qpath else original_read(p)),\
         mock.patch.object(N,'committed_head',lambda path:(bad_raw,'a'*40)),mock.patch.object(P,'committed',committed),\
         mock.patch.object(P,'validate_inputs',lambda *args:dict(proof)),self.assertRaisesRegex(ValueError,'qualification-source-pin'):
     N.live_gate(self.inputs,output_path=pathlib.Path(tmp)/'other',acquisition=acq,queries=queries,_test_hooks=hooks)
    self.assertEqual(events,[])
  finally:N.OWNED_FILES=old_owned;N.SOURCE_FILES=old_sources
