def load(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_loader(name, loader)
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    return module

def sha(raw): return hashlib.sha256(raw).hexdigest()
def canonical(value):
    return (json.dumps(value, sort_keys=True, separators=(",", ":"),
                        ensure_ascii=False, allow_nan=False) + "\n").encode()

            self.files[name]=sha(raw)

        reg={"schema":"exp103-anchor-registration/1","study_id":"EXP-103",

             "reader_transport_sha256":sha(self.reader.read_bytes()),"failed_run_pins":pins}

    def install_patches(self):

        QUAL.HERE=self.exp103;QUAL.REPO=self.repo

        # EXP-102 was already independently qualified; keep this dependency synthetic.
        class OldQualification:
            def verify(self, **kwargs): return {"status":"qualified"}
        class Loader:
            def __init__(self,name,path): self.path=path
            def load_module(self):
                if str(self.path).endswith("continuation-authority.py.txt"): return AUTH
                return OldQualification()

class QualificationAndAuthorityTests(unittest.TestCase):
    def setUp(self): self.fx=QualificationFixture()
    def tearDown(self): self.fx.close()

    def test_synthetic_registration_and_all_eight_failed_lineage_pins_qualify(self):
        result=self.fx.verify()
        self.assertEqual(result["status"],"qualified")
        self.assertEqual(result["study_id"],"EXP-103")
        self.assertEqual(result["failed_exp102_clock_sha256"],sha(self.fx.lineage_paths["clock"].read_bytes()))

    def test_nested_real_sourcefileloader_keeps_exp102_module_context(self):
        """EXP-103's loader name must not collide with EXP-102's nested EXP-101 load."""
        q102=self.fx.exp102/"qualification-preflight.py.txt"
        q101dir=self.fx.repo/"docs/experiments/evidence/EXP-101"
        q101dir.mkdir(parents=True)
        q101dir.joinpath("qualification-preflight.py.txt").write_text(
            "from pathlib import Path\nHERE = Path(__file__).resolve().parent\n",encoding="utf-8")
        q102.write_text(
            "from pathlib import Path\nimport importlib.machinery\n"
            "HERE = Path(__file__).resolve().parent\n"
            "def verify(**kwargs):\n"
            "    importlib.machinery.SourceFileLoader('exp102_original_qualification', "
            "str(HERE.parent / 'EXP-101/qualification-preflight.py.txt')).load_module()\n"
            "    return {'status': 'qualified'}\n",encoding="utf-8")
        previous=QUAL.importlib.machinery.SourceFileLoader
        real_loader=REAL_SOURCE_FILE_LOADER
        captured=[]
        class NestedLoader:
            def __init__(self,name,path): self.name,self.path=name,str(path)
            def load_module(self):
                if self.path.endswith("EXP-102/qualification-preflight.py.txt"):
                    self.assert_unique_name()
                    module=real_loader(self.name,self.path).load_module();captured.append(module)
                    return module
                if self.path.endswith("continuation-authority.py.txt"): return AUTH
                return real_loader(self.name,self.path).load_module()
            def assert_unique_name(self):
                if self.name!="exp103_previous_qualification":
                    raise AssertionError("EXP-103 must use a distinct module name for EXP-102")
        QUAL.importlib.machinery.SourceFileLoader=NestedLoader
        try:
            self.fx.verify()
            self.assertEqual(len(captured),1)
            self.assertEqual(captured[0].HERE.resolve(),self.fx.exp102.resolve())
        finally:
            QUAL.importlib.machinery.SourceFileLoader=previous

    def test_missing_unqualified_registration_trim_and_private_pin_drift_fail_closed(self):
        qpath=self.fx.exp103/"qualification.json"
        original=qpath.read_bytes()
        q=json.loads(original);qpath.unlink()
        with self.assertRaisesRegex(ValueError,"qualification-missing"): self.fx.verify()
        qpath.write_bytes(original)
        q["qualified"]=False;qpath.write_bytes(canonical(q));self.fx.blobs[("HEAD","docs/experiments/evidence/EXP-103/qualification.json")]=qpath.read_bytes()
        with self.assertRaisesRegex(ValueError,"not-qualified"): self.fx.verify()
        qpath.write_bytes(original);self.fx.blobs[("HEAD","docs/experiments/evidence/EXP-103/qualification.json")]=original
        regpath=self.fx.exp103/"registration-integrity.json"; reg=regpath.read_bytes();regpath.unlink()
        with self.assertRaisesRegex(ValueError,"registration-missing"): self.fx.verify()
        regpath.write_bytes(reg)
        q=json.loads(original);q["source_files"].pop("docs/experiments/evidence/EXP-103/grading-driver.py.txt")
        qpath.write_bytes(canonical(q));self.fx.blobs[("HEAD","docs/experiments/evidence/EXP-103/qualification.json")]=qpath.read_bytes()
        with self.assertRaisesRegex(ValueError,"source-inventory"): self.fx.verify()
        qpath.write_bytes(original);self.fx.blobs[("HEAD","docs/experiments/evidence/EXP-103/qualification.json")]=original
        self.fx.scheduler.write_bytes(b"changed scheduler")
        with self.assertRaisesRegex(ValueError,"scheduler-changed"): self.fx.verify()

    def test_lineage_requires_terminal_zero_acceptance_one_charge_and_all_bindings(self):
        auditpath=self.fx.lineage_paths["terminal_audit"]
        audit=json.loads(auditpath.read_bytes());audit["accepted_grader_results"]=1
        auditpath.write_bytes(canonical(audit))
        self.fx.update_registration_and_qualification()
        with self.assertRaisesRegex(ValueError,"failed-lineage-not-terminal"):
            self.fx.verify()
        with self.assertRaisesRegex(ValueError,"failed-lineage-inventory"):
            paths=dict(self.fx.lineage_paths);paths.pop("view_proof")
            QUAL.verify(original_entrypoints=[],previous_continuation_entrypoint=self.fx.scheduler,
                previous_failed_run_files={},continuation_entrypoint=self.fx.scheduler,
                reader_entrypoint=self.fx.reader,failed_run_files=paths)

    def test_authority_protocol_keeps_execution_caps_order_and_namespace(self):
        AUTH.validate_protocol(self.fx.protocol,self.fx.old_protocol)
        for mutate in (lambda p:p["execution"].__setitem__("fresh_answer_assessments",33),
                       lambda p:p["immutable_science"].__setitem__("source_limit","changed"),
                       lambda p:p["operative_execution_order"].__setitem__("precedence","old first"),
                       lambda p:p["continuation"].__setitem__("physical_actor_namespace","/root/old")):
            bad=json.loads(json.dumps(self.fx.protocol));mutate(bad)
            with self.subTest(bad=bad.get("continuation",{}).get("physical_actor_namespace")),self.assertRaises(ValueError):
                AUTH.validate_protocol(bad,self.fx.old_protocol)

    def test_new_clock_binds_fresh_exp103_identity_and_rejects_failed_exp102_clock(self):
        old={"schema":"exp101-grading-clock/1","stage_id":str(uuid.uuid4()),
             "source_revision":"b"*40,"protocol_sha256":"0"*64}
        protocol_raw=canonical(self.fx.protocol)
        current={"schema":"exp101-grading-clock/1","stage_id":str(uuid.uuid4()),
                 "source_revision":self.fx.rev,"protocol_sha256":sha(protocol_raw)}
        result=AUTH.bind_clock(clock_raw=canonical(current),failed_clock_raw=canonical(old),
            protocol_raw=protocol_raw,qualified_source_revision=self.fx.rev)
        self.assertEqual(result["schema"],"exp103-current-clock-authority/1")
        self.assertEqual(result["study_id"],"EXP-103")
        with self.assertRaisesRegex(ValueError,"fresh-clock-required"):
            AUTH.bind_clock(clock_raw=canonical(old),failed_clock_raw=canonical(old),
                protocol_raw=protocol_raw,qualified_source_revision=self.fx.rev)
        reused=dict(current,stage_id=old["stage_id"])
        with self.assertRaisesRegex(ValueError,"fresh-clock-required"):
            AUTH.bind_clock(clock_raw=canonical(reused),failed_clock_raw=canonical(old),
                protocol_raw=protocol_raw,qualified_source_revision=self.fx.rev)
        wrongrev=dict(current,source_revision="c"*40)
        with self.assertRaisesRegex(ValueError,"fresh-clock-required"):
            AUTH.bind_clock(clock_raw=canonical(wrongrev),failed_clock_raw=canonical(old),
                protocol_raw=protocol_raw,qualified_source_revision=self.fx.rev)
        wrongprotocol=dict(current,protocol_sha256="f"*64)
        with self.assertRaisesRegex(ValueError,"fresh-clock-required"):
            AUTH.bind_clock(clock_raw=canonical(wrongprotocol),failed_clock_raw=canonical(old),
                protocol_raw=protocol_raw,qualified_source_revision=self.fx.rev)

if __name__ == "__main__": unittest.main()
