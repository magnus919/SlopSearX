def digest(raw):return hashlib.sha256(raw).hexdigest()
def atomic(path,value):
    path=Path(path);raw=value if type(value)is bytes else json.dumps(value,ensure_ascii=False,separators=(',',':'),allow_nan=False).encode()
    tmp=path.with_name(path.name+'.tmp')
    with tmp.open('xb') as stream:stream.write(raw);stream.flush();os.fsync(stream.fileno())
    os.replace(tmp,path);sync_directory(path.parent)
def sync_pending(path,value):
    raw=json.dumps(value,ensure_ascii=False,separators=(',',':'),allow_nan=False).encode()
    with Path(path).open('xb') as stream:stream.write(raw);stream.flush();os.fsync(stream.fileno())
    sync_directory(Path(path).parent)
def sync_directory(path):
    fd=os.open(path,os.O_RDONLY)
    try:os.fsync(fd)
    finally:os.close(fd)
def durable_mkdirs(path):
    path=Path(path);missing=[];probe=path
    while not probe.exists():missing.append(probe);probe=probe.parent
    path.mkdir(parents=True,exist_ok=True,mode=0o700)
    for directory in missing:sync_directory(directory);sync_directory(directory.parent)
def committed(path,revision):return P.committed(path,revision)
def committed_head(path):

def study_identity(protocol_raw,registration_raw,input_bytes):
    if type(protocol_raw)is not bytes or type(registration_raw)is not bytes or type(input_bytes)is not dict or set(input_bytes)!=set(P.PINS):
        raise ValueError('study-identity-inputs')
    if any(type(value)is not bytes for value in input_bytes.values()):raise ValueError('study-identity-inputs')
    return digest(protocol_raw+registration_raw+b''.join(name.encode()+b'\0'+input_bytes[name] for name in sorted(input_bytes)))

def usage_admissible(budget,bounds):

            and budget['output_tokens']+32_000<=bounds['reported_output_tokens'])

def fixture_client_factory(mode='scores',stats=None):
    stats=stats if stats is not None else {'clients':0,'requests':0,'closed':0}
    class CountingAsyncClient(httpx.AsyncClient):
        async def aclose(self):
            if not getattr(self,'_exp075_closed_counted',False):self._exp075_closed_counted=True;stats['closed']+=1
            await super().aclose()
    async def handler(request):
        stats['requests']+=1
        parsed=P.strict_json(request.content)
        if mode=='redirect':return httpx.Response(302,headers={'Location':'https://fixture.invalid/redirect'},content=b'')
        if mode=='http-error':return httpx.Response(503,content=P.response_fixture(parsed))
        if mode=='missing-usage':
            payload=P.strict_json(P.response_fixture(parsed));payload.pop('usage',None);return httpx.Response(200,content=P.wire(payload))
        if mode=='overflow':return httpx.Response(200,content=b'x'*(T.MAX_RESPONSE_BYTES+10))
        if mode=='credential-echo':return httpx.Response(200,content=request.headers['authorization'].encode())
        if mode=='slow-joint' and any(k.startswith('purpose:') for k in parsed['questions']):
            await asyncio.sleep(2)
        score_by_id=None
        if mode=='mixed':score_by_id={key:(9 if key.startswith('w0:') else 3) for key in parsed['questions']}
        return httpx.Response(200,content=P.response_fixture(parsed,score_by_id=score_by_id))
    def factory(**kwargs):
        stats['clients']+=1
        return CountingAsyncClient(transport=httpx.MockTransport(handler),**kwargs)
    factory.stats=stats
    return factory

async def run(acquisition,queries,outdir,offline=True,source_integrity=None,prebuilt=None,client_factory=None,
              api_key=None,qualification_bytes=None,build_joint_fn=None,admission=None,input_bytes=None):

    if outdir.exists() or outdir.is_symlink():raise ValueError('output-exists')
    if not outdir.parent.is_dir() or not os.access(outdir.parent,os.W_OK):raise ValueError('output-parent')

        if admission is not None:raise ValueError('offline-admission-not-allowed')
        api_key='offline-fixture-non-secret'
    else:

                or _ISSUED_ADMISSIONS.get(id(admission)) is not admission
                or type(input_bytes)is not dict or type(qualification_bytes)is not bytes):raise ValueError('live-admission-required')
        admission.used=True;_ISSUED_ADMISSIONS.pop(id(admission),None)

                or client_factory is not None or build_joint_fn is not None):

        expected_acquisition,expected_queries=P.source_data(input_bytes)
        expected_prepared=P.prebuild(expected_acquisition,expected_queries)
        if acquisition!=expected_acquisition or queries!=expected_queries or prepared!=expected_prepared:
            raise ValueError('live-input-object-mismatch')
        api_key=admission.api_key
    outdir.mkdir();receipts=outdir/'receipts';receipts.mkdir()
    if not offline:atomic(outdir/'qualification-integrity.json',qualification_bytes)
    if offline and client_factory is None:client_factory=fixture_client_factory()
    owner=T.AsyncHTTPTransport(client_factory=client_factory)

            'source_integrity':source_integrity,'neutral':None,'operations':[],'budget':dict(budget),'wall_seconds':0.0}

    async def physical_call(body_or_factory,opid,arm,phase_deadline,usage_cap,preview_sha=None):
        nonlocal call_index
        index=call_index;call_index+=1;path=receipts/('call-%02d.json'%index)
        base={'status':'pending','dispatched':False,'operation':opid,'arm':arm,'request_hex':None,
              'request_sha256':preview_sha,'raw_response_hex':None,'response_body_sha256':None,'response_bytes':0,
              'response_complete':False,'http_status':None,'usage':None,'usage_status':'not_dispatched',
              'elapsed_ms':None,'timings':{},'transport':{'kind':'inprocess-async-httpx','trust_env':False,
              'follow_redirects':False,'retries':0,'concurrency':1,'response_limit_bytes':T.MAX_RESPONSE_BYTES,
              'request_sha256':preview_sha,'response_sha256':None,'client_closed':False},'failure':None}
        pending_ns=time.perf_counter_ns();sync_pending(path,base);pending_ms=(time.perf_counter_ns()-pending_ns)/1e6
        build_ms=serialization_ms=0.0
        try:
            if type(body_or_factory)is bytes:
                raw_request=body_or_factory
            else:
                bstart=time.perf_counter_ns();body=body_or_factory();build_ms=(time.perf_counter_ns()-bstart)/1e6
                sstart=time.perf_counter_ns();raw_request=P.wire(body);serialization_ms=(time.perf_counter_ns()-sstart)/1e6
            request_sha=digest(raw_request)
            if preview_sha is not None and request_sha!=preview_sha:raise ValueError('preview-byte-mismatch')
            if len(raw_request)>bounds['legacy_control_request_bytes'] and arm=='w0':raise ValueError('control-request-bound')
            if len(raw_request)>bounds['request_bytes']:raise ValueError('candidate-request-bound')
            base.update(request_hex=raw_request.hex(),request_sha256=request_sha)
            base['transport'].update(request_sha256=request_sha)
            base['timings'].update({'pending_receipt_fsync_ms':pending_ms,'build_ms':build_ms,'serialization_ms':serialization_ms})
            atomic(path,base)
            if phase_deadline-time.perf_counter()<=0:raise TimeoutError('whole-phase-deadline')
            if budget['attempts']>=bounds['calls']:raise ValueError('call-count')
            if not usage_admissible(budget,bounds):
                raise ValueError('total-usage-admission')
            timeout_s=min(1.0,phase_deadline-time.perf_counter())
            call_start=time.perf_counter_ns()
            def dispatch():
                if phase_deadline-time.perf_counter()<=0:raise T.DispatchDenied()
                fsync_start=time.perf_counter_ns();base['dispatched']=True;base['usage_status']='unknown';budget['attempts']+=1
                atomic(path,base);dispatch_fsync=(time.perf_counter_ns()-fsync_start)/1e6
                if phase_deadline-time.perf_counter()<=0:
                    base['dispatched']=False;base['usage_status']='not_dispatched';budget['attempts']-=1
                    base.update(status='failed',failure={'type':'DispatchDenied','code':'whole-phase-deadline'})
                    atomic(path,base);raise T.DispatchDenied()
                base['timings']['dispatch_receipt_fsync_ms']=dispatch_fsync
            try:
                if type(api_key)is not str or not api_key:raise ValueError('missing-credential')
                raw,status,complete,transport_timings=await owner.exchange(raw_request,api_key,timeout_s,endpoint,on_dispatch=dispatch)
                rec=owner.last_attempt or {};usage=P.usage_from_raw(raw)
                base.update(status='returned',raw_response_hex=raw.hex(),response_body_sha256=digest(raw),response_bytes=len(raw),
                            response_complete=complete,http_status=status,usage=usage,usage_status='known' if usage is not None else 'unknown')
                if usage:
                    budget['input_tokens']+=usage['input_tokens'];budget['output_tokens']+=usage['output_tokens']
                else:budget['unknown_usage']=True
                base['transport'].update(response_sha256=digest(raw),client_closed=bool(rec.get('client_closed')))
                base['timings'].update(transport_timings)
                base['elapsed_ms']=(time.perf_counter_ns()-call_start)/1e6
                if status!=200:raise CallFailure(base,'http-status')
                if not complete:raise CallFailure(base,'incomplete-response')
                if usage is None:raise CallFailure(base,'unknown-usage')
                if usage['input_tokens']>64_000 or usage['output_tokens']>32_000:raise CallFailure(base,'per-call-usage')
                if budget['input_tokens']>bounds['reported_input_tokens'] or budget['output_tokens']>bounds['reported_output_tokens']:
                    raise CallFailure(base,'total-usage')
            except T.TransportFailure as error:
                rec=owner.last_attempt or {};raw=error.raw;usage=P.usage_from_raw(raw) if raw is not None else None
                base.update(status='failed',http_status=error.http_status,response_complete=error.response_complete,
                            usage=usage,usage_status='known' if usage is not None else ('unknown' if error.dispatched else 'not_dispatched'),
                            failure={'type':'TransportFailure','code':error.code},elapsed_ms=(time.perf_counter_ns()-call_start)/1e6)
                if raw is not None and (raw or error.response_complete):
                    base.update(raw_response_hex=raw.hex(),response_body_sha256=digest(raw),response_bytes=len(raw))
                    base['transport']['response_sha256']=digest(raw)
                if usage is not None:
                    budget['input_tokens']+=usage['input_tokens'];budget['output_tokens']+=usage['output_tokens']
                elif error.dispatched:budget['unknown_usage']=True
                base['transport']['client_closed']=bool(rec.get('client_closed'));base['timings'].update(error.timings)
                raise CallFailure(base,error.code) from None
            except T.DispatchDenied:
                rec=owner.last_attempt or {};base.update(status='failed',usage_status='not_dispatched',
                    failure={'type':'DispatchDenied','code':'whole-phase-deadline'},elapsed_ms=(time.perf_counter_ns()-call_start)/1e6)
                base['transport']['client_closed']=bool(rec.get('client_closed'));base['timings'].update(rec.get('timings',{}))
                raise CallFailure(base,'whole-phase-deadline') from None
            except CallFailure:
                raise
            except BaseException as error:
                if isinstance(error,asyncio.CancelledError):
                    rec=owner.last_attempt or {};base.update(status='failed',usage_status='unknown' if base['dispatched'] else 'not_dispatched',
                         failure={'type':'CancelledError','code':'cancelled'},elapsed_ms=(time.perf_counter_ns()-call_start)/1e6)
                    if base['dispatched']:
                        raw=rec.get('raw');base['raw_response_hex']=raw.hex() if type(raw)is bytes else None
                        base['response_body_sha256']=digest(raw) if type(raw)is bytes else None
                        base['response_bytes']=len(raw) if type(raw)is bytes else 0
                        base['transport']['response_sha256']=base['response_body_sha256'];budget['unknown_usage']=True
                    base['transport']['client_closed']=bool(rec.get('client_closed'));base['timings'].update(rec.get('timings',{}))
                else:
                    base.update(status='failed',usage_status='not_dispatched' if not base['dispatched'] else 'unknown',
                                failure={'type':type(error).__name__,'code':str(error) if str(error).replace('-','').isalnum() else type(error).__name__.lower()},
                                elapsed_ms=(time.perf_counter_ns()-call_start)/1e6)
                    if base['dispatched'] and base['usage'] is None:budget['unknown_usage']=True
                raise CallFailure(base,base['failure']['code']) from None
            fsync_start=time.perf_counter_ns();atomic(path,base)
            base['timings']['response_receipt_fsync_ms']=(time.perf_counter_ns()-fsync_start)/1e6
            atomic(path,base)
            return raw,base
        except CallFailure as error:
            if error.record is not base:base=error.record
            if base.get('status')=='pending':base.update(status='failed',usage_status='not_dispatched',
                failure={'type':type(error).__name__,'code':error.code})
            fsync_start=time.perf_counter_ns();atomic(path,base);receipt_ms=(time.perf_counter_ns()-fsync_start)/1e6
            base['timings']['response_receipt_fsync_ms']=receipt_ms
            atomic(path,base)
            error.record=base
            raise
        except BaseException as error:
            if isinstance(error,CallFailure):raise
            base.update(status='failed',usage_status='not_dispatched',failure={'type':type(error).__name__,'code':type(error).__name__.lower()})
            fsync_start=time.perf_counter_ns();atomic(path,base);base['timings']['response_receipt_fsync_ms']=(time.perf_counter_ns()-fsync_start)/1e6
            atomic(path,base);raise CallFailure(base,base['failure']['code']) from None

    def blank_operation(item,case,cards,control):

                'baseline_phase_ms':None,'candidate_phase_ms':None,'paired_wall_ms':None,
                'candidate_build_ms':None,'candidate_serialization_ms':None,'phase_endpoint_after_fsync':False,
                'baseline_receipt':None,'candidate_body_receipt':None,'final_receipt':None,'failure':None}

    async def operation(item,case,cards,control,joint,control_raw,joint_raw):
        paired_start=time.perf_counter();opid=item['case']+'-'+item['variant'];op=blank_operation(item,case,cards,control)
        # Independent fresh W0 control phase.
        phase_start=time.perf_counter();deadline=phase_start+1.0
        try:
            raw,rec=await physical_call(control_raw,opid,'w0',deadline,bounds['legacy_control_request_bytes'],digest(control_raw))
            op['calls'].append(rec)
            parse_start=time.perf_counter_ns();parsed=P.parse_control(raw,control)
            op['arms']['w0']={'status':'valid','order_ids':parsed['order_ids'],'scores':parsed['scores'],
                'raw_answers':parsed['raw_answers'],'rational_scores':{},'usage':parsed['usage'],'elapsed_ms':rec['elapsed_ms']}
            parse_ms=(time.perf_counter_ns()-parse_start)/1e6
            sidecar=receipts/(opid+'-baseline.json')
            fsync_start=time.perf_counter_ns();atomic(sidecar,{'receipt_status':'baseline-valid','operation':op});fsync_ms=(time.perf_counter_ns()-fsync_start)/1e6
