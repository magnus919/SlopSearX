import ast,hashlib,json,subprocess
from pathlib import Path
paths=subprocess.check_output(['git','ls-files','slopsearx','engines'],text=True).splitlines()
files=[];calls=[];definitions=[];constructors=[];references=[]
for name in paths:
 p=Path(name)
 if p.suffix!='.py':continue
 source=p.read_text();tree=ast.parse(source);files.append({'path':name,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()})
 for node in ast.walk(tree):
  if isinstance(node,(ast.FunctionDef,ast.AsyncFunctionDef)) and node.name=='set_error':definitions.append({'path':name,'line':node.lineno})
  if isinstance(node,ast.Attribute) and node.attr=='set_error':references.append({'path':name,'line':node.lineno})
  if isinstance(node,ast.Call):
   target=node.func
   if (isinstance(target,ast.Attribute) and target.attr=='set_error') or (isinstance(target,ast.Name) and target.id=='set_error'):calls.append({'path':name,'line':node.lineno,'call':ast.get_source_segment(source,node)})
   if isinstance(target,ast.Name) and target.id=='SearchCache':constructors.append({'path':name,'line':node.lineno})
result={'baseline':'058575daa037ed20737a9ca0ecfdf4f2149838ae','registration':'5e31742','files':files,'set_error_definitions':definitions,'set_error_calls':calls,'set_error_attribute_references':references,'SearchCache_constructors':constructors,'decision':'blocked' if not calls else 'manual path review required','runtime_replay_calls':0}
out=Path('docs/experiments/evidence/EXP-030');out.mkdir(exist_ok=True);(out/'audit.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k!='files'},indent=2))
