class FakeClock:
    def __init__(self) -> None:
        self.value = 1000.0

    def __call__(self) -> float:
        return self.value

    async def sleep(self, seconds: float) -> None:
        self.value += seconds


class FakeResponse:
    def __init__(self, status: int, body: bytes = b"[]", headers: dict[str, str] | None = None,
                 chunks: list[bytes] | None = None) -> None:
        self.status = status
        self.headers = headers or {}
        self.body = body
        self.chunks = chunks
        self.closed = False

    async def iter_bytes(self):
        for part in (self.chunks if self.chunks is not None else [self.body]):
            await asyncio.sleep(0)
            yield part

    async def close(self) -> None:
        self.closed = True


class FakeTransport:
    def __init__(self, replies: list[Any], clock: FakeClock | None = None,
                 advance: float = 0.0) -> None:
        self.replies = list(replies)
        self.requests = []
        self.clock = clock
        self.advance = advance

    async def __call__(self, request):
        self.requests.append(request)
        if self.clock is not None:
            self.clock.value += self.advance
        reply = self.replies.pop(0)
        if isinstance(reply, BaseException):
            raise reply
        return reply

def digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()
