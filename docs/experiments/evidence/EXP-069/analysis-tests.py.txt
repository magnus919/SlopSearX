import copy,importlib.machinery,importlib.util,json,pathlib,unittest
ROOT=pathlib.Path(__file__).resolve().parent
l=importlib.machinery.SourceFileLoader('analysis069',str(ROOT/'analyze.py.txt'));s=importlib.util.spec_from_loader(l.name,l);A=importlib.util.module_from_spec(s);l.exec_module(A)

def fixture():
    protocol=json.loads((ROOT/'protocol.json').read_text());acq={'pools':{}};refs={'A':{},'B':{}};queries=[];ops=[]
    for i in range(1,14):
        q=f'q{i}';cards=[{'id':f'c{j}','title':f'title{j}','url':f'https://example.org/{j}','snippet':'fixture'} for j in range(12)];acq['pools'][q]={'cards':cards};queries.append({'id':q,'kind':'navigation' if i>=9 else 'research','target_url':'https://example.org/8'})
        for who in refs:refs[who][q]={c['id']:{'lead':3 if j>=8 else 0,'facets':['f1'],'visible':1,'uncertain':False,'rationale':'fixture'} for j,c in enumerate(cards)}
    usage={'input_tokens':1,'output_tokens':1}
    neutral={arm:{'status':'valid','questions':80,'synthetic_input':True,'elapsed_ms':100,'usage':dict(usage)} for arm in ('gate','relative')}
    for plan in protocol['schedule']:
        q=plan['case'];base=[c['id'] for c in acq['pools'][q]['cards']]
        native=base[8:]+base[:8] if plan['variant']=='rotate' else base
        relative=base[8:]+base[:8];scores={c:4-j/100 for j,c in enumerate(relative)}
        probabilities={c:0.75 for c in base}
        w0={'status':'valid','order_ids':base,'scores':{c:9-j/100 for j,c in enumerate(base)},'elapsed_ms':100,'usage':dict(usage)}
        rel={'status':'valid','order_ids':relative,'scores':scores,'elapsed_ms':100,'usage':dict(usage)}
        gate={'status':'valid','order_ids':A.canonical_ids(acq['pools'][q]['cards']),'probabilities':probabilities,'elapsed_ms':100,'usage':dict(usage)}
        ops.append({'case':q,'variant':plan['variant'],'id':q+'-'+plan['variant'],'provenance_order_ids':native,'candidate_phase_ms':250,'arms':{'w0':w0,'gate':gate,'relative':rel}})
    run={'wall_seconds':1,'status':'complete','synthetic':False,'neutral':neutral,'budget':{'attempts':65,'input_tokens':65,'output_tokens':65,'unknown_usage':False},'operations':ops}
    return run,acq,refs,queries,protocol

class Tests(unittest.TestCase):
    def test_complete_quality_never_authorizes_adoption(self):
        r=A.analyze(*fixture());self.assertTrue(r['development_all_gates_passed']);self.assertFalse(r['adoption_eligible']);self.assertEqual(r['schema'],'exp069-paired-analysis/1')
    def test_synthetic_cannot_be_quality(self):
        x=fixture();x[0]['synthetic']=True;r=A.analyze(*x);self.assertFalse(r['development_quality_passed']);self.assertFalse(r['quality_analysis_performed']);self.assertEqual(r['outcome'],'synthetic_fixture_only')
    def test_other_assessor_veto(self):
        x=fixture()
        for rows in x[2]['B'].values():
            for cid,row in rows.items():row['lead']=3 if int(cid[1:])<4 else 0
        self.assertFalse(A.analyze(*x)['development_quality_passed'])
    def test_gate_inventory_boolean_range_nonfinite_and_missing_are_rejected(self):
        for bad in ('bool','range','nan','missing'):
            x=fixture();p=x[0]['operations'][0]['arms']['gate']['probabilities']
            if bad=='bool':p['c0']=True
            elif bad=='range':p['c0']=1.01
            elif bad=='nan':p['c0']=float('nan')
            else:del p['c0']
            self.assertEqual(A.analyze(*x)['outcome'],'incomplete',bad)
    def test_half_boundary_and_probability_magnitude_do_not_rank_within_bucket(self):
        x=fixture();op=x[0]['operations'][0];gate=op['arms']['gate'];score=op['arms']['relative']
        # c0 is exactly .5 and belongs after every >.5 card. Among >.5 cards,
        # relative Score—not the difference between .51 and .99—determines order.
        gate['probabilities'].update({'c0':.5,'c1':.51,'c2':.99})
        score['scores'].update({'c0':4,'c1':1,'c2':3})
        canonical=A.canonical_ids(x[1]['pools']['q1']['cards'])
        score['order_ids']=sorted(canonical,key=lambda c:(0 if gate['probabilities'][c]>.5 else 1,-score['scores'][c],canonical.index(c)))
        self.assertFalse(A.analyze(*x)['structural_failures'])
        score['order_ids'].reverse()
        self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
    def test_composite_deadline_and_exact_65_call_usage(self):
        x=fixture();r=A.analyze(*x);self.assertFalse(r['structural_failures'])
        x=fixture();x[0]['operations'][0]['candidate_phase_ms']=199.9;self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
        x=fixture();x[0]['operations'][0]['arms']['gate']['elapsed_ms']=700;x[0]['operations'][0]['candidate_phase_ms']=700;self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
        x=fixture();x[0]['budget']['attempts']=64;self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
        x=fixture();x[0]['budget']['input_tokens']=66;self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
    def test_two_neutral_calls_and_summed_usage_are_required(self):
        x=fixture();del x[0]['neutral']['gate'];self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
        x=fixture();x[0]['neutral']['relative']['questions']=79;self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
        x=fixture();x[0]['neutral']['gate']['usage']['output_tokens']=64001;self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
    def test_relative_scores_still_use_canonical_ties_and_fractional_scores(self):
        x=fixture();row=x[0]['operations'][0]['arms']['relative'];row['scores']={c:2.5 for c in row['order_ids']};cards=x[1]['pools']['q1']['cards'];row['order_ids']=A.canonical_ids(cards)
        gate=x[0]['operations'][0]['arms']['gate']
        row['order_ids']=sorted(row['order_ids'],key=lambda c:(0 if gate['probabilities'][c]>.5 else 1,-row['scores'][c],row['order_ids'].index(c)))
        self.assertFalse(A.analyze(*x)['structural_failures'])
    def test_registered_schedule_requires_exact_three_arm_inventory(self):
        x=fixture();x[0]['operations'][0]['arms'].pop('gate');self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
        x=fixture();x[4]['schedule'][0]['arms']=['w0','relative'];self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
    def test_missing_or_duplicate_navigation_inputs_cannot_pass(self):
        x=list(fixture());x[3]=[q for q in x[3] if q['kind']!='navigation'];self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
        x=list(fixture());x[3].append(copy.deepcopy(x[3][-1]));self.assertEqual(A.analyze(*x)['outcome'],'incomplete')
    def test_absent_navigation_is_unmet_recall(self):
        x=fixture();x[3][9]['target_url']='https://example.org/missing';r=A.analyze(*x);self.assertTrue(r['navigation_gate_met']);self.assertFalse(r['acquisition_recall_met']);self.assertFalse(r['development_all_gates_passed'])

if __name__=='__main__':unittest.main()
