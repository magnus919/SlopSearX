"""Inert paired analysis; exposed development is never production adoption."""
import math,random,statistics

CALL_ARMS=('w0','gate','relative')
QUALITY_ARMS=('w0','relative')
TOKEN_CAPS={'input_tokens':3000000,'output_tokens':160000}

def ndcg(order,rows):
    def dcg(ids):return sum((2**rows[c]['lead']-1)/math.log2(i+2) for i,c in enumerate(ids[:10]))
    ideal=dcg(sorted(rows,key=lambda c:-rows[c]['lead']))
    return dcg(order)/ideal if ideal else 0.0

def useful(order,rows):return sum(rows[c]['lead']>=2 for c in order[:10])

def covered(order,rows):return {f for c in order[:10] if rows[c]['lead']>=2 for f in rows[c]['facets']}

def interval(values):
    rng=random.Random(6501);samples=sorted(statistics.mean(rng.choices(values,k=len(values))) for _ in range(10000))
    return [samples[249],samples[9749]]

def finite_number(value,low,high):
    return type(value) in (int,float) and math.isfinite(value) and low<=value<=high

def canonical_ids(cards):
    return [c['id'] for c in sorted(cards,key=lambda c:tuple(c[f].encode('utf-8') for f in ('url','title','snippet','id')))]

def analyze(run,acquisition,references,queries,protocol):
    schedule=protocol['schedule'];operations=run.get('operations',[])
    expected=[(x['case'],x['variant']) for x in schedule]
    actual=[(x.get('case'),x.get('variant')) for x in operations]
    structural=[]
    query_ids=[q.get('id') for q in queries]
    if len(query_ids)!=13 or set(query_ids)!={f'q{i}' for i in range(1,14)}:structural.append('query-inventory')
    if {q.get('id') for q in queries if q.get('kind')=='navigation'}!={f'q{i}' for i in range(9,14)}:structural.append('navigation-inventory')
    if actual!=expected:structural.append('registered_schedule')
    if any(x.get('arms')!=list(CALL_ARMS) for x in schedule):structural.append('registered_arm_schedule')
    if run.get('status')!='complete':structural.append('incomplete_run')
    if run.get('budget',{}).get('unknown_usage')is not False:structural.append('unknown_usage')
    if run.get('budget',{}).get('attempts')!=65:structural.append('physical_call_inventory')
    if type(run.get('synthetic'))is not bool:structural.append('missing_execution_identity')
    for key,cap in TOKEN_CAPS.items():
        value=run.get('budget',{}).get(key)
        if type(value)is not int or not 0<=value<=cap:structural.append('budget-'+key)

    # Exactly two neutral calls, each with the same bounded maximum-pool input.
    neutral=run.get('neutral',{})
    if type(neutral)is not dict or set(neutral)!={'gate','relative'}:structural.append('neutral-call-inventory')
    else:
        for arm in ('gate','relative'):
            row=neutral[arm]
            if type(row)is not dict or row.get('status')!='valid' or row.get('questions')!=80 or row.get('synthetic_input')is not True:structural.append('neutral-'+arm+'-acceptance')
            if type(row)is dict and not finite_number(row.get('elapsed_ms'),0,1000):structural.append('neutral-'+arm+'-deadline')

    usage_rows=[]
    if type(neutral)is dict:usage_rows.extend(row for row in neutral.values() if type(row)is dict)
    for op in operations:
        if type(op.get('arms'))is dict:usage_rows.extend(row for row in op['arms'].values() if type(row)is dict)
    totals={key:0 for key in TOKEN_CAPS}
    for row in usage_rows:
        usage=row.get('usage',{})
        if type(usage)is not dict or set(usage)!=set(totals) or any(type(v)is not int or not 0<=v<=64000 for v in usage.values()):structural.append('per-call-usage')
        else:
            for key in totals:totals[key]+=usage[key]
    if any(totals[key]!=run.get('budget',{}).get(key) for key in totals):structural.append('usage-sum')
    wall=run.get('wall_seconds')
    if not finite_number(wall,0,7200):structural.append('wall-deadline')

    for op in operations:
        cards=acquisition['pools'][op['case']]['cards'];ids=[c['id'] for c in cards]
        if op['variant']=='rotate':
            offset=20%len(ids);ids=ids[offset:]+ids[:offset]
        if op.get('provenance_order_ids')!=ids:structural.append(op['case']+'-provenance')
        canonical=canonical_ids(cards)
        if type(op.get('arms'))is not dict or set(op['arms'])!=set(CALL_ARMS):structural.append(op['case']+'-arms');continue
        for arm in CALL_ARMS:
            row=op['arms'][arm];order=row.get('order_ids',[])
            if row.get('status')!='valid':structural.append(op['case']+'-'+arm+'-failure')
            if len(order)!=len(ids) or len(set(order))!=len(order) or set(order)!=set(ids):structural.append(op['case']+'-'+arm+'-membership')
            if arm=='w0' and (set(order[:40])!=set(ids[:40]) or order[40:]!=ids[40:]):structural.append(op['case']+'-w0-tail')
            if arm=='w0':
                scores=row.get('scores',{})
                if type(scores)is not dict or set(scores)!=set(ids[:40]) or any(not finite_number(v,0,9) for v in scores.values()):structural.append(op['case']+'-w0-scores')
                elif order!=sorted(ids[:40],key=lambda c:-scores[c])+ids[40:]:structural.append(op['case']+'-w0-score_order')
            elif arm=='gate':
                probabilities=row.get('probabilities',{})
                if type(probabilities)is not dict or set(probabilities)!=set(ids) or any(not finite_number(v,0,1) for v in probabilities.values()):structural.append(op['case']+'-gate-probabilities')
                elif order!=canonical:structural.append(op['case']+'-gate-canonical_order')
            elif arm=='relative':
                scores=row.get('scores',{})
                if type(scores)is not dict or set(scores)!=set(ids) or any(not finite_number(v,0,4) for v in scores.values()):structural.append(op['case']+'-relative-scores')
                else:
                    probabilities=op['arms']['gate'].get('probabilities',{})
                    if type(probabilities)is dict and set(probabilities)==set(ids) and all(finite_number(v,0,1) for v in probabilities.values()):
                        expected_order=sorted(canonical,key=lambda c:(0 if probabilities[c]>.5 else 1,-scores[c],canonical.index(c)))
                        if order!=expected_order:structural.append(op['case']+'-relative-combined_order')
            elapsed=row.get('elapsed_ms')
            if not finite_number(elapsed,0,1000):structural.append(op['case']+'-'+arm+'-deadline')
        phase=op.get('candidate_phase_ms');gate_ms=op['arms']['gate'].get('elapsed_ms');score_ms=op['arms']['relative'].get('elapsed_ms')
        if not finite_number(phase,0,1000) or not finite_number(gate_ms,0,1000) or not finite_number(score_ms,0,1000) or phase+1e-6<gate_ms+score_ms:structural.append(op['case']+'-candidate-phase')

    result={'schema':'exp069-paired-analysis/1','dataset':'exposed development; not untouched confirmation','adoption_eligible':False,'quality_analysis_performed':False,'structural_failures':structural,'synthetic':run.get('synthetic'),'assessors':{}}
    if structural:return dict(result,outcome='incomplete')
    bykey={(op['case'],op['variant']):op for op in operations}
    stability=[]
    for q in ('q1','q4','q7','q8'):
        base=set(bykey[q,'base']['arms']['relative']['order_ids'][:10])
        for variant in ('repeat','rotate'):
            other=set(bykey[q,variant]['arms']['relative']['order_ids'][:10]);overlap=len(base&other)/min(10,len(base))
            stability.append({'case':q,'variant':variant,'overlap':overlap,'passed':overlap>=.8})
    nav=[]
    for q in queries:
        if q['kind']!='navigation':continue
        target=q['target_url'].rstrip('/');cards=acquisition['pools'][q['id']]['cards']
        targets={c['id'] for c in cards if c['url'].rstrip('/')==target}
        orders=bykey[q['id'],'base']['arms']
        nav.append({'case':q['id'],'target_present':bool(targets),'target_ids':sorted(targets),'relative_target_top1':bool(targets and orders['relative']['order_ids'][0] in targets)})
    result.update(stability=stability,navigation=nav,natural_long_pools=sum(len(p['cards'])>40 for p in acquisition['pools'].values()))
    for assessor in ('A','B'):
        records=[]
        for q in protocol['primary_cases']:
            rows=references[assessor][q];cards=acquisition['pools'][q]['cards'];assert set(rows)=={c['id'] for c in cards}
            for row in rows.values():
                assert type(row['lead'])is int and 0<=row['lead']<=3
            orders=bykey[q,'base']['arms'];metrics={arm:{'ndcg10':ndcg(orders[arm]['order_ids'],rows),'useful10':useful(orders[arm]['order_ids'],rows),'covered_facets':sorted(covered(orders[arm]['order_ids'],rows))} for arm in QUALITY_ARMS}
            delta=metrics['relative']['ndcg10']-metrics['w0']['ndcg10']
            lost=sorted(set(metrics['w0']['covered_facets'])-set(metrics['relative']['covered_facets']))
            records.append({'case':q,'metrics':metrics,'relative_minus_w0':delta,'lost_w0_facets':lost,'per_case_passed':delta>=-.03-1e-12 and not lost and metrics['relative']['useful10']>=metrics['w0']['useful10']})
        deltas=[r['relative_minus_w0'] for r in records];mean=statistics.mean(deltas);ci=interval(deltas)
        result['assessors'][assessor]={'cases':records,'mean_relative_minus_w0':mean,'bootstrap95':ci,'quality_gate_passed':mean>=.05-1e-12 and ci[0]>0 and all(r['per_case_passed'] for r in records)}
    result['quality_analysis_performed']=run.get('synthetic')is False
    result['development_quality_passed']=run.get('synthetic')is False and all(r['quality_gate_passed'] for r in result['assessors'].values()) and all(r['passed'] for r in stability)
    result['navigation_gate_met']=all(r['relative_target_top1'] for r in nav if r['target_present'])
    result['acquisition_recall_met']=all(r['target_present'] for r in nav)
    result['development_all_gates_passed']=result['development_quality_passed'] and result['navigation_gate_met'] and result['acquisition_recall_met']
    result['outcome']='synthetic_fixture_only' if run.get('synthetic')is True else 'development_pass' if result['development_all_gates_passed'] else 'development_quality_pass_navigation_unmet' if result['development_quality_passed'] else 'development_rejected'
    return result
