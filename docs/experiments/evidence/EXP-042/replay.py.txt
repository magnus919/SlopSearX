def _load(name: str, path: pathlib.Path, expected: str | None = None):
    raw = path.read_bytes()
    digest = hashlib.sha256(raw).hexdigest()
    if expected and digest != expected: raise ValueError(f"pinned source changed: {name} {digest}")
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_loader(name, loader)
    mod = importlib.util.module_from_spec(spec); loader.exec_module(mod); return mod

def group_pool(pool_id,cards,query):
    """Return post-v3 representatives and complete reversible membership map."""
    from slopsearx.adapter import SearchResult
    from slopsearx.scholarly import group_publications
    feed=[SearchResult(url=c["url"],title=c["title"],content=c["snippet"],engine="frozen_projection",engines={"frozen_projection"}) for c in cards]
    grouped=group_publications({"frozen_projection":feed},query)["frozen_projection"]
    original={c["id"]:c for c in cards}; result=[]; mapping={}; used=set()
    for ordinal,item in enumerate(grouped):
        members=(item.work_group or {}).get("members") or [{"original_position":next(i+1 for i,c in enumerate(cards) if c["id"] not in used and c["url"]==item.url and c["title"]==item.title and c["snippet"]==item.content)}]
        member_ids=[]
        for m in members:
            pos=m.get("original_position")
            if type(pos)is not int or not 1<=pos<=len(cards): raise ValueError(f"bad original position in {pool_id}")
            cid=cards[pos-1]["id"]; member_ids.append(cid)
        rep_candidates=[cid for cid in member_ids if original[cid]["url"]==item.url and original[cid]["title"]==item.title and original[cid]["snippet"]==item.content]
        if not rep_candidates: raise ValueError(f"representative is not an original row: {pool_id}")
        rep=rep_candidates[0]
        if rep in used: raise ValueError("duplicate representative")
        used.add(rep)
        row=dict(original[rep]); row["id"]=rep; result.append(row)
        gid=f"{pool_id}-g{ordinal:03d}"
        for cid in member_ids: mapping[cid]={"group_id":gid,"representative_id":rep}
    if set(mapping)!=set(original) or len(result)!=len(set(c["id"] for c in result)): raise ValueError(f"grouping membership failure: {pool_id}")
    return result,mapping

def wire(body): return json.dumps(body,ensure_ascii=False).encode("utf-8")

def request_eligibility(body,cards,cap,incumbent_ids):
    ids=[c["id"] for c in cards]; fallback=list(incumbent_ids)
    if len(ids)!=len(set(ids)) or len(fallback)!=len(ids) or set(fallback)!=set(ids): raise ValueError("fallback must be exact full membership")
    reasons=[]
    if len(cards)>MAX_CARDS: reasons.append("candidate_count")
    if len(body)>cap: reasons.append("request_bytes")
    return {"eligible":not reasons,"reasons":reasons,"request_bytes":len(body),"cap":cap,"fallback_order_ids":None if not reasons else fallback}

def req_body(contract, task, cards, facets=None, mixed=False):
    questions={}
    for i,c in enumerate(cards):
        qid=f"score-{i:03d}" if mixed else c["id"]
        questions[qid]={"type":"score","instructions":contract["instructions"].replace("{id}",c["id"]),"criteria":list(contract["criteria"])}
        if mixed:
            for j,_f in enumerate(facets):
                nid=f"noul-{i:03d}-{j:02d}"
                questions[nid]={"type":"noul","instructions":NUOL_TEMPLATE.format(candidate_id=c["id"],facet_index=j)}
    state={"query":task,"candidates":cards}
    if mixed: state["requested_facets"]=facets
    return {"model":MODEL,"state":state,"questions":questions}

def expected_specs(root,data,grouped,contracts):
    frozen,contexts,facets,pools,nav,baseline,selection,*_=data; specs=[]
    def add(rid,pool,kind,task,cards,scorecards=None,fs=None,role=None,repeat=None,target=None,rotated=False):

        cap=CAP_W0 if kind=="W0" else CAP_E

        for c in cards: H.validate_card(c)
        decision=request_eligibility(wire(body),cards,cap,[c["id"] for c in cards])
        if not decision["eligible"]: raise ValueError(f"request over registered limit: {rid}: {decision['reasons']}")
        qids=[(f"score-{i:03d}" if kind=="E" else c["id"]) for i,c in enumerate(scorecards)]
        if mixed: qids += [f"noul-{i:03d}-{j:02d}" for i in range(len(scorecards)) for j in range(len(fs))]
        specs.append({"id":rid,"pool_id":pool,"kind":kind,"role":role or kind,"task":task,"facets":fs or [],"candidate_ids":[c["id"] for c in cards],"score_ids":[c["id"] for c in scorecards],"score_question_ids":[f"score-{i:03d}" for i in range(len(scorecards))],"question_ids":qids,"request_cap":cap,"target_id":target,"repeat_of":repeat,"body":wire(body)})
    # Software pools are grouped before every arm; W0 scores prefix 40 and appends untouched tail.
    for q in SOFTWARE:
        cards=grouped[q]
        add(f"{q}-w0",q,"W0",contexts[q]["keyword"],cards,cards[:40],role="W0")
        add(f"{q}-e",q,"E",contexts[q]["purpose"],cards,cards,facets[q],role="E")
    for p in EXTENDED:
        cards=grouped[p]; task=pools[p]["query"]; purpose=pools[p]["purpose"]
        add(f"{p}-w0",p,"W0",task,cards,cards[:40],role="W0")
        add(f"{p}-d",p,"D",purpose,cards,cards,role="D")
        add(f"{p}-e",p,"E",purpose,cards,cards,facets[p],role="E")
        add(f"{p}-repeat",p,"E",purpose,cards,cards,facets[p],role="exact_repeat",repeat=f"{p}-e")
        rotated=cards[20:]+cards[:20]
        add(f"{p}-rotated20",p,"E",purpose,rotated,rotated,facets[p],role="rotated20",repeat=f"{p}-e",rotated=True)

    for row in nav:
        q,cid=row["query_id"],row["candidate_id"]; cards=grouped[q]
        # exact target rows survive grouping; their representative ID remains the original ID.
        target=next((c["id"] for c in cards if c["url"]==row["expected_target_url"]),None)

        for kind,task in (("W0",row["query"]),("D",row["query"])):
            add(f"nav-{q}-{cid}-{kind.lower()}",q,kind,task,cards,cards,role="navigation",target=cid)
    if len(specs)!=CALL_COUNT or len({s["id"] for s in specs})!=CALL_COUNT: raise ValueError("plan must be exactly 48 calls")
    return specs

def stable_scores(scores,ids): return sorted(ids,key=lambda cid:(-scores[cid],ids.index(cid)))

def selector(scores,nouls,facets,ids):
    ranked=stable_scores(scores,ids); k=min(10,len(ids)); selected=ranked[:k]; reserved=[]; insertions=[]
    def qualified(cid,f): return scores[cid]>=5 and nouls[cid][f]>=.8
    while len(insertions)<4:
        covered={fi for fi in range(len(facets)) if any(qualified(cid,fi) for cid in selected)}
        outsiders=[cid for cid in ranked if cid not in selected and any(qualified(cid,fi) for fi in range(len(facets)) if fi not in covered)]
        if not outsiders: break
        def add_count(cid): return sum(qualified(cid,fi) for fi in range(len(facets)) if fi not in covered)
        chosen=sorted(outsiders,key=lambda cid:(-add_count(cid),-scores[cid],ids.index(cid)))[0]
        evictable=[]
        for cid in selected:
            if cid in reserved: continue
            if any(qualified(cid,fi) and sum(qualified(other,fi) for other in selected)==1 for fi in range(len(facets))): continue
            evictable.append(cid)
        if not evictable: break
        evict=sorted(evictable,key=lambda cid:(scores[cid],-ids.index(cid)))[0]
        selected.remove(evict); selected.append(chosen); reserved.append(chosen); insertions.append({"inserted":chosen,"evicted":evict,"new_facets":[facets[i]["id"] for i in range(len(facets)) if i not in covered and qualified(chosen,i)]})
    out=sorted(selected,key=lambda cid:(-scores[cid],ids.index(cid)))+[cid for cid in ranked if cid not in selected]
    if len(out)!=len(ids) or set(out)!=set(ids): raise ValueError("selector membership failure")
    return out,insertions

def validate_mixed(data,record):
    errors=[]
    if type(data)is not dict or set(data)!={"model","answers","usage","_elapsed_ms"}: return {},{},["envelope"]
    if data["model"]!=MODEL: errors.append("model")
    usage=data["usage"]
    if type(usage)is not dict or set(usage)!={"input_tokens","output_tokens"} or any(type(v)is not int or v<0 for v in usage.values()): errors.append("usage")
    ans=data["answers"]; expected=set(record["question_ids"])
    if type(ans)is not dict or set(ans)!=expected: return {},{},errors+["answer_membership"]
    scores={}; nouls={cid:{} for cid in record["candidate_ids"]}
    for i,cid in enumerate(record["score_ids"]):
        qid=f"score-{i:03d}" if record["kind"]=="E" else cid; a=ans[qid]
        if type(a)is not dict or not {"type","score"}<=set(a) or set(a)-{"type","score","confidence","legend","probabilities"} or a.get("type")!="score": errors.append("score_type:"+qid); continue
        v=a.get("score")
        if isinstance(v,bool) or not isinstance(v,(int,float)) or not math.isfinite(v) or not 0<=v<=9: errors.append("score_value:"+qid); continue
        # Reuse documented optional Score metadata validation from pinned helper.
        sub={"model":MODEL,"answers":{cid:a}}
        parsed,es=H.validate_response(sub,{"candidate_ids":[cid]})
        if es: errors.extend(es); continue
        scores[cid]=float(v)
        if record["kind"]=="E":
            for j,_ in enumerate(record["facets"]):
                nid=f"noul-{i:03d}-{j:02d}"; n=ans[nid]
                if type(n)is not dict or set(n)!={"type","noul"} or n.get("type")!="noul": errors.append("noul_type:"+nid); continue
                nv=n.get("noul")
                if isinstance(nv,bool) or not isinstance(nv,(int,float)) or not math.isfinite(nv) or not 0<=nv<=1: errors.append("noul_value:"+nid); continue
                nouls[cid][j]=float(nv)
    if len(scores)!=len(record["score_ids"]): errors.append("score_missing")
    if record["kind"]=="E" and any(len(v)!=len(record["facets"]) for v in nouls.values()): errors.append("noul_missing")
    if errors: return {},{},errors
    return scores,nouls,errors

def read_receipts(root,manifest,require_all=False):
    out={}; errors=[]
    for req in manifest["requests"]:
        p=root/"receipts"/(req["id"]+".json")
        if not p.exists(): errors.append("missing:"+req["id"]); continue

        except Exception: errors.append("malformed:"+req["id"]); continue
        allowed={"schema","id","request_sha256","model","answers","usage","remote_http_ms","transport_wall_ms","transport_exit_code","response_bytes","response_sha256","fallback_order_ids","status","validation_errors","error"}
        if type(obj)is not dict or set(obj)!=allowed or obj["schema"]!="exp042-receipt/1" or obj["id"]!=req["id"] or obj["request_sha256"]!=req["request_sha256"] or obj["status"]!="valid" or obj["validation_errors"]!=[]:
            errors.append("receipt_identity:"+req["id"]); continue
        data={"model":obj["model"],"answers":obj["answers"],"usage":obj["usage"],"_elapsed_ms":obj["remote_http_ms"]}
        scores,nouls,es=validate_mixed(data,req)

        except Exception: raw_data=None
        raw_elapsed=raw_data.get("_elapsed_ms") if type(raw_data)is dict else None
        _,_,raw_errors=validate_mixed(raw_data,req) if type(raw_data)is dict else ({},{},["envelope"])
        raw_elapsed_valid=not isinstance(raw_elapsed,bool) and isinstance(raw_elapsed,(int,float)) and math.isfinite(raw_elapsed) and 0<=raw_elapsed<=MAX_REMOTE_MS
        raw_matches=(type(raw_data)is dict and not raw_errors and raw_elapsed_valid and raw_data.get("model")==obj["model"] and raw_data.get("answers")==obj["answers"] and raw_data.get("usage")==obj["usage"] and raw_elapsed==remote)
        if (not raw_matches or es or inp is None or outtok is None or isinstance(remote,bool) or not isinstance(remote,(int,float)) or not math.isfinite(remote) or remote<0 or remote>MAX_REMOTE_MS or isinstance(wall,bool) or not isinstance(wall,(int,float)) or not math.isfinite(wall) or wall<0 or wall>MAX_WALL*1000 or type(obj["transport_exit_code"]) is not int or obj["transport_exit_code"]!=0 or type(obj["response_bytes"]) is not int or obj["response_bytes"]<0 or obj["response_bytes"]>2_000_000 or not isinstance(obj["response_sha256"],str) or not re.fullmatch(r"[0-9a-f]{64}",obj["response_sha256"]) or raw is None or len(raw)!=obj["response_bytes"] or sha(raw)!=obj["response_sha256"]):
            errors.append("invalid_receipt:"+req["id"]); continue

def read_history(root,manifest,refhash):
    p=root/"attempts.jsonl"
    if not p.exists(): return set(),set(),None
    events=[strict_json(x) for x in p.read_bytes().splitlines() if x]
    if any(type(e)is not dict for e in events): raise ValueError("attempt ledger event must be an object")
    starts=[e for e in events if e.get("event")=="run_started"]
    if len(starts)!=1 or starts[0].get("reference_hashes")!=refhash or starts[0].get("prepared_manifest_sha256")!=sha((root/"prepared-manifest.json").read_bytes()): raise ValueError("run history is not bound to frozen refs/manifest")
    if set(starts[0])!={"event","prepared_manifest_sha256","reference_hashes","at_utc"}: raise ValueError("malformed run start event")
    H.parse_run_start_time(starts[0].get("at_utc")); begun=set(); done=set(); terminal=False; byid={r["id"]:r for r in manifest["requests"]}; start_events={}; finish_events={}
    for e in events:
        event=e.get("event")
        if event not in {"run_started","attempt_started","attempt_finished","runner_stopped"}: raise ValueError("unknown attempt ledger event")
        if event=="run_started": continue
        if event=="attempt_started":
            rid=e.get("request_id")
            allowed={"event","attempt","request_id","request_sha256","reference_hashes","preflight_input_token_estimate","preflight_output_token_estimate","preflight_is_heuristic","at_utc"}
            if set(e)!=allowed or e.get("reference_hashes")!=refhash or terminal or rid not in byid or rid in begun or e.get("attempt")!=1 or e.get("request_sha256")!=byid[rid]["request_sha256"]: raise ValueError("duplicate/unknown/mismatched attempt")
            if begun-done: raise ValueError("attempts are not serial")
            begun.add(rid); start_events[rid]=e
        if event=="attempt_finished":
            rid=e.get("request_id")
            allowed={"event","attempt","request_id","request_sha256","status","error","input_tokens","output_tokens","at_utc"}
            if set(e)!=allowed or rid not in begun or rid in done or e.get("attempt")!=1 or e.get("request_sha256")!=byid[rid]["request_sha256"] or e.get("status") not in {"valid","invalid"} or (e.get("status")=="valid")!=(e.get("error") is None) or any(v is not None and (type(v)is not int or v<0) for v in (e.get("input_tokens"),e.get("output_tokens"))): raise ValueError("invalid attempt completion ledger")
            done.add(rid)
            finish_events[rid]=e
            rp=root/"receipts"/(rid+".json")
            if not rp.is_file(): raise ValueError("completed ledger attempt has no durable receipt")
            rec=load_json(rp)
            if type(rec)is not dict or rec.get("request_sha256")!=e["request_sha256"] or rec.get("status")!=e["status"] or rec.get("error")!=e["error"]: raise ValueError("receipt and attempt completion differ")
            usage=rec.get("usage")
            if e["status"]=="valid":
                if type(usage)is not dict or e.get("input_tokens")!=usage.get("input_tokens") or e.get("output_tokens")!=usage.get("output_tokens"): raise ValueError("valid attempt usage differs from receipt")
            elif e.get("input_tokens") is not None or e.get("output_tokens") is not None:
                if type(usage)is not dict or e.get("input_tokens")!=usage.get("input_tokens") or e.get("output_tokens")!=usage.get("output_tokens"): raise ValueError("invalid attempt usage differs from receipt")
            if e["status"]!="valid": terminal=True
        if event=="runner_stopped":
            allowed_shapes=(
                {"event","reason","at_utc"},
                {"event","request_id","reason","at_utc"},
                {"event","request_id","reason","fallback_order_ids","at_utc"},
                {"event","request_id","reason","input_estimate","output_estimate","at_utc"},
            )
            if set(e) not in allowed_shapes or not isinstance(e.get("reason"),str) or not e["reason"]: raise ValueError("malformed stop event")
            terminal=True
    if terminal: raise ValueError("terminal stop recorded; never resume/retry this study")
    if begun-done: raise ValueError("uncertain prior attempt; no retry is allowed")
    return begun,done,starts[0]["at_utc"]

def ranking(req,receipt):
    scores=receipt["scores"]; ids=req["score_ids"]; score_order=stable_scores(scores,ids)
    if req["kind"]=="W0":
        return score_order+[cid for cid in req["candidate_ids"] if cid not in set(ids)]
    if req["kind"]!="E": return score_order
    if len(req["facets"])<1 or len(req["facets"])>4: return list(req["candidate_ids"])
    selected,diag=selector(scores,receipt["nouls"],req["facets"],ids)
    return selected
