    def test_selector_thresholds_multicoverage_ties_reservation_and_membership(self):
        ids=[f"c{i}" for i in range(12)]; facets=[{"id":f"f{i}","description":"f"} for i in range(4)]
        scores={cid:float(9-i/10) for i,cid in enumerate(ids)}
        n={cid:{j:0.0 for j in range(4)} for cid in ids}
        self.assertEqual(R.selector(scores,n,facets,ids)[0],R.stable_scores(scores,ids))
        n["c10"]={0:.8,1:.8,2:.8,3:.8}
        out,ins=R.selector(scores,n,facets,ids)
        self.assertEqual(len(out),len(ids)); self.assertEqual(set(out),set(ids)); self.assertLessEqual(len(ins),4)
        self.assertEqual(ins[0]["inserted"],"c10")
        self.assertEqual(ins[0]["new_facets"],["f0","f1","f2","f3"])
        scores2=dict(scores); scores2["c10"]=5.0; n2={cid:{j:0.0 for j in range(4)} for cid in ids}; n2["c10"][0]=.8
        self.assertIn("c10",R.selector(scores2,n2,facets,ids)[0][:10])
        scores3=dict(scores); scores3["c10"]=4.999
        self.assertNotIn("c10",R.selector(scores3,n2,facets,ids)[0][:10])
        n3={cid:{j:0.0 for j in range(4)} for cid in ids}; n3["c10"][0]=.7999
        self.assertNotIn("c10",R.selector(scores2,n3,facets,ids)[0][:10])
