def load(name,path):
    loader=importlib.machinery.SourceFileLoader(name,str(path));spec=importlib.util.spec_from_loader(name,loader);mod=importlib.util.module_from_spec(spec);loader.exec_module(mod);return mod

def card(i):return {"id":"x%02d"%i,"url":"https://example.test/%02d"%i,"title":"Example source %02d"%i,"snippet":"A bounded public metadata excerpt %02d."%i}
def case():return {"id":"q1","query":"ordinary query","purpose":"Find strong public reading leads for this task.","facets":[{"id":"f1","description":"implementation"},{"id":"f2","description":"validation"}]}
def response(request,*,usage=True,noul=.5,score=5,metadata=True):
    answers={}
    for qid,q in request["questions"].items():
        if q["type"]=="score":answers[qid]={"type":"score","score":score,"confidence":.8}
        else:answers[qid]={"type":"noul","noul":noul,**({"confidence":.01,"ignored":"metadata"} if metadata else {})}
    data={"model":P.MODEL,"answers":answers}
    if usage:data["usage"]={"input_tokens":11,"output_tokens":3}
    return json.dumps(data,allow_nan=False,separators=(",",":")).encode()

class PrepareTests(unittest.TestCase):
    def test_binary_request_has_one_canonical_card_local_question_per_full_pool_id(self):
        cards=[card(i) for i in range(80)];built=P.bodies(case(),cards)
        req=built["binary"]
        self.assertEqual(req["state"],{"purpose":case()["purpose"],"requested_facets":case()["facets"]})
        self.assertEqual(list(req["questions"]),built["ids"])
        self.assertEqual(len(req["questions"]),80)
        for cid,q in req["questions"].items():
            self.assertEqual(set(q),{"type","instructions","criteria"});self.assertEqual(q["type"],"noul")
            self.assertEqual(q["instructions"],{"candidate":next(c for c in built["cards"] if c["id"]==cid),"question":P.PROTOCOL["instructions"]})
            self.assertEqual(q["criteria"],P.PROTOCOL["criteria"])
        self.assertNotIn("candidates",req["state"])

    def test_canonical_binary_body_unchanged_by_input_rotation_and_w0_keeps_native_tail(self):
        cards=[card(i) for i in range(59)];base=P.bodies(case(),cards);rot=P.bodies(case(),cards,"rotate")
        self.assertEqual(base["binary"],rot["binary"])
        self.assertEqual(rot["provenance_order_ids"],base["provenance_order_ids"][20:]+base["provenance_order_ids"][:20])
        self.assertEqual(base["w0"]["state"]["query"],case()["query"])
        self.assertEqual(len(base["w0"]["questions"]),40)
        self.assertEqual(base["tail_ids"],["x%02d"%i for i in range(40,59)])

    def test_binary_partition_preserves_exact_half_positions_and_incumbent_ties(self):
        incumbent=["a","b","c","d","e","f"]
        probs={"a":.5,"b":.91,"c":.12,"d":.72,"e":.5,"f":.31}
        self.assertEqual(P.binary_order(incumbent,probs),["a","b","d","c","e","f"])
        self.assertEqual(P.binary_order(incumbent,{"a":.5,"b":.5000001,"c":.4999999,"d":.5,"e":.5,"f":.5}),["a","b","c","d","e","f"])
        all_neutral={cid:.5 for cid in incumbent};self.assertEqual(P.binary_order(incumbent,all_neutral),incumbent)
        self.assertEqual(P.binary_order(incumbent,{cid:.8 for cid in incumbent}),incumbent)
        self.assertEqual(P.binary_order(incumbent,{cid:.2 for cid in incumbent}),incumbent)

    def test_w0_score_reconstruction_and_binary_answer_metadata(self):
        built=P.bodies(case(),[card(i) for i in range(45)])
        scores=P.validate_score_response(response(built["w0"]),built["w0"],list(built["w0"]["questions"]))
        self.assertEqual(P.order_scores(scores["scores"],list(built["w0"]["questions"])),list(built["w0"]["questions"]))
        actual=P.validate_response(response(built["binary"],noul=.5),built["binary"],built["ids"])
        self.assertEqual(set(actual["probabilities"]),set(built["ids"]));self.assertEqual(actual["usage"],{"input_tokens":11,"output_tokens":3})

    def test_strict_noul_schema_rejects_bad_inventory_bool_nonfinite_duplicate_and_oversize(self):
        built=P.bodies(case(),[card(0),card(1)]);body=built["binary"]
        with self.assertRaises(ValueError):P.validate_response(response(body).replace(P.MODEL.encode(),b"wrong"),body,built["ids"])
        with self.assertRaises(ValueError):P.validate_response(response(body,noul=True),body,built["ids"])
        bad=response(body).replace(b'"noul":0.5',b'"noul":NaN')
        with self.assertRaises(ValueError):P.validate_response(bad,body,built["ids"])
        duplicate=b'{"model":"jev-1.13.0","model":"jev-1.13.0","answers":{}}'
        with self.assertRaisesRegex(ValueError,"duplicate-json-key"):P.validate_response(duplicate,body,built["ids"])
        with self.assertRaises(ValueError):P.validate_response(b" "*2_000_001,body,built["ids"])
        bad_data=json.loads(response(body));bad_data["answers"].pop(built["ids"][0])
        with self.assertRaisesRegex(ValueError,"answer-membership"):P.validate_response(json.dumps(bad_data).encode(),body,built["ids"])

    def test_known_usage_is_recovered_even_when_answer_is_invalid(self):
        built=P.bodies(case(),[card(0),card(1)]);raw=response(built["binary"],noul=1.5)
        self.assertEqual(P.usage_from_raw(raw),{"input_tokens":11,"output_tokens":3})
        with self.assertRaisesRegex(ValueError,"noul-range"):P.validate_response(raw,built["binary"],built["ids"])
        self.assertIsNone(P.validate_response(response(built["binary"],usage=False),built["binary"],built["ids"])["usage"])

    def test_rejects_bad_card_id_binding_and_size_before_transport(self):
        built=P.bodies(case(),[card(0),card(1)]);request=json.loads(P.wire(built["binary"]))
        first=next(iter(request["questions"]));request["questions"][first]["instructions"]["candidate"]["id"]="other"
        with self.assertRaisesRegex(ValueError,"binary-binding"):P.validate_request(P.wire(request))
        with self.assertRaises(ValueError):P.bodies(case(),[card(i) for i in range(81)])

    def test_http_boundary_rejects_empty_or_oversized_w0_card_fields(self):
        built=P.bodies(case(),[card(0),card(1)])
        for field,value in (("url",""),("url","u"*513),("title","t"*257),("snippet","s"*1201),("id","i"*81)):
            request=json.loads(P.wire(built["w0"]));request["state"]["candidates"][0][field]=value
            with self.subTest(field=field,length=len(value)),self.assertRaises(ValueError):P.validate_request(P.wire(request))

    def test_invalid_binary_answer_with_known_usage_fails_once_and_falls_back_to_fresh_w0(self):

        class FakeOwned:

                FakeOwned.calls+=1;request=P.strict_json(body)
                if FakeOwned.calls==1:self.last_raw=response(request,noul=.5);return self.last_raw,1.0
                if FakeOwned.calls==2:self.last_raw=response(request,score=5);return self.last_raw,1.0
