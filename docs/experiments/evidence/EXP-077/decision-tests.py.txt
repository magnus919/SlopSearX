def arm(status='complete'):
 return {'facet_assessments':[{'facet_id':'f1','status':status,'scope_match':'match','entailment':'supports','applicability':'applicable','unsupported_claim_indices':[],'unsupported_material_claims':[],'missed_qualifications':[],'claims':[{'references':['verified-evidence-fixture']}],'evidence_dispositions':[{'disposition':'used_with_support'},{'disposition':'not_acquired'}]}]}
class DecisionTests(unittest.TestCase):
 def test_identity_is_only_diagnostic(self):
  rows={str(i):{'lead':2,'facets':['f1']} for i in range(12)}
  r=m.ranking_case(list(rows),list(reversed(rows)),rows);self.assertTrue(r['passes']);self.assertEqual(len(r['identity_displaced_diagnostic']),2)
 def test_whole_pool_membership(self):
  with self.assertRaises(ValueError):m.ranking_case(['a'],['a','a'],{'a':{'lead':2,'facets':[]}})
 def test_positive_mean_not_enough(self):
  r=m.ranking_endpoint([{'delta':x,'passes':True} for x in [-.03]*4+[.08]*4]);self.assertGreater(r['mean_delta'],.02);self.assertFalse(r['passes'])
 def test_effect_bar(self):
  self.assertTrue(m.ranking_endpoint([{'delta':.021,'passes':True}]*8)['passes'])
  self.assertFalse(m.ranking_endpoint([{'delta':.019,'passes':True}]*8)['passes'])
 def test_two_additional(self):
  p=[(str(i),arm('partial') if i<2 else arm(),arm()) for i in range(8)];r=m.task_use_endpoint(p);self.assertTrue(r['passes']);self.assertEqual(r['wins'],2)
 def test_regression_cannot_be_averaged(self):
  p=[(str(i),arm('partial') if i<3 else arm(),arm('partial') if i==7 else arm()) for i in range(8)];r=m.task_use_endpoint(p);self.assertEqual(r['additional_completed'],2);self.assertFalse(r['passes'])
 def test_no_anchor_cannot_self_certify(self):
  a=arm();a['facet_assessments'][0]['claims']=[];self.assertFalse(m.answer_endpoint(a)['complete'])
 def test_missing_capture_not_automatic_unknown(self):self.assertTrue(m.answer_endpoint(arm())['complete'])
 def test_unknown_critical_prevents_pass(self):
  a=arm();a['facet_assessments'][0]['entailment']='uncertain';r=m.answer_endpoint(a);self.assertFalse(r['complete']);self.assertEqual(r['unknown_critical_facets'],['f1'])
 def test_material_unsupported_prevents_completion(self):
  a=arm();a['facet_assessments'][0]['unsupported_claim_indices']=[0];r=m.answer_endpoint(a);self.assertFalse(r['complete']);self.assertTrue(r['material_failures'])
 def test_independent_endpoints_not_pooled(self):
