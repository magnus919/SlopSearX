"""Offline contract and gate tests for the EXP-060 analysis adapter."""
from __future__ import annotations

import copy
import importlib.machinery
import json
import unittest
from pathlib import Path

HERE = Path(__file__).resolve().parent
analysis = importlib.machinery.SourceFileLoader(
    "exp060_analysis_tests", str(HERE / "analyze.py.txt")
).load_module()
protocol = json.loads((HERE / "protocol.json").read_text(encoding="utf-8"))
cohort = json.loads((HERE / "queries.json").read_text(encoding="utf-8"))
questions = {item["id"]: item for item in cohort}


def _cards(qid: str, count: int) -> list[dict[str, str]]:
    question = questions[qid]
    rows = []
    for i in range(count):
        url = question.get("target_url", f"https://example.invalid/{qid}/{i}")
        rows.append({"id": f"{qid}-c{i}", "title": f"card {i}", "url": url if question["kind"] == "navigation" and i == 0 else f"{url}/{i}", "snippet": "visible fixture"})
    if question["kind"] == "navigation":
        rows[0]["url"] = question["target_url"]
    return rows


def fixture() -> dict:
    pools = {}
    pool_sizes = {qid: (41 if qid in {"q1", "q4", "q7", "q8"} else 12) for qid in questions}
    for qid, question in questions.items():
        cards = _cards(qid, pool_sizes[qid])
        pools[qid] = {"kind": question["kind"], "cards": cards}
        if question["kind"] == "navigation":
            pools[qid]["target_url"] = question["target_url"]

    references = {reviewer: {} for reviewer in ("A", "B")}
    facets_by_q = {
        qid: [facet["id"] for facet in questions[qid].get("facets", [])]
        for qid in (*analysis.PRIMARY, *analysis.NAVIGATION)
    }
    for reviewer in ("A", "B"):
        for qid in (*analysis.PRIMARY, *analysis.NAVIGATION):
            facets = facets_by_q[qid]
            rows = {}
            for i, card in enumerate(pools[qid]["cards"]):
                tags = [facets[i]] if i < len(facets) else []
                rows[card["id"]] = {
                    "lead": 0 if qid in analysis.NAVIGATION else (3 if i < 2 else (2 if i == 2 else 0)),
                    "visible": 0 if qid in analysis.NAVIGATION else (2 if i < 3 else 0),
                    "facets": tags,
                    "uncertain": False,
                    "rationale": "visible fixture evidence",
                }
            references[reviewer][qid] = rows

    operations = []
    for planned in protocol["ordered_case_plan"]:
        opid = planned["id"]
        qid, mode = opid.split("-", 1)
        base = [card["id"] for card in pools[qid]["cards"]]
        cards = copy.deepcopy(pools[qid]["cards"])
        if mode == "rotate20":
            shift = 20 % len(cards)
            cards = cards[shift:] + cards[:shift]
        input_ids = [card["id"] for card in cards]
        if qid in analysis.NAVIGATION:
            target_id = next(card["id"] for card in pools[qid]["cards"] if card["url"] == questions[qid]["target_url"])
            order = [target_id] + [cid for cid in base if cid != target_id]
            orders = {"w0": order, "d": order[:], "e": order[:]}
            attempts = [
                {"arm": "w0", "remote_http_ms": 250, "status": "valid"},
                {"arm": "d", "remote_http_ms": 250, "status": "valid"},
            ]
            remote_dfe = 250.0
            terminal = "navigation_bypass"
        else:
            fids = facets_by_q[qid]
            # The incumbent covers two useful facets; E moves the third useful
            # reading lead into the top ten without displacing a useful card.
            w0 = [f"{qid}-c0", f"{qid}-c1"] + [f"{qid}-c{i}" for i in range(3, 11)] + [f"{qid}-c2"] + [f"{qid}-c{i}" for i in range(11, len(base))]
            e = [f"{qid}-c0", f"{qid}-c1", f"{qid}-c2"] + [f"{qid}-c{i}" for i in range(3, 10)] + [f"{qid}-c{i}" for i in range(10, len(base))]
            orders = {"w0": w0, "d": w0[:], "e": e}
            attempts = [
                {"arm": "w0", "remote_http_ms": 250, "status": "valid"},
                {"arm": "d", "remote_http_ms": 250, "status": "valid"},
                {"arm": "f", "remote_http_ms": 250, "status": "valid"},
                *[{"arm": "e", "round": round_no, "remote_http_ms": 100, "status": "valid"} for round_no in range(1, 5)],
            ]
            remote_dfe = 900.0
            terminal = "round_limit"
        operations.append({
            "id": opid,
            "case": qid,
            "mode": mode,
            "cards": cards,
            "orders": orders,
            "status": "valid",
            "terminal_reason": terminal,
            "remaining_remote_ms": 1100.0,
            "remote_d_f_e_ms": remote_dfe,
            "attempts": attempts,
        })
    return {
        "schema": "exp060-run-artifacts/1",
        "reference_source": "fresh_exp060",
        "historical_labels_reused": False,
        "pools": pools,
        "operations": operations,
        "references": references,
    }


class AnalysisAdapterTests(unittest.TestCase):
    def test_controlled_complete_fixture_passes_unchanged_gates(self):
        result = analysis.analyze(fixture())
        self.assertEqual(result["outcome"], "development_confirmation_passed")
        self.assertFalse(result["adoption_eligible"])
        self.assertTrue(all(result["gates"].values()))
        self.assertEqual(result["natural_primary_pools_over_40"], ["q1", "q4", "q7", "q8"])
        for reviewer in ("A", "B"):
            self.assertGreaterEqual(result["references"][reviewer]["primary_E_minus_W0"]["mean_gain"], 0.03)
            self.assertEqual(result["references"][reviewer]["primary_E_minus_W0"]["bootstrap"]["draws"], 10000)
            self.assertEqual(result["references"][reviewer]["primary_E_minus_W0"]["bootstrap"]["seed"], 91940)

    def test_missing_operation_is_inconclusive_not_a_reduced_sample(self):
        run = fixture()
        run["operations"][-1]["status"] = "not-invoked"
        result = analysis.analyze(run)
        self.assertEqual(result["outcome"], "inconclusive")
        self.assertFalse(result["quality_analysis_performed"])

    def test_reference_inventory_must_cover_every_projected_card(self):
        run = fixture()
        run["references"]["A"]["q1"].pop("q1-c40")
        result = analysis.analyze(run)
        self.assertEqual(result["outcome"], "inconclusive")

    def test_pool_cannot_exceed_frozen_eighty_card_cap(self):
        run = fixture()
        run["pools"]["q2"]["cards"] = [
            {"id": f"q2-extra-{i}", "title": "extra", "url": f"https://example.invalid/extra/{i}", "snippet": "fixture"}
            for i in range(81)
        ]
        self.assertEqual(analysis.analyze(run)["outcome"], "inconclusive")

    def test_boolean_grade_is_not_an_integer_grade(self):
        run = fixture()
        run["references"]["B"]["q2"]["q2-c0"]["lead"] = True
        self.assertEqual(analysis.analyze(run)["outcome"], "inconclusive")

    def test_unknown_facet_is_rejected_as_invalid_measurement(self):
        run = fixture()
        run["references"]["A"]["q3"]["q3-c0"]["facets"] = ["invented-facet"]
        self.assertEqual(analysis.analyze(run)["outcome"], "inconclusive")

    def test_duplicate_or_missing_order_identity_is_inconclusive(self):
        run = fixture()
        run["operations"][0]["orders"]["e"][0] = run["operations"][0]["orders"]["e"][1]
        self.assertEqual(analysis.analyze(run)["outcome"], "inconclusive")

    def test_reused_historical_reference_metadata_is_inconclusive(self):
        run = fixture()
        run["historical_labels_reused"] = True
        self.assertEqual(analysis.analyze(run)["outcome"], "inconclusive")

    def test_no_primary_uplift_rejects_instead_of_inconclusive(self):
        run = fixture()
        for op in run["operations"]:
            if op["case"] in analysis.PRIMARY:
                op["orders"]["e"] = op["orders"]["w0"][:]
        result = analysis.analyze(run)
        self.assertEqual(result["outcome"], "rejected")
        self.assertFalse(result["gates"]["both_reference_quality"])

    def test_full_thirteen_query_reference_inventory_includes_navigation_metadata(self):
        run = fixture()
        self.assertEqual(set(run["references"]["A"]), set((*analysis.PRIMARY, *analysis.NAVIGATION)))
        nav = next(op for op in run["operations"] if op["case"] == "q9")
        target = questions["q9"]["target_url"]
        self.assertEqual(run["pools"]["q9"]["target_url"], target)
        self.assertIn(target, [card["url"] for card in nav["cards"]])
        del run["references"]["B"]["q13"]
        self.assertEqual(analysis.analyze(run)["outcome"], "inconclusive")

    def test_two_useful_identities_can_be_replaced_when_net_count_and_facets_hold(self):
        run = fixture()
        # Exchange two useful identities for two useful identities with the
        # same facet support. EXP-042's frozen guard is net count, not identity.
        for reviewer in ("A", "B"):
            run["references"][reviewer]["q1"]["q1-c2"]["lead"] = 0
            for cid, facet_idx in (("q1-c11", 0), ("q1-c12", 1)):
                run["references"][reviewer]["q1"][cid]["lead"] = 2
                run["references"][reviewer]["q1"][cid]["facets"] = [questions["q1"]["facets"][facet_idx]["id"]]
        q1e = next(op for op in run["operations"] if op["id"] == "q1-base")["orders"]["e"]
        q1e[0], q1e[11] = q1e[11], q1e[0]
        q1e[1], q1e[12] = q1e[12], q1e[1]
        result = analysis.analyze(run)
        comparison = result["references"]["A"]["per_pool"]["q1"]["w0"]
        self.assertEqual(comparison["net_useful_loss"], 0)
        self.assertEqual(comparison["lost_useful_count"], 2)
        self.assertEqual(comparison["lost_useful_facets"], [])
        self.assertTrue(result["references"]["A"]["useful_retention_gate"])

    def test_navigation_must_preserve_exact_w0_and_rank_fresh_d_top3(self):
        run = fixture()
        nav = next(op for op in run["operations"] if op["case"] == "q9")
        nav["orders"]["d"][0], nav["orders"]["d"][3] = nav["orders"]["d"][3], nav["orders"]["d"][0]
        result = analysis.analyze(run)
        self.assertEqual(result["outcome"], "rejected")
        self.assertFalse(result["gates"]["five_navigation_targets"])

    def test_all_four_repeat_rotate_triples_are_required(self):
        run = fixture()
        op = next(op for op in run["operations"] if op["id"] == "q8-rotate20")
        op["status"] = "failed"
        result = analysis.analyze(run)
        self.assertEqual(result["outcome"], "inconclusive")

    def test_rotation_must_preserve_ids_and_apply_fixed_left_twenty_order(self):
        run = fixture()
        op = next(op for op in run["operations"] if op["id"] == "q4-rotate20")
        op["cards"] = op["cards"][1:] + op["cards"][:1]
        self.assertEqual(analysis.analyze(run)["outcome"], "inconclusive")

    def test_four_natural_large_pools_are_a_separate_conjunctive_gate(self):
        run = fixture()
        for qid in ("q1", "q4", "q7", "q8"):
            new_cards = run["pools"][qid]["cards"][:40]
            run["pools"][qid]["cards"] = new_cards
            keep = {card["id"] for card in new_cards}
            for op in run["operations"]:
                if op["case"] == qid:
                    qmode = op["mode"]
                    op["cards"] = copy.deepcopy(new_cards)
                    if qmode == "rotate20":
                        op["cards"] = op["cards"][20:] + op["cards"][:20]
                    op["orders"] = {arm: [cid for cid in order if cid in keep] for arm, order in op["orders"].items()}
            for reviewer in ("A", "B"):
                run["references"][reviewer][qid] = {cid: row for cid, row in run["references"][reviewer][qid].items() if cid in keep}
        result = analysis.analyze(run)
        self.assertEqual(result["outcome"], "inconclusive")
        self.assertFalse(result["gates"]["four_natural_primary_pools_over_40"])

    def test_valid_successful_budget_stop_is_analyzed(self):
        run = fixture()
        op = next(op for op in run["operations"] if op["id"] == "q2-base")
        op["attempts"] = [a for a in op["attempts"] if not (a["arm"] == "e" and a["round"] == 4)]
        op["remote_d_f_e_ms"] -= 100
        op["terminal_reason"] = "remote_budget_stop"
        op["remaining_remote_ms"] = 900
        self.assertTrue(analysis.analyze(run)["quality_analysis_performed"])

    def test_budget_stop_at_exact_full_deadline_is_invalid(self):
        run = fixture()
        op = next(op for op in run["operations"] if op["id"] == "q2-base")
        op["terminal_reason"] = "remote_budget_stop"
        op["remaining_remote_ms"] = 1000
        self.assertEqual(analysis.analyze(run)["outcome"], "inconclusive")

    def test_timing_aggregate_must_bind_to_owned_attempts(self):
        run = fixture()
        run["operations"][0]["remote_d_f_e_ms"] += 10
        self.assertEqual(analysis.analyze(run)["outcome"], "inconclusive")

    def test_per_call_deadline_is_enforced(self):
        run = fixture()
        run["operations"][0]["attempts"][1]["remote_http_ms"] = 1000.1
        run["operations"][0]["remote_d_f_e_ms"] += 750.1
        result = analysis.analyze(run)
        self.assertEqual(result["outcome"], "rejected")
        self.assertFalse(result["gates"]["fresh_candidate_remote_budget"])

    def test_candidate_cumulative_deadline_is_enforced(self):
        run = fixture()
        op = run["operations"][0]
        for attempt in op["attempts"]:
            if attempt["arm"] == "e":
                attempt["remote_http_ms"] = 400
        op["remote_d_f_e_ms"] = 250 + 250 + 4 * 400
        result = analysis.analyze(run)
        self.assertEqual(result["outcome"], "rejected")
        self.assertFalse(result["gates"]["fresh_candidate_remote_budget"])


if __name__ == "__main__":
    unittest.main(verbosity=2)
