def load_module(name: str, path: pathlib.Path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_loader(name, loader)
    mod = importlib.util.module_from_spec(spec)
    loader.exec_module(mod)
    return mod

def root_path(root=None):
    return pathlib.Path(root or HERE).resolve()


def _hash(path: pathlib.Path) -> str:
    return sha(path.read_bytes())


def wire(obj: Any) -> bytes:
    # Match the normal production serializer used by TypeSafe requests.
    return json.dumps(obj, ensure_ascii=False).encode("utf-8")


def verify_registered_inputs(root):
    root = root_path(root)
    registration = REPO / "docs/experiments/EXP-045-contextual-source-replacement.md"
    pins = load_json(root / "source-pins.json")
    integ = load_json(root / "registration-integrity.json")
    if type(integ) is not dict or integ.get("schema") != "exp045-registration-integrity/1":
        raise ValueError("EXP-045 registration integrity schema mismatch")
    expected_local = {"source-pins.json", "design-review.md", "registration-review.md"}
    files = integ.get("files")
    if type(files) is not dict or set(files) != expected_local:
        raise ValueError("EXP-045 registration file inventory mismatch")
    for name, digest in files.items():
        if _hash(root / name) != digest:
            raise ValueError(f"EXP-045 registration artifact hash mismatch: {name}")
    if _hash(registration) != integ.get("registration_sha256"):
        raise ValueError("EXP-045 registration contract hash mismatch")
    if integ.get("new_provider_calls") != 0 or integ.get("new_search_calls") != 0 or integ.get("candidate_implemented") is not False:
        raise ValueError("EXP-045 registration declares prior execution or implementation")
    if type(pins) is not dict or pins.get("schema") != "exp045-source-pins/1" or pins.get("baseline_revision") != "4616b0a8bfabf7f2823c9eace062ec882b9428dd" or set(pins.get("sources", {})) != EXPECTED_SOURCE_NAMES:
        raise ValueError("EXP-045 source pin inventory mismatch")
    if len(pins["sources"]) != EXPECTED_SOURCE_PINS:
        raise ValueError("EXP-045 source pin count mismatch")
    for rel, digest in pins["sources"].items():
        path = REPO / rel
        if not path.is_file() or _hash(path) != digest:
            raise ValueError(f"EXP-045 baseline source pin mismatch: {rel}")
    return pins, integ


def validate_baseline():
    # EXP-044 validates the existing EXP-043 receipts and returns references in A/B -> pool shape.
    baseline = BASE.validate_published_result()
    data = BASE.source_data()
    grouped = BASE.groups_in_memory(data)
    specs = BASE.specs_in_memory(data, grouped)
    if len(specs) != EXPECTED_OLD_REQUESTS:
        raise ValueError("pinned baseline request count mismatch")
    specmap = {s["id"]: s for s in specs}
    if len(specmap) != EXPECTED_OLD_REQUESTS:
        raise ValueError("pinned baseline request IDs are not unique")
    d_ids = [s["id"] for s in specs if s["kind"] == "D" and s["role"] != "navigation"]
    if len(d_ids) != EXPECTED_REAL_OPERATIONS:
        raise ValueError("expected exactly 21 substantive D variants")
    refs, refhashes = BASE.analysis_references(data, grouped)
    # Do not transpose again: the pinned EXP-043 helper already returns A/B -> pool -> card.
    if set(refs) != {"A", "B"} or any(set(refs[r]) != set(SOFTWARE) | set(NATURAL) for r in refs):
        raise ValueError("sealed reference shape mismatch")
    return {"baseline": baseline, "data": data, "grouped": grouped, "specs": specs, "specmap": specmap, "refs": refs, "refhashes": refhashes}


def _stable_score_order(spec, rec):
    ids = list(spec["candidate_ids"])
    scores = rec.get("scores")
    if type(scores) is not dict or set(scores) != set(ids):
        raise ValueError("D score membership mismatch")
    if any(not _valid_number(v, 0, 9) for v in scores.values()):
        raise ValueError("invalid D score")
    return sorted(ids, key=lambda cid: (-float(scores[cid]), ids.index(cid)))


def _w0_order(spec, rec):
    """Use the pinned W0 adapter: score its returned head, then append the frozen tail."""
    order = BASE.BASE.score_order(spec, rec)
    ids = list(spec["candidate_ids"])
    if len(order) != len(ids) or len(set(order)) != len(ids) or set(order) != set(ids):
        raise ValueError("W0 inherited order does not preserve the complete candidate pool")
    return order


def _valid_number(value, lo, hi):
    return type(value) in (int, float) and math.isfinite(value) and lo <= value <= hi


def initial_state(spec, d_receipt, facets=None):
    if spec.get("kind") != "D" or spec.get("role") == "navigation" or spec.get("facets"):
        raise ValueError("initial state requires a substantive D variant")
    ids = list(spec["candidate_ids"])
    if len(ids) != len(set(ids)) or not 1 <= len(ids) <= 80:
        raise ValueError("D candidate membership invalid")
    order = _stable_score_order(spec, d_receipt)
    cards = spec["candidate_cards"] if "candidate_cards" in spec else None
    # EXP-043 specs carry cards in the submitted request body; recover their exact projection.
    request = strict_json(spec["body"])
    state0 = request.get("state")
    candidates = state0.get("candidates") if type(state0) is dict else None
    if type(candidates) is not list or [c.get("id") for c in candidates] != ids:
        raise ValueError("D state candidate order differs from frozen membership")
    if type(spec.get("task")) is not str or not spec["task"]:
        raise ValueError("D caller purpose missing")
    facets = facets or []
    if type(facets) is not list or len(facets) > 4:
        raise ValueError("facet contract invalid")
    scores = {cid: float(d_receipt["scores"][cid]) for cid in ids}
    selected = order[:min(TOP_K, len(order))]
    return {
        "candidates": candidates,
        "purpose": spec["task"],
        "facets": facets,
        "scores": scores,
        "order": order,
        "selected_ids": selected,
        "removable_ids": list(selected),
        "eligible_ids": [cid for cid in ids if cid not in selected and scores[cid] >= MIN_SCORE],
        "protected_ids": [],
        "evicted_ids": [],
        "step": 1,
    }

def state_for_round(state):
    ids = [c["id"] for c in state["candidates"]]
    if len(ids) > 80 or len(ids) != len(set(ids)) or set(ids) != set(state["scores"]) or set(ids) != set(state["order"]):
        raise ValueError("current pool membership corrupt")
    if len(state["selected_ids"]) != min(TOP_K, len(ids)) or not set(state["selected_ids"]) <= set(ids):
        raise ValueError("selected set membership corrupt")
    removable = [cid for cid in state["order"] if cid in set(state["selected_ids"]) and cid not in set(state["protected_ids"])]
    eligible = [cid for cid in ids if cid not in set(state["selected_ids"]) and cid not in set(state["evicted_ids"]) and cid not in set(state["protected_ids"]) and state["scores"][cid] >= MIN_SCORE]
    return {
        "candidates": state["candidates"],
        "purpose": state["purpose"],
        "facets": state["facets"],
        "scores": state["scores"],
        "order": state["order"],
        "selected_ids": list(state["selected_ids"]),
        "removable_ids": removable,
        "eligible_ids": eligible,
        "protected_ids": list(state["protected_ids"]),
        "evicted_ids": list(state["evicted_ids"]),
        "step": state["step"],
    }


def build_batch(state):
    state = state_for_round(state)
    if state["step"] > MAX_ROUNDS:
        return None
    removable = state["removable_ids"]
    eligible = state["eligible_ids"]
    if not removable or not eligible:
        return None
    questions = {}
    global_criteria = {"KEEP": GLOBAL_KEEP}
    for victim in removable:
        global_criteria[f"remove_{victim}"] = GLOBAL_REMOVE.format(victim_id=victim)
    questions["victim"] = {"type": "choice", "instructions": GLOBAL_INSTRUCTIONS, "criteria": global_criteria}
    for victim in removable:
        criteria = {"KEEP": REPLACEMENT_KEEP}
        for candidate in eligible:
            criteria[f"replace_{candidate}"] = REPLACEMENT_OPTION.replace("<victim_id>", victim).replace("<candidate_id>", candidate)
        questions[f"replacement_for_{victim}"] = {
            "type": "choice",
            "instructions": REPLACEMENT_INSTRUCTIONS.replace("<victim_id>", victim),
            "criteria": criteria,
        }
    if len(questions) > 11 or sum(len(q["criteria"]) for q in questions.values()) > 721:
        raise ValueError("registered Choice question/option envelope exceeded")
    return {"model": MODEL, "state": state, "questions": questions}


def checked_body(batch):
    """Serialize once and enforce registered byte limits before a transport attempt."""
    body = wire(batch)
    if len(wire(batch["state"])) > MAX_STATE:
        raise ValueError("state_bytes")
    if len(body) > MAX_BODY:
        raise ValueError("request_bytes")
    return body


def validate_choice_response(parsed, batch):
    if type(parsed) is not dict or set(parsed) != {"model", "answers", "usage", "_elapsed_ms"} or parsed.get("model") != MODEL:
        raise ValueError("Choice response envelope/model invalid")
    elapsed = parsed["_elapsed_ms"]
    if not _valid_number(elapsed, 0, MAX_REMOTE_MS):
        raise ValueError("Choice elapsed time invalid")
    usage = parsed["usage"]
    if type(usage) is not dict or set(usage) != {"input_tokens", "output_tokens"} or any(type(x) is not int or x < 0 for x in usage.values()):
        raise ValueError("Choice usage invalid or unknown")
    answers = parsed["answers"]
    questions = batch["questions"]
    if type(answers) is not dict or set(answers) != set(questions):
        raise ValueError("Choice answer IDs mismatch")
    chosen = {}
    distributions = {}
    for qid, q in questions.items():
        answer = answers[qid]
        if type(answer) is not dict or set(answer) != {"type", "choice", "probabilities", "confidence"} or answer.get("type") != "choice":
            raise ValueError(f"Choice answer shape invalid: {qid}")
        criteria = q["criteria"]
        probs = answer.get("probabilities")
        if type(probs) is not dict or set(probs) != set(criteria):
            raise ValueError(f"Choice probability option set mismatch: {qid}")
        if any(not _valid_number(v, 0, 1) for v in probs.values()) or abs(sum(float(v) for v in probs.values()) - 1.0) > 1e-6:
            raise ValueError(f"Choice probabilities invalid: {qid}")
        if not _valid_number(answer.get("confidence"), 0, 1):
            raise ValueError(f"Choice confidence invalid: {qid}")
        reported = answer.get("choice")
        if reported not in criteria or float(probs[reported]) != max(float(v) for v in probs.values()):
            raise ValueError(f"Reported choice is not a maximum: {qid}")
        # Resolve probability ties deterministically: KEEP first, then option insertion order.
        maximum = max(float(v) for v in probs.values())
        tied = [option for option in criteria if float(probs[option]) == maximum]
        chosen[qid] = "KEEP" if "KEEP" in tied else tied[0]
        distributions[qid] = {k: float(v) for k, v in probs.items()}
    return chosen, distributions


def apply_global_action(state, choices):
    current = state_for_round(state)
    victim_choice = choices.get("victim")
    if victim_choice == "KEEP":
        return current, {"choice": "KEEP"}, "global_keep"
    removable = current["removable_ids"]
    eligible = current["eligible_ids"]
    victim_options = [f"remove_{cid}" for cid in removable]
    if victim_choice not in victim_options:
        raise ValueError("global action names a non-removable victim")
    victim = victim_choice[len("remove_"):]
    replacement_key = f"replacement_for_{victim}"
    choice = choices.get(replacement_key)
    if choice == "KEEP":
        return current, {"choice": "KEEP", "victim": victim}, "branch_keep"
    expected = {f"replace_{cid}" for cid in eligible}
    if choice not in expected:
        raise ValueError("conditional action names an ineligible replacement")
    candidate = choice[len("replace_"):]
    selected = [cid for cid in current["selected_ids"] if cid != victim]
    selected.append(candidate)
    order = current["order"]
    selected.sort(key=order.index)
    new = dict(current)
    new["selected_ids"] = selected
    new["protected_ids"] = list(current["protected_ids"]) + [candidate]
    new["evicted_ids"] = list(current["evicted_ids"]) + [victim]
    new["step"] = current["step"] + 1
    if new["step"] > MAX_ROUNDS + 1:
        raise ValueError("round limit exceeded")
    return state_for_round(new), {"victim": victim, "replacement": candidate}, "accepted_replacement"


def relative_selection(spec, d_receipt, step_receipts, w0_order=None, facets=None):
    """Replay validated sequential Choice receipts; malformed/incomplete means W0 fallback."""
    state = initial_state(spec, d_receipt, facets)
    initial = list(state["order"])
    fallback_order = list(w0_order) if w0_order is not None else initial
    if len(fallback_order) != len(initial) or set(fallback_order) != set(initial):
        raise ValueError("W0 fallback membership mismatch")
    trace = {"steps": [], "fallback": None, "full_membership_preserved": True}
    try:
        for index, rec in enumerate(step_receipts):
            if index >= MAX_ROUNDS:
                raise ValueError("too_many_steps")
            batch = build_batch(state)
            if batch is None:
                raise ValueError("request_after_deterministic_stop")
            chosen, probs = validate_choice_response(rec["parsed"], batch)
            new, action, stop = apply_global_action(state, chosen)
            trace["steps"].append({"step": state["step"], "action": action, "stop": stop, "probabilities": probs})
            state = new
            if stop != "accepted_replacement":
                if index != len(step_receipts) - 1:
                    raise ValueError("steps_after_terminal_choice")
                break
        else:
            if step_receipts and len(step_receipts) == MAX_ROUNDS and state["step"] == MAX_ROUNDS + 1:
                trace["terminal"] = "four_round_limit"
        if len(step_receipts) > MAX_ROUNDS:
            raise ValueError("too_many_steps")
        order = [cid for cid in state["order"] if cid in set(state["selected_ids"])] + [cid for cid in state["order"] if cid not in set(state["selected_ids"])]
        if len(order) != len(initial) or set(order) != set(initial):
            raise ValueError("membership_changed")
        trace["selected_ids"] = list(state["selected_ids"])
        trace["evicted_ids"] = list(state["evicted_ids"])
        trace["protected_ids"] = list(state["protected_ids"])
        trace["fallback"] = False
        return order, trace, False
    except (KeyError, TypeError, ValueError, OverflowError) as exc:
        trace["fallback"] = type(exc).__name__
        return fallback_order, trace, True


def operation_inventory(verified):
    specs = verified["specmap"]
    operations = []
    for spec in verified["specs"]:
        if spec["kind"] != "D" or spec["role"] == "navigation":
            continue
        suffix = "" if spec.get("repeat_of") is None else "-" + spec["id"].split("-d", 1)[-1]
        # Pair W0 is keyed by the underlying pool, unchanged across D variants.
        w0id = f"{spec['pool_id']}-w0"
        f_spec = next((candidate for candidate in verified["specs"] if candidate["kind"] == "F" and candidate.get("pair_id") == spec.get("pair_id")), None)
        if f_spec is None or f_spec["candidate_ids"] != spec["candidate_ids"] or f_spec["task"] != spec["task"] or not f_spec.get("facets"):
            raise ValueError(f"missing/mismatched paired F facet contract for {spec['id']}")
        if w0id not in specs:
            raise ValueError(f"missing W0 for {spec['id']}")
        operations.append({
            "id": spec["id"], "pool_id": spec["pool_id"], "variant": "main" if not suffix else suffix[1:],
            "candidate_ids": list(spec["candidate_ids"]), "d_request_id": spec["id"], "f_request_id": f_spec["id"], "w0_request_id": w0id,
            "facets": f_spec["facets"],
            "max_rounds": MAX_ROUNDS, "step_request_ids": [f"{spec['id']}-choice-step{n}" for n in range(1, MAX_ROUNDS + 1)],
            "deterministic_state_builder": "EXP045/replay.py.txt:initial_state+build_batch",
        })
    if len(operations) != EXPECTED_REAL_OPERATIONS or len({x["id"] for x in operations}) != EXPECTED_REAL_OPERATIONS:
        raise ValueError("real operation inventory is not exactly 21")
    return operations


def navigation_inventory(verified):
    nav = [s for s in verified["specs"] if s.get("role") == "navigation" and s.get("kind") == "D"]
    if len(nav) != 5:
        raise ValueError("expected five frozen navigation controls")
    rows = []
    for d in nav:
        matches = [s for s in verified["specs"] if s.get("role") == "W0" and s.get("pool_id") == d.get("pool_id") and s.get("target_id") == d.get("target_id") and s["id"].startswith("nav-")]
        if len(matches) != 1:
            raise ValueError("navigation W0 companion is missing or ambiguous")
        rows.append({"d_request_id": d["id"], "w0_request_id": matches[0]["id"], "target_id": d["target_id"], "candidate_ids": list(d["candidate_ids"]), "facet_policy": "bypassed"})
    return rows


def make_neutral_fixture():
    # Neutral maximum-schema body: 10 selected cards and 70 eligible cards, four facets.
    candidates = [{"id": f"c{i:02d}", "title": f"Synthetic source {i:02d}", "url": f"https://example.invalid/{i:02d}", "snippet": "synthetic fixture"} for i in range(80)]
    state = {
        "candidates": candidates,
        "purpose": "Synthetic neutral Choice-envelope acceptance test; do not infer task truth.",
        "facets": [{"id": f"facet-{j}", "description": f"Synthetic facet {j}"} for j in range(4)],
        "scores": {c["id"]: (9.0 if i < 10 else 5.0) for i, c in enumerate(candidates)},
        "order": [c["id"] for c in candidates], "selected_ids": [c["id"] for c in candidates[:10]],
        "removable_ids": [c["id"] for c in candidates[:10]], "eligible_ids": [c["id"] for c in candidates[10:]],
        "protected_ids": [], "evicted_ids": [], "step": 1,
    }
    batch = build_batch(state)
    if batch is None or len(batch["questions"]) != 11 or sum(len(q["criteria"]) for q in batch["questions"].values()) != 721:
        raise ValueError("neutral maximum Choice inventory mismatch")
    body = checked_body(batch)
    state_bytes = len(wire(batch["state"]))
    if len(body) > MAX_BODY or state_bytes > MAX_STATE:
        raise ValueError("neutral body/state exceeds registered byte bounds")
    return {"body": body, "state": state, "state_bytes": state_bytes, "body_sha256": sha(body), "body_bytes": len(body), "question_count": 11, "option_count": 721, "candidate_ids": [c["id"] for c in candidates]}

def verify(root, require_publications=False, offline_revision=None, neutral_revision=None, require_neutral=True):
    root = root_path(root)

def _verify_publication_revision(root, filename, revision):
    if not isinstance(revision, str) or not re.fullmatch(r"[0-9a-f]{40}", revision):
        raise ValueError(f"invalid publication revision for {filename}")
    try:
        subprocess.run(["git", "merge-base", "--is-ancestor", revision, "origin/main"], cwd=REPO, check=True, capture_output=True, timeout=15)
        published = subprocess.run(["git", "show", f"{revision}:docs/experiments/evidence/EXP-045/{filename}"], cwd=REPO, check=True, capture_output=True, timeout=15).stdout
    except Exception as exc:
        raise ValueError(f"could not verify published EXP-045 gate artifact {filename}") from exc
    current = (root / filename).read_bytes()
    if published != current:
        raise ValueError(f"published gate artifact differs from local {filename}")
    return sha(published)


def publication_gates(root, required=False, offline_revision=None, neutral_revision=None, require_neutral=True):
    root = root_path(root)
    offline = root / "offline-qualification.json"
    neutral = root / "neutral-receipt.json"
    if not offline.is_file():
        if required:
            raise ValueError("offline qualification receipt is required")
        return {"offline": False, "neutral": False}
    qual = load_json(offline)
    qkeys = {"schema", "status", "files", "fixture_tests_passed", "manual_checks_passed", "synthetic_operations", "primary_deltas", "pool_inventory", "neutral_quality_failed", "provider_calls", "search_calls"}
    expected_files = {"replay.py.txt", "fixture_tests.py.txt", "manual-choice-checks.py.txt", "publication-gates-checks.py.txt", "transport-fixtures.py.txt", "transport-qualification.json", "prepared-plan.json", "neutral-body.json", "neutral-inventory.json", "source-pins.json", "registration-integrity.json"}
    if type(qual) is not dict or set(qual) != qkeys or qual.get("schema") != "exp045-offline-qualification/1" or qual.get("status") != "passed" or type(qual.get("files")) is not dict or set(qual["files"]) != expected_files:
        raise ValueError("offline qualification receipt schema/inventory mismatch")
    for name, digest in qual["files"].items():
        if not isinstance(digest, str) or not re.fullmatch(r"[0-9a-f]{64}", digest) or _hash(root / name) != digest:
            raise ValueError(f"offline qualification does not bind {name}")
    if type(qual["fixture_tests_passed"]) is not int or qual["fixture_tests_passed"] < 10 or type(qual["manual_checks_passed"]) is not int or qual["manual_checks_passed"] < 1 or type(qual.get("synthetic_operations")) is not int or qual["synthetic_operations"] != 21 or type(qual.get("primary_deltas")) is not dict or qual.get("primary_deltas") != {"A": 8, "B": 8} or type(qual.get("pool_inventory")) is not int or qual.get("pool_inventory") != 13 or qual.get("neutral_quality_failed") is not True or type(qual.get("provider_calls")) is not int or qual.get("provider_calls") != 0 or type(qual.get("search_calls")) is not int or qual.get("search_calls") != 0:
        raise ValueError("offline qualification evidence does not meet required checks")
    if required and offline_revision is None:
        raise ValueError("--offline-publication-revision is required for live actions")
    offline_pub_hash = _verify_publication_revision(root, "offline-qualification.json", offline_revision) if offline_revision else None
    if not neutral.is_file():
        if require_neutral:
            raise ValueError("neutral acceptance receipt is required")
        return {"offline": True, "offline_receipt_sha256": offline_pub_hash or _hash(offline), "neutral": False}
    rec = load_json(neutral)
    rkeys = {"schema", "status", "error", "model", "body_sha256", "raw_sha256", "raw_bytes", "response", "usage", "remote_http_ms", "transport_wall_ms", "transport_exit_code", "attempt_count", "retry_count", "attempts_sha256"}
    if type(rec) is not dict or set(rec) != rkeys or rec.get("schema") != "exp045-neutral-receipt/1" or rec.get("status") != "accepted" or rec.get("error") is not None or rec.get("model") != MODEL or rec.get("body_sha256") != _hash(root / "neutral-body.json") or type(rec.get("attempt_count")) is not int or rec.get("attempt_count") != 1 or type(rec.get("retry_count")) is not int or rec.get("retry_count") != 0 or type(rec.get("raw_bytes")) is not int:
        raise ValueError("neutral acceptance receipt identity mismatch")
    rawpath = root / "neutral.response.bin"
    parsed_path = root / "neutral-parsed.json"
    attempts_path = root / "neutral-attempts.jsonl"
    if not rawpath.is_file() or not parsed_path.is_file() or not attempts_path.is_file():
        raise ValueError("neutral raw/parsed/attempt evidence missing")
    raw = rawpath.read_bytes()
    if len(raw) != rec["raw_bytes"] or sha(raw) != rec["raw_sha256"] or len(raw) > MAX_RESPONSE:
        raise ValueError("neutral raw response integrity mismatch")
    parsed = parse_transport(raw)
    if load_json(parsed_path) != parsed:
        raise ValueError("neutral parsed response differs from raw response")
    neutral_batch = strict_json((root / "neutral-body.json").read_bytes())
    chosen, _ = validate_choice_response(parsed, neutral_batch)
    if rec["response"] != parsed or rec["usage"] != parsed["usage"] or rec["remote_http_ms"] != parsed["_elapsed_ms"]:
        raise ValueError("neutral receipt structured response is not raw-bound")
    if rec.get("transport_exit_code") != 0 or not _valid_number(rec.get("transport_wall_ms"), 0, MAX_WALL_SECONDS * 1000) or not isinstance(rec.get("attempts_sha256"), str) or not re.fullmatch(r"[0-9a-f]{64}", rec["attempts_sha256"]):
        raise ValueError("neutral transport/attempt evidence malformed")
    attempt_bytes = attempts_path.read_bytes()
    if sha(attempt_bytes) != rec["attempts_sha256"]:
        raise ValueError("neutral attempt ledger hash mismatch")
    events = [strict_json(line) for line in attempt_bytes.splitlines() if line]
    if len(events) != 2 or [e.get("event") for e in events] != ["attempt_started", "attempt_finished"]:
        raise ValueError("neutral acceptance must have exactly one start and finish")
    start, finish = events
    expected_id = "neutral-acceptance"
    if start.get("request_id") != expected_id or start.get("attempt") != 1 or start.get("request_sha256") != rec["body_sha256"] or start.get("remote_deadline_ms") != MAX_REMOTE_MS or finish.get("request_id") != expected_id or finish.get("attempt") != 1 or finish.get("request_sha256") != rec["body_sha256"] or finish.get("response_sha256") != rec["raw_sha256"] or finish.get("response_bytes") != len(raw) or finish.get("status") != "valid" or finish.get("error") is not None or finish.get("input_tokens") != parsed["usage"]["input_tokens"] or finish.get("output_tokens") != parsed["usage"]["output_tokens"] or finish.get("remote_http_ms") != parsed["_elapsed_ms"]:
        raise ValueError("neutral attempt ledger is not bound to the accepted raw result")
    if parsed["_elapsed_ms"] > MAX_REMOTE_MS or parsed["usage"]["input_tokens"] > MAX_CALL_INPUT:
        raise ValueError("neutral request exceeds observed provider limits")
    if required and neutral_revision is None:
        raise ValueError("--neutral-publication-revision is required for live actions")
    neutral_pub_hash = _verify_publication_revision(root, "neutral-receipt.json", neutral_revision) if neutral_revision else None
    return {"offline": True, "offline_receipt_sha256": offline_pub_hash, "neutral": True, "neutral_receipt_sha256": neutral_pub_hash, "neutral_selected_options": len(chosen)}


def parse_transport(raw):
    parsed = H.parse_transport_result(raw)
    if type(parsed) is not dict or parsed.get("model") != MODEL:
        raise ValueError("transport result model mismatch")
    return parsed


def request_envelope(body, deadline_ms):
    if type(deadline_ms) is not int or not 1 <= deadline_ms <= MAX_REMOTE_MS:
        raise ValueError("remote HTTP deadline must be 1..1000 ms")
    # The private wrapper decodes and posts these exact bytes; never parse/re-serialize
    # the production request between the frozen body hash and the HTTP transport.
    return wire({"body_base64": base64.b64encode(body).decode("ascii"), "http_deadline_ms": deadline_ms})


def _append(path, event):
    with path.open("ab") as f:
        f.write(wire(event))
        f.write(b"\n")
        f.flush()
        import os
        os.fsync(f.fileno())


def _run_one(root, argv, opid, state, operation_elapsed_ms, total_usage):
    batch = build_batch(state)
    if batch is None:
        return None, None
    body = checked_body(batch)
    state_bytes = len(wire(batch["state"]))
    remaining = MAX_OPERATION_REMOTE_MS - operation_elapsed_ms
    if remaining <= 0:
        raise AttemptStop("combined_operation_budget_exhausted")
    deadline_ms = min(MAX_REMOTE_MS, int(remaining))
    if deadline_ms <= 0:
        raise AttemptStop("combined_operation_budget_exhausted")
    qid = f"{opid}-choice-step{state['step']}"
    qdir = root / "requests"
    qdir.mkdir(parents=True, exist_ok=True)
    body_path = qdir / f"{qid}.json"
    immutable(body_path, body)
    body_sha = sha(body)
    envelope = request_envelope(body, deadline_ms)
    ledger = root / "attempts.jsonl"
    start_event = {"event": "attempt_started", "attempt": 1, "request_id": qid, "request_sha256": body_sha, "parent_response_sha256": state.get("parent_response_sha256"), "operation_id": opid, "step": state["step"], "remote_deadline_ms": deadline_ms, "at_utc": dt.datetime.now(dt.timezone.utc).isoformat()}
    _append(ledger, start_event)
    raw = b""
    exitcode, wall, overflow, timeout = -1, None, False, False
    error = None
    parsed = None
    try:
        raw, exitcode, wall, overflow, timeout = H.bounded_process(argv, envelope, MAX_WALL_SECONDS)
    except Exception:
        error = "transport_process_error"
    if error is None and timeout:
        error = "transport_timeout"
    elif error is None and overflow:
        error = "response_overflow"
    elif error is None and exitcode != 0:
        error = "transport_exit"
    if error is None:
        try:
            parsed = parse_transport(raw)
            choices, distributions = validate_choice_response(parsed, batch)
        except Exception:
            error = "invalid_choice_response"
            choices, distributions = {}, {}
    else:
        choices, distributions = {}, {}
    usage = parsed.get("usage") if type(parsed) is dict else None
    inp = usage.get("input_tokens") if type(usage) is dict and type(usage.get("input_tokens")) is int else None
    outtok = usage.get("output_tokens") if type(usage) is dict and type(usage.get("output_tokens")) is int else None
    remote_ms = parsed.get("_elapsed_ms") if type(parsed) is dict else None
    if error is None and (inp is None or outtok is None):
        error = "unknown_usage"
    usage_over = False
    if inp is not None and outtok is not None:
        usage_over = total_usage[0] + inp > MAX_TOTAL_INPUT or total_usage[1] + outtok > MAX_TOTAL_OUTPUT
        # Preserve known actual usage even when this response causes a stop.
        total_usage[0] += inp
        total_usage[1] += outtok
    if error is None and inp > MAX_CALL_INPUT:
        error = "per_call_input_limit_exceeded"
    if error is None and remote_ms > deadline_ms:
        error = "remote_deadline_exceeded"
    if error is None and usage_over:
        error = "cumulative_usage_exceeded"
    result = {"schema": "exp045-step-receipt/1", "id": qid, "request_sha256": body_sha, "parent_response_sha256": state.get("parent_response_sha256"), "response_sha256": sha(raw), "response_bytes": len(raw), "model": parsed.get("model") if type(parsed) is dict else None, "usage": usage, "remote_http_ms": remote_ms, "transport_wall_ms": wall, "transport_exit_code": exitcode, "status": "valid" if error is None else "invalid", "error": error, "answers": parsed.get("answers") if type(parsed) is dict else None, "choices": choices if error is None else None, "distributions": distributions if error is None else None}
    receipt_dir = root / "receipts"
    receipt_dir.mkdir(parents=True, exist_ok=True)
    immutable(receipt_dir / f"{qid}.response.bin", raw)
    immutable(receipt_dir / f"{qid}.json", pretty(result))
    _append(ledger, {"event": "attempt_finished", "attempt": 1, "request_id": qid, "request_sha256": body_sha, "parent_response_sha256": state.get("parent_response_sha256"), "response_sha256": sha(raw), "response_bytes": len(raw), "status": result["status"], "error": error, "input_tokens": inp, "output_tokens": outtok, "remote_http_ms": remote_ms, "at_utc": dt.datetime.now(dt.timezone.utc).isoformat()})
    if error:
        raise AttemptStop(error, remote_elapsed_ms=remote_ms, attempted=True)
    return {"parsed": parsed, "choices": choices, "distributions": distributions, "response_sha256": sha(raw), "receipt": result}, body_sha


def run_neutral(root, transport_file, offline_revision):
    """Perform the one synthetic batch-acceptance attempt; never retry."""
    root = root_path(root)
    # The offline qualification is required here, but this is the phase that creates
    # the neutral receipt; its absence is expected until after this single attempt.
    verify(root, False, offline_revision=offline_revision, require_neutral=False)
    gate = publication_gates(root, True, offline_revision=offline_revision, require_neutral=False)
    if not gate["offline"] or gate["neutral"]:
        raise ValueError("offline qualification must be published and neutral acceptance must not already exist")
    names = {"neutral-attempts.jsonl", "neutral.response.bin", "neutral-parsed.json", "neutral-receipt.json"}
    if any((root / name).exists() for name in names):
        raise ValueError("neutral attempt artifacts already exist; no retry/resume")
    body = (root / "neutral-body.json").read_bytes()
    batch = strict_json(body)
    if sha(body) != load_json(root / "neutral-inventory.json")["body_sha256"]:
        raise ValueError("frozen neutral body hash mismatch")
    argv = H.read_transport_argv(transport_file)
    ledger = root / "neutral-attempts.jsonl"
    request_sha = sha(body)
    _append(ledger, {"event": "attempt_started", "attempt": 1, "request_id": "neutral-acceptance", "request_sha256": request_sha, "parent_response_sha256": None, "remote_deadline_ms": MAX_REMOTE_MS, "at_utc": dt.datetime.now(dt.timezone.utc).isoformat()})
    raw = b""
    exitcode, wall, overflow, timeout = -1, None, False, False
    parsed = {}
    error = None
    try:
        raw, exitcode, wall, overflow, timeout = H.bounded_process(argv, request_envelope(body, MAX_REMOTE_MS), MAX_WALL_SECONDS)
    except Exception:
        error = "transport_process_error"
    if error is None and timeout: error = "transport_timeout"
    elif error is None and overflow: error = "response_overflow"
    elif error is None and exitcode != 0: error = "transport_exit"
    if error is None:
        try:
            parsed = parse_transport(raw)
            validate_choice_response(parsed, batch)
        except Exception:
            error = "invalid_neutral_response"
    usage = parsed.get("usage") if type(parsed) is dict else None
    inp = usage.get("input_tokens") if type(usage) is dict and type(usage.get("input_tokens")) is int else None
    outtok = usage.get("output_tokens") if type(usage) is dict and type(usage.get("output_tokens")) is int else None
    elapsed = parsed.get("_elapsed_ms") if type(parsed) is dict else None
    if error is None and (inp is None or outtok is None): error = "unknown_usage"
    if error is None and inp > MAX_CALL_INPUT: error = "per_call_input_limit_exceeded"
    if error is None and (inp > MAX_TOTAL_INPUT or outtok > MAX_TOTAL_OUTPUT): error = "neutral_usage_guard_exceeded"
    if error is None and elapsed > MAX_REMOTE_MS: error = "neutral_remote_deadline_exceeded"
    _append(ledger, {"event": "attempt_finished", "attempt": 1, "request_id": "neutral-acceptance", "request_sha256": request_sha, "parent_response_sha256": None, "response_sha256": sha(raw), "response_bytes": len(raw), "status": "valid" if error is None else "invalid", "error": error, "input_tokens": inp, "output_tokens": outtok, "remote_http_ms": elapsed, "at_utc": dt.datetime.now(dt.timezone.utc).isoformat()})
    immutable(root / "neutral.response.bin", raw)
    immutable(root / "neutral-parsed.json", pretty(parsed if type(parsed) is dict else {}))
    receipt = {"schema": "exp045-neutral-receipt/1", "status": "accepted" if error is None else "invalid", "error": error, "model": parsed.get("model") if type(parsed) is dict else None, "body_sha256": request_sha, "raw_sha256": sha(raw), "raw_bytes": len(raw), "response": parsed if type(parsed) is dict else {}, "usage": usage, "remote_http_ms": elapsed, "transport_wall_ms": wall, "transport_exit_code": exitcode, "attempt_count": 1, "retry_count": 0, "attempts_sha256": _hash(ledger)}
    immutable(root / "neutral-receipt.json", pretty(receipt))
    print(json.dumps({"neutral_status": receipt["status"], "error": error, "provider_attempts": 1, "retries": 0, "provider_calls": 1}, indent=2))
    return receipt


def run(root, transport_file, offline_revision=None, neutral_revision=None):
    root = root_path(root)
    gate = verify(root, True, offline_revision, neutral_revision, True)
    if not gate["publication_gates"]["offline"] or not gate["publication_gates"]["neutral"]:
        raise ValueError("offline and neutral publication gates are required")
    if (root / "attempts.jsonl").exists() or (root / "receipts").exists() or (root / "requests").exists():
        raise ValueError("EXP-045 run directory already contains attempts; no resume or retry")
    verified = validate_baseline()
    specs = verified["specmap"]
    # All raw EXP-043 W0/D receipts are already strictly validated by the pinned source chain.
    old_receipts = verified["baseline"]["receipts"]
    argv = H.read_transport_argv(transport_file)
    neutral_rec = load_json(root / "neutral-receipt.json")
    total_usage = [neutral_rec["usage"]["input_tokens"], neutral_rec["usage"]["output_tokens"]]
    if total_usage[0] > MAX_TOTAL_INPUT or total_usage[1] > MAX_TOTAL_OUTPUT:
        raise ValueError("neutral usage already exceeds cumulative guard")
    start = dt.datetime.now(dt.timezone.utc).isoformat()
    ledger = root / "attempts.jsonl"
    _append(ledger, {"event": "run_started", "prepared_plan_sha256": _hash(root / "prepared-plan.json"), "offline_qualification_sha256": gate["publication_gates"]["offline_receipt_sha256"], "neutral_receipt_sha256": gate["publication_gates"]["neutral_receipt_sha256"], "at_utc": start})
    completed = []
    case_attempts = 0
    terminal_reason = None
    ops = operation_inventory(verified)
    if len(ops) * MAX_ROUNDS + 1 > MAX_TOTAL_ATTEMPTS:
        raise ValueError("registered total-attempt bound inconsistent")
    for op_index, op in enumerate(ops):
        dspec = specs[op["d_request_id"]]
        drec = old_receipts[op["d_request_id"]]
        wspec = specs[op["w0_request_id"]]
        wrec = old_receipts[op["w0_request_id"]]
        fallback = _w0_order(wspec, wrec)
        if not op.get("facets"):
            completed.append({"operation_id": op["id"], "status": "bypass_no_facets", "order": fallback})
            continue
        state = initial_state(dspec, drec, op["facets"])
        elapsed = float(drec["remote_http_ms"])
        elapsed_complete = True
        step_receipts = []
        trace = {"steps": []}
        outcome = "four_round_cap"
        failure = None
        for _ in range(MAX_ROUNDS):
            if build_batch(state) is None:
                outcome = "deterministic_no_action"
                break
            if H.run_window_remaining(start) <= 0:
                failure = "four_hour_window_expired"
                break
            if total_usage[0] >= MAX_TOTAL_INPUT or total_usage[1] >= MAX_TOTAL_OUTPUT:
                failure = "cumulative_usage_guard"
                break
            try:
                # Size validation precedes the durable attempt event and transport.
                try:
                    checked_body(build_batch(state))
                except ValueError as exc:
                    if str(exc) in {"state_bytes", "request_bytes"}:
                        raise AttemptStop("request_or_state_bound_exceeded") from None
                    raise
                if case_attempts >= MAX_REAL_ATTEMPTS or case_attempts + 1 + 1 > MAX_TOTAL_ATTEMPTS:
                    raise AttemptStop("study_attempt_cap")
                case_attempts += 1
                response, _ = _run_one(root, argv, op["id"], state, elapsed, total_usage)
                step_receipts.append(response)
                elapsed += float(response["parsed"]["_elapsed_ms"])
                # Keep selected-result evolution strictly deterministic and bind next body to response bytes.
                new_state, action, stop = apply_global_action(state, response["choices"])
                trace["steps"].append({"step": state["step"], "action": action, "stop": stop, "response_sha256": response["response_sha256"], "request_sha256": response["receipt"]["request_sha256"]})
                new_state["parent_response_sha256"] = response["response_sha256"]
                state = new_state
                if stop != "accepted_replacement":
                    outcome = stop
                    break
            except Exception as exc:
                failure = exc.code if isinstance(exc, AttemptStop) else type(exc).__name__
                if isinstance(exc, AttemptStop) and exc.attempted:
                    if _valid_number(exc.remote_elapsed_ms, 0, MAX_REMOTE_MS):
                        elapsed += float(exc.remote_elapsed_ms)
                    else:
                        elapsed_complete = False
                break
        if failure:
            order = fallback
            status = "full_w0_fallback"
        else:
            order = [cid for cid in state["order"] if cid in set(state["selected_ids"])] + [cid for cid in state["order"] if cid not in set(state["selected_ids"])]
            status = "completed"
        if len(order) != len(dspec["candidate_ids"]) or set(order) != set(dspec["candidate_ids"]):
            order, status, failure = fallback, "full_w0_fallback", "membership_validation_failed"
        completed.append({"operation_id": op["id"], "status": status, "reason": failure or outcome, "order": order, "trace": trace, "remote_http_ms": elapsed if elapsed_complete else None, "remote_http_complete": elapsed_complete, "steps": len(step_receipts)})
        if failure:
            terminal_reason = failure
            _append(ledger, {"event": "runner_stopped", "request_id": op["id"], "reason": failure, "at_utc": dt.datetime.now(dt.timezone.utc).isoformat()})
            for rest in ops[op_index + 1:]:
                wspec2 = specs[rest["w0_request_id"]]
                wrec2 = old_receipts[rest["w0_request_id"]]
                completed.append({"operation_id": rest["id"], "status": "not_attempted_terminal_stop", "reason": failure, "order": _w0_order(wspec2, wrec2), "steps": 0})
            break
    actual_attempts = sum(1 for line in ledger.read_bytes().splitlines() if line and strict_json(line).get("event") == "attempt_started")
    if actual_attempts > MAX_REAL_ATTEMPTS or actual_attempts + 1 > MAX_TOTAL_ATTEMPTS:
        raise ValueError("durable attempt count exceeds registered maximum")
    finished_rows = [strict_json(line) for line in ledger.read_bytes().splitlines() if line and strict_json(line).get("event") == "attempt_finished"]
    usage_complete = all(type(row.get("input_tokens")) is int and type(row.get("output_tokens")) is int for row in finished_rows)
    # Unknown usage is never presented as a study total. known_usage is only a
    # lower bound over responses whose typed usage fields were actually observed.
    usage = {"input_tokens": total_usage[0], "output_tokens": total_usage[1]} if usage_complete else None
    immutable(root / "operation-results.json", pretty({"schema": "exp045-operation-results/1", "status": "complete" if terminal_reason is None else "stopped", "terminal_reason": terminal_reason, "operations": completed, "usage_complete": usage_complete, "usage": usage, "known_usage": {"input_tokens": total_usage[0], "output_tokens": total_usage[1]}, "provider_calls": actual_attempts, "neutral_attempts": 1, "total_attempts": actual_attempts + 1, "search_calls": 0}))


def analyze_rankings(ranks, refs, facets):
    # Exact frozen EXP-042 analysis path, including its pinned nDCG/bootstrap callbacks.
    return BASE.analyze_rankings(ranks, refs, facets)


def build_rankings_from_results(verified, results):
    """Adapt the complete 21-operation result inventory into the frozen analyzer shape."""
    operations = operation_inventory(verified)
    rows = results.get("operations") if type(results) is dict else None
    if type(rows) is not list or len(rows) != EXPECTED_REAL_OPERATIONS:
        raise ValueError("operation result list is not the complete 21-case inventory")
    result_by_id = {row.get("operation_id"): row for row in rows if type(row) is dict}
    expected_ids = {op["id"] for op in operations}
    if set(result_by_id) != expected_ids or len(result_by_id) != len(rows):
        raise ValueError("operation result IDs differ from frozen inventory")
    ranks = {}
    facets = {p: [row["id"] for row in verified["data"][2][p]] for p in (*SOFTWARE, *NATURAL)}
    for pool in ALL_POOLS:
        w0id, did = f"{pool}-w0", f"{pool}-d"
        w0 = _w0_order(verified["specmap"][w0id], verified["baseline"]["receipts"][w0id])
        d = _stable_score_order(verified["specmap"][did], verified["baseline"]["receipts"][did])
        order = result_by_id[did].get("order")
        members = set(verified["specmap"][did]["candidate_ids"])
        if type(order) is not list or len(order) != len(members) or len(set(order)) != len(order) or set(order) != members:
            raise ValueError(f"E order membership invalid for {pool}")
        ranks[pool] = {"w0": w0, "d_star": d, "e": order}
        if pool in EXTENDED:
            ranks[pool]["d"] = d
            for suffix, arm in (("repeat", "repeat"), ("rotated20", "rotated")):
                opid = f"{pool}-d-{suffix}"
                repeat_order = result_by_id[opid].get("order")
                if type(repeat_order) is not list or len(repeat_order) != len(members) or len(set(repeat_order)) != len(repeat_order) or set(repeat_order) != members:
                    raise ValueError(f"{arm} order membership invalid for {pool}")
                ranks[pool][arm] = repeat_order
    navigation = []
    for q, cid in BASE.BASE.NAV_IDS:
        for kind, key in (("W0", f"nav-{q}-{cid}-w0"), ("D", f"nav-{q}-{cid}-d")):
            spec = verified["specmap"][key]
            order = _stable_score_order(spec, verified["baseline"]["receipts"][key])
            if cid not in order:
                raise ValueError("navigation target is absent from its order")
            if kind == "W0":
                w0_rank = order.index(cid) + 1
            else:
                d_rank = order.index(cid) + 1
        navigation.append({"query_id": q, "target_id": cid, "w0_rank": w0_rank, "d_rank": d_rank, "passed": w0_rank <= 3 and d_rank <= 3, "facet_policy_bypassed": True})
    if len(ranks) != 13 or len(navigation) != 5:
        raise ValueError("analysis inputs lost a pool or navigation control")
    return ranks, facets, navigation


def _validate_case_journal(root, verified, results):
    """Bind every persisted action to its exact request, raw response, and parent."""
    ledger_path = root / "attempts.jsonl"
    if not ledger_path.is_file():
        raise ValueError("case attempt ledger missing")
    events = [strict_json(line) for line in ledger_path.read_bytes().splitlines() if line]
    if not events or events[0].get("event") != "run_started":
        raise ValueError("case ledger lacks one run-start event")
    starts, finishes, by_id = [], [], {}
    usage_complete = True
    active = None
    saw_start = False
    stopped = False
    previous_time = None
    run_keys = {"event", "prepared_plan_sha256", "offline_qualification_sha256", "neutral_receipt_sha256", "at_utc"}
    start_keys = {"event", "attempt", "request_id", "request_sha256", "parent_response_sha256", "operation_id", "step", "remote_deadline_ms", "at_utc"}
    finish_keys = {"event", "attempt", "request_id", "request_sha256", "parent_response_sha256", "response_sha256", "response_bytes", "status", "error", "input_tokens", "output_tokens", "remote_http_ms", "at_utc"}
    stop_keys = {"event", "request_id", "reason", "at_utc"}
    for index, event in enumerate(events):
        if type(event) is not dict or type(event.get("event")) is not str:
            raise ValueError("case ledger event malformed")
        when = event.get("at_utc")
        try:
            parsed_when = dt.datetime.fromisoformat(when.replace("Z", "+00:00"))
        except Exception as exc:
            raise ValueError("case ledger timestamp malformed") from exc
        if previous_time is not None and parsed_when < previous_time:
            raise ValueError("case ledger timestamps out of order")
        previous_time = parsed_when
        kind = event["event"]
        if kind == "run_started":
            if index != 0 or saw_start or active is not None or set(event) != run_keys:
                raise ValueError("case ledger run start malformed/duplicated")
            saw_start = True
        elif kind == "attempt_started":
            if not saw_start or stopped or active is not None or set(event) != start_keys or type(event.get("attempt")) is not int or event["attempt"] != 1:
                raise ValueError("case attempt start chronology/shape invalid")
            rid = event.get("request_id")
            if type(rid) is not str or rid in by_id or not re.fullmatch(r".+-choice-step[1-4]", rid) or not re.fullmatch(r"[0-9a-f]{64}", str(event.get("request_sha256"))):
                raise ValueError("case attempt identity invalid")
            parent = event.get("parent_response_sha256")
            if parent is not None and (type(parent) is not str or not re.fullmatch(r"[0-9a-f]{64}", parent)):
                raise ValueError("parent response hash malformed")
            if type(event.get("remote_deadline_ms")) is not int or not 1 <= event["remote_deadline_ms"] <= MAX_REMOTE_MS or type(event.get("step")) is not int or event["step"] not in range(1, MAX_ROUNDS + 1):
                raise ValueError("case attempt deadline/step invalid")
            active = event
            starts.append(event)
        elif kind == "attempt_finished":
            if active is None or set(event) != finish_keys or event.get("request_id") != active.get("request_id") or event.get("attempt") != 1 or event.get("request_sha256") != active.get("request_sha256") or event.get("parent_response_sha256") != active.get("parent_response_sha256"):
                raise ValueError("case completion does not match an open attempt")
            if event.get("status") not in {"valid", "invalid"} or event.get("error") is not None and type(event.get("error")) is not str:
                raise ValueError("case completion status/error malformed")
            if event.get("response_bytes") is not None and (type(event["response_bytes"]) is not int or event["response_bytes"] < 0 or event["response_bytes"] > MAX_RESPONSE):
                raise ValueError("case response byte count invalid")
            for key in ("input_tokens", "output_tokens"):
                if event.get(key) is not None and (type(event[key]) is not int or event[key] < 0):
                    raise ValueError("case completion usage invalid")
            if event.get("input_tokens") is None or event.get("output_tokens") is None:
                usage_complete = False
            if event.get("remote_http_ms") is not None and not _valid_number(event["remote_http_ms"], 0, MAX_REMOTE_MS):
                raise ValueError("case completion HTTP elapsed invalid")
            finishes.append(event)
            by_id[event["request_id"]] = (active, event)
            active = None
        elif kind == "runner_stopped":
            if active is not None or stopped or set(event) != stop_keys or index != len(events) - 1 or type(event.get("reason")) is not str:
                raise ValueError("runner stop chronology/shape invalid")
            stopped = True
        else:
            raise ValueError("unknown case ledger event")
    if active is not None or not saw_start or len(starts) > MAX_REAL_ATTEMPTS or len(starts) != len(finishes):
        raise ValueError("case attempt/finish count invalid")
    if results.get("provider_calls") != len(starts) or results.get("total_attempts") != len(starts) + 1 or results.get("neutral_attempts") != 1 or results.get("search_calls") != 0:
        raise ValueError("operation result attempt counts are not ledger-bound")
    receipts = root / "receipts"
    requests = root / "requests"
    known = set()
    states_after = {}
    known_usage = [0, 0]
    neutral = load_json(root / "neutral-receipt.json")
    if type(neutral.get("usage")) is not dict or any(type(neutral["usage"].get(k)) is not int or neutral["usage"][k] < 0 for k in ("input_tokens", "output_tokens")):
        raise ValueError("published neutral usage is unavailable")
    known_usage = [neutral["usage"]["input_tokens"], neutral["usage"]["output_tokens"]]
    operations = operation_inventory(verified)
    operation_index = {op["id"]: i for i, op in enumerate(operations)}
    last_op_index = -1
    per_op_elapsed = {}
    per_op_elapsed_complete = {}
    attempted_per_op = {}
    prior_attempt_terminal = False
    for start, finish in zip(starts, finishes):
        rid = start["request_id"]
        matches = [op for op in operations if any(rid == f"{op['id']}-choice-step{n}" for n in range(1, MAX_ROUNDS + 1))]
        if len(matches) != 1:
            raise ValueError("attempt ID is not in the registered case inventory")
        op = matches[0]
        idx = operation_index[op["id"]]
        if idx < last_op_index or prior_attempt_terminal:
            raise ValueError("case operations were dispatched out of registered order")
        last_op_index = idx
        attempted_per_op.setdefault(op["id"], []).append(rid)
        if by_id.get(rid) and by_id[rid][1].get("status") == "invalid":
            prior_attempt_terminal = True
    for op in operations:
        opid = op["id"]
        result = next((item for item in results["operations"] if item.get("operation_id") == opid), None)
        if result is None:
            raise ValueError("operation result missing registered ID")
        w0 = verified["baseline"]["receipts"][op["w0_request_id"]]
        d = verified["baseline"]["receipts"][op["d_request_id"]]
        fallback = _w0_order(verified["specmap"][op["w0_request_id"]], w0)
        if result.get("status") in {"bypass_no_facets", "not_attempted_terminal_stop"}:
            if result.get("order") != fallback and result.get("status") == "not_attempted_terminal_stop":
                raise ValueError("terminal-unattempted operation is not W0 fallback")
            continue
        state = initial_state(verified["specmap"][op["d_request_id"]], d, op["facets"])
        op_elapsed = d.get("remote_http_ms")
        if not _valid_number(op_elapsed, 0, MAX_OPERATION_REMOTE_MS):
            raise ValueError("historical D elapsed time invalid")
        op_elapsed = float(op_elapsed)
        operation_steps = result.get("trace", {}).get("steps", [])
        for step_trace in operation_steps:
            rid = f"{opid}-choice-step{state['step']}"
            if rid not in by_id:
                raise ValueError("operation trace has no durable attempt")
            known.add(rid)
            start, finish = by_id[rid]
            if start.get("operation_id") != opid or start.get("step") != state["step"]:
                raise ValueError("attempt operation/step identity mismatch")
            reqpath = requests / f"{rid}.json"
            recpath = receipts / f"{rid}.json"
            rawpath = receipts / f"{rid}.response.bin"
            if not reqpath.is_file() or not recpath.is_file() or not rawpath.is_file():
                raise ValueError("attempt request or receipt artifact missing")
            body = reqpath.read_bytes()
            batch = build_batch(state)
            if batch is None or body != checked_body(batch) or sha(body) != start.get("request_sha256") or sha(body) != step_trace.get("request_sha256"):
                raise ValueError("persisted request differs from deterministic next-step body")
            parent = state.get("parent_response_sha256")
            if start.get("parent_response_sha256") != parent:
                raise ValueError("parent response hash chain mismatch")
            expected_deadline = min(MAX_REMOTE_MS, int(MAX_OPERATION_REMOTE_MS - op_elapsed))
            if expected_deadline <= 0 or start.get("remote_deadline_ms") != expected_deadline:
                raise ValueError("attempt remote deadline does not reflect historical D plus prior Choice calls")
            raw = rawpath.read_bytes()
            rec = load_json(recpath)
            parsed = parse_transport(raw)
            if sha(raw) != finish.get("response_sha256") or len(raw) != finish.get("response_bytes") or rec.get("response_sha256") != sha(raw) or rec.get("response_bytes") != len(raw) or parsed.get("model") != MODEL:
                raise ValueError("raw response bytes are not history/receipt bound")
            if rec.get("status") != finish.get("status") or rec.get("error") != finish.get("error") or rec.get("request_sha256") != sha(body) or rec.get("parent_response_sha256") != parent:
                raise ValueError("response receipt/history fields disagree")
            choices, distributions = validate_choice_response(parsed, batch)
            if rec.get("schema") != "exp045-step-receipt/1" or rec.get("id") != rid or rec.get("model") != MODEL or rec.get("answers") != parsed["answers"] or rec.get("usage") != parsed["usage"] or rec.get("remote_http_ms") != parsed["_elapsed_ms"] or rec.get("choices") != choices or rec.get("distributions") != distributions or finish.get("input_tokens") != parsed["usage"]["input_tokens"] or finish.get("output_tokens") != parsed["usage"]["output_tokens"] or finish.get("remote_http_ms") != parsed["_elapsed_ms"]:
                raise ValueError("structured receipt/ledger projection differs from raw response")
            if parsed["_elapsed_ms"] > expected_deadline or parsed["usage"]["input_tokens"] > MAX_CALL_INPUT:
                raise ValueError("Choice response exceeded remote deadline or input limit")
            known_usage[0] += parsed["usage"]["input_tokens"]
            known_usage[1] += parsed["usage"]["output_tokens"]
            op_elapsed += float(parsed["_elapsed_ms"])
            new_state, action, stop = apply_global_action(state, choices)
            if action != step_trace.get("action") or stop != step_trace.get("stop") or step_trace.get("response_sha256") != sha(raw):
                raise ValueError("replayed action differs from frozen operation trace")
            if stop != "accepted_replacement" and step_trace is not operation_steps[-1]:
                raise ValueError("trace continues after a deterministic KEEP")
            new_state["parent_response_sha256"] = sha(raw)
            state = new_state
        states_after[opid] = state
        per_op_elapsed[opid] = op_elapsed
        per_op_elapsed_complete[opid] = True
        reconstructed = [cid for cid in state["order"] if cid in set(state["selected_ids"])] + [cid for cid in state["order"] if cid not in set(state["selected_ids"])]
        if result.get("status") == "completed" and result.get("order") != reconstructed:
            raise ValueError("completed operation ranking does not replay from raw answers")
        if result.get("status") == "full_w0_fallback" and result.get("order") != fallback:
            raise ValueError("failed operation did not return full W0 fallback")
    if set(by_id) != known:
        # One invalid final attempt may not appear in an accepted step trace. It is still bound to W0 fallback.
        orphaned = set(by_id) - known
        for rid in orphaned:
            start, finish = by_id[rid]
            if finish.get("status") != "invalid":
                raise ValueError("valid durable attempt is absent from the operation trace")
            recpath, rawpath = receipts / f"{rid}.json", receipts / f"{rid}.response.bin"
            if not recpath.is_file() or not rawpath.is_file():
                raise ValueError("invalid attempt raw/receipt missing")
            raw, rec = rawpath.read_bytes(), load_json(recpath)
            if len(raw) != finish.get("response_bytes") or sha(raw) != finish.get("response_sha256") or rec.get("response_bytes") != len(raw) or rec.get("response_sha256") != sha(raw) or rec.get("status") != "invalid" or rec.get("request_sha256") != start.get("request_sha256"):
                raise ValueError("invalid attempt raw/receipt/history binding failed")
            opid = start.get("operation_id")
            if opid not in states_after:
                raise ValueError("failed orphan attempt has no registered operation state")
            state = states_after[opid]
            expected_id = f"{opid}-choice-step{state['step']}"
            if rid != expected_id or start.get("step") != state["step"]:
                raise ValueError("failed attempt is not the exact next deterministic step")
            batch = build_batch(state)
            if batch is None:
                raise ValueError("failed attempt occurred after a deterministic terminal stop")
            reqpath = requests / f"{rid}.json"
            if not reqpath.is_file():
                raise ValueError("failed attempt request body missing")
            body = reqpath.read_bytes()
            if body != checked_body(batch) or sha(body) != start["request_sha256"] or start.get("parent_response_sha256") != state.get("parent_response_sha256"):
                raise ValueError("failed attempt request/state/parent binding mismatch")
            d_elapsed = float(verified["baseline"]["receipts"][next(op for op in operation_inventory(verified) if op["id"] == opid)["d_request_id"]]["remote_http_ms"])
            prior_elapsed = sum(float(by_id[f"{opid}-choice-step{n}"][1]["remote_http_ms"]) for n in range(1, state["step"]) if f"{opid}-choice-step{n}" in by_id and _valid_number(by_id[f"{opid}-choice-step{n}"][1].get("remote_http_ms"), 0, MAX_REMOTE_MS))
            expected_deadline = min(MAX_REMOTE_MS, int(MAX_OPERATION_REMOTE_MS - d_elapsed - prior_elapsed))
            if expected_deadline <= 0 or start.get("remote_deadline_ms") != expected_deadline:
                raise ValueError("failed attempt deadline does not match cumulative operation budget")
            try:
                parsed_failed = parse_transport(raw)
            except Exception:
                parsed_failed = None
            if parsed_failed is not None:
                raw_usage = parsed_failed.get("usage")
                raw_in = raw_usage.get("input_tokens") if type(raw_usage) is dict else None
                raw_out = raw_usage.get("output_tokens") if type(raw_usage) is dict else None
                if parsed_failed.get("_elapsed_ms") != finish.get("remote_http_ms") or raw_in != finish.get("input_tokens") or raw_out != finish.get("output_tokens") or rec.get("usage") != raw_usage or rec.get("remote_http_ms") != parsed_failed.get("_elapsed_ms") or rec.get("error") != finish.get("error"):
                    raise ValueError("failed attempt timing/usage does not match raw response")
                if _valid_number(parsed_failed.get("_elapsed_ms"), 0, MAX_REMOTE_MS):
                    per_op_elapsed[opid] = d_elapsed + prior_elapsed + float(parsed_failed["_elapsed_ms"])
                else:
                    per_op_elapsed_complete[opid] = False
                usage = parsed_failed.get("usage")
                if type(usage) is dict and type(usage.get("input_tokens")) is int and type(usage.get("output_tokens")) is int:
                    known_usage[0] += usage["input_tokens"]
                    known_usage[1] += usage["output_tokens"]
                else:
                    usage_complete = False
            else:
                if rec.get("usage") is not None or rec.get("remote_http_ms") is not None or rec.get("error") != finish.get("error"):
                    raise ValueError("unparseable failed response projection differs from raw-bound receipt")
                usage_complete = False
                per_op_elapsed_complete[opid] = False
            failed_op = next(op for op in operations if op["id"] == opid)
            failed_w0 = _w0_order(verified["specmap"][failed_op["w0_request_id"]], verified["baseline"]["receipts"][failed_op["w0_request_id"]])
            if result_by_id_for_failure := next((x for x in results["operations"] if x.get("operation_id") == opid), None):
                if result_by_id_for_failure.get("status") != "full_w0_fallback" or result_by_id_for_failure.get("order") != failed_w0:
                    raise ValueError("failed attempt is not represented by a full-W0 operation fallback")
    # Account only observed usage; missing values make the study total unknown.
    expected_known_usage = {"input_tokens": known_usage[0], "output_tokens": known_usage[1]}
    if results.get("usage_complete") is not usage_complete or results.get("known_usage") != expected_known_usage:
        raise ValueError("reported usage completeness/known lower bound differs from raw attempts")
    expected_usage = expected_known_usage if usage_complete else None
    if results.get("usage") != expected_usage:
        raise ValueError("reported study usage is incorrectly represented as complete/incomplete")
    for opid, expected_elapsed in per_op_elapsed.items():
        result = next(row for row in results["operations"] if row.get("operation_id") == opid)
        timing_complete = per_op_elapsed_complete.get(opid, False)
        if result.get("remote_http_complete") is not timing_complete:
            raise ValueError("operation elapsed completeness differs from raw attempt timing")
        observed_elapsed = result.get("remote_http_ms")
        if timing_complete:
            if not _valid_number(observed_elapsed, 0, MAX_OPERATION_REMOTE_MS + MAX_ROUNDS * MAX_REMOTE_MS) or not math.isclose(float(observed_elapsed), float(expected_elapsed), rel_tol=0.0, abs_tol=1e-6):
                raise ValueError("operation elapsed total differs from historical D plus raw response timings")
        elif observed_elapsed is not None:
            raise ValueError("unknown operation elapsed time must not be reported as complete")
    expected_request_files = {f"{rid}.json" for rid in by_id}
    expected_receipt_files = expected_request_files | {f"{rid}.response.bin" for rid in by_id}
    if set(p.name for p in requests.iterdir()) != expected_request_files or set(p.name for p in receipts.iterdir()) != expected_receipt_files:
        raise ValueError("orphan or missing request/receipt artifact")
    if bool(stopped) != (results.get("status") == "stopped") or results.get("terminal_reason") != (events[-1].get("reason") if stopped else None):
        raise ValueError("runner stop event does not match operation result status")
    return {"attempts": len(starts), "completed_attempts": len(finishes), "ledger_sha256": sha(ledger_path.read_bytes()), "usage_complete": usage_complete, "known_usage": expected_known_usage, "operation_elapsed_complete": per_op_elapsed_complete, "operation_elapsed_ms": per_op_elapsed}


def compute_report(root):
    root = root_path(root)
    verified = validate_baseline()
    results = load_json(root / "operation-results.json")
    if type(results) is not dict or results.get("schema") != "exp045-operation-results/1" or len(results.get("operations", [])) != EXPECTED_REAL_OPERATIONS:
        raise ValueError("incomplete EXP-045 operation result inventory")
    journal = _validate_case_journal(root, verified, results)
    result_by_id = {row["operation_id"]: row for row in results["operations"]}
    if len(result_by_id) != EXPECTED_REAL_OPERATIONS:
        raise ValueError("duplicate/missing EXP-045 operation results")
    ranks, facets, navrows = build_rankings_from_results(verified, results)
    analysis = analyze_rankings(ranks, verified["refs"], facets)
    status_complete = all(row["status"] == "completed" for row in results["operations"])
    nav_pass = len(navrows) == 5 and all(row["passed"] for row in navrows)
    resource_codes = {"combined_operation_budget_exhausted", "four_hour_window_expired", "cumulative_usage_guard", "cumulative_usage_exceeded", "per_call_input_limit_exceeded", "remote_deadline_exceeded", "transport_timeout", "response_overflow", "request_or_state_bound_exceeded", "study_attempt_cap", "neutral_usage_guard_exceeded", "neutral_remote_deadline_exceeded"}
    reasons = [row.get("reason") for row in results["operations"] if row.get("status") != "completed"]
    resource_failure = any(reason in resource_codes for reason in reasons)
    protocol_failure = any(row.get("status") == "full_w0_fallback" and row.get("reason") not in resource_codes for row in results["operations"]) or not journal["usage_complete"]
    supported = status_complete and analysis["development_screen_passed"] and nav_pass and journal["attempts"] <= MAX_REAL_ATTEMPTS and results.get("total_attempts") == journal["attempts"] + 1
    if supported:
        decision = "supports_only_untouched_confirmation"
    elif resource_failure:
        decision = "not_supported_for_resource_envelope"
    elif protocol_failure or not status_complete:
        decision = "inconclusive_protocol_or_incomplete"
    else:
        decision = "development_candidate_rejected"

def analyze(root, offline_revision=None, neutral_revision=None):
    root = root_path(root)
    publication_gates(root, True, offline_revision, neutral_revision, True)
    report = compute_report(root)
    immutable(root / "analysis.json", pretty(report))
    print(json.dumps({"decision": report["decision"], "quality_passed": report["development_analysis"]["development_screen_passed"], "navigation_passed": report["navigation"]["passed"], "provider_calls": 0}, indent=2))
    return report


def selftest(root):
    import os
    env = dict(os.environ)
    env["PYTHONPATH"] = str(REPO) + (os.pathsep + env["PYTHONPATH"] if env.get("PYTHONPATH") else "")
    proc = subprocess.run([sys.executable, str(root_path(root) / "fixture_tests.py.txt")], cwd=str(REPO), env=env, capture_output=True, text=True, timeout=240)
    if proc.returncode:
        raise RuntimeError(proc.stdout + proc.stderr)
    print(proc.stdout.rstrip())
    print("EXP-045 offline fixtures passed; provider/search calls=0")


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--root", type=pathlib.Path, default=HERE)
    g = p.add_mutually_exclusive_group(required=True)
    g.add_argument("--prepare", action="store_true")
    g.add_argument("--verify", action="store_true")
    g.add_argument("--run", metavar="PRIVATE_TRANSPORT_ARGV_FILE")
    g.add_argument("--neutral", metavar="PRIVATE_TRANSPORT_ARGV_FILE")
    g.add_argument("--analyze", action="store_true")
    g.add_argument("--selftest", action="store_true")
    p.add_argument("--offline-publication-revision")
    p.add_argument("--neutral-publication-revision")
    a = p.parse_args()
    if a.prepare:
        prepare(a.root)
    elif a.verify:
        print(json.dumps(verify(a.root, bool(a.offline_publication_revision or a.neutral_publication_revision), a.offline_publication_revision, a.neutral_publication_revision, bool(a.neutral_publication_revision)), indent=2))
    elif a.neutral:
        run_neutral(a.root, a.neutral, a.offline_publication_revision)
    elif a.run:
        run(a.root, a.run, a.offline_publication_revision, a.neutral_publication_revision)
    elif a.analyze:
        analyze(a.root, a.offline_publication_revision, a.neutral_publication_revision)
    else:
        selftest(a.root)


if __name__ == "__main__":
    main()
