def synthetic_spec(n=20):
    ids = [f"c{i:02d}" for i in range(n)]
    candidates = [{"id": cid, "title": f"Synthetic {cid}", "url": f"https://example.invalid/{cid}", "snippet": "neutral fixture"} for cid in ids]
    body = R.wire({"model": R.MODEL, "state": {"candidates": candidates}, "questions": {}})
    return {"kind": "D", "role": "D", "candidate_ids": ids, "body": body, "task": "Synthetic source-selection purpose for fixture validation.", "facets": []}, {cid: float(max(0, 9 - i // 3)) for i, cid in enumerate(ids)}


def response_for(batch, picks=None, probs_factory=None):
    picks = picks or {}
    answers = {}
    for qid, q in batch["questions"].items():
        options = list(q["criteria"])
        if probs_factory:
            probs = probs_factory(qid, options)
        else:
            chosen = picks.get(qid, "KEEP")
            probs = {o: (1.0 if o == chosen else 0.0) for o in options}
        chosen_id = picks.get(qid, max(probs, key=probs.get))
        answers[qid] = {"type": "choice", "choice": chosen_id, "probabilities": probs, "confidence": 1.0}
    return {"model": R.MODEL, "answers": answers, "usage": {"input_tokens": 100, "output_tokens": 40}, "_elapsed_ms": 10.0}


def current_state(n=20, facet_count=2):
    spec, scores = synthetic_spec(n)
    facets = [{"id": f"f{i}", "description": f"Fixture facet {i}"} for i in range(facet_count)]
    return R.initial_state(spec, {"scores": scores}, facets), spec, scores, facets


class TestChoiceBatch(unittest.TestCase):
    def test_neutral_has_11_questions_and_721_options(self):
        neutral = R.make_neutral_fixture()
        body = R.strict_json(neutral["body"])
        self.assertEqual(len(body["questions"]), 11)
        self.assertEqual(sum(len(q["criteria"]) for q in body["questions"].values()), 721)
        self.assertEqual(neutral["question_count"], 11)
        self.assertEqual(neutral["option_count"], 721)
        self.assertLessEqual(neutral["body_bytes"], R.MAX_BODY)
        self.assertLessEqual(neutral["state_bytes"], R.MAX_STATE)

    def test_exact_instruction_templates_and_option_order(self):
        state, _, _, _ = current_state()
        batch = R.build_batch(state)
        self.assertEqual(batch["questions"]["victim"]["instructions"], R.GLOBAL_INSTRUCTIONS)
        self.assertEqual(list(batch["questions"]["victim"]["criteria"])[0], "KEEP")
        victim = state["removable_ids"][0]
        branch = batch["questions"][f"replacement_for_{victim}"]
        self.assertEqual(list(branch["criteria"])[0], "KEEP")
        self.assertIn(victim, branch["instructions"])
        self.assertEqual(batch["state"]["facets"], state["facets"])

    def test_complete_response_parses_and_resolves_keep_ties(self):
        state, *_ = current_state()
        batch = R.build_batch(state)
        parsed = response_for(batch, probs_factory=lambda qid, options: {o: 0.5 if o in ("KEEP", options[-1]) else 0.0 for o in options})
        chosen, _ = R.validate_choice_response(parsed, batch)
        self.assertTrue(all(value == "KEEP" for value in chosen.values()))

    def test_missing_extra_and_nonfinite_distribution_rejected(self):
        state, *_ = current_state()
        batch = R.build_batch(state)
        valid = response_for(batch)
        bad = dict(valid); bad["answers"] = dict(valid["answers"]); bad["answers"].pop(next(iter(batch["questions"])))
        with self.assertRaises(ValueError): R.validate_choice_response(bad, batch)
        bad = response_for(batch); bad["answers"]["victim"]["probabilities"]["extra"] = 0
        with self.assertRaises(ValueError): R.validate_choice_response(bad, batch)
        bad = response_for(batch); bad["answers"]["victim"]["probabilities"]["KEEP"] = math.nan
        with self.assertRaises(ValueError): R.validate_choice_response(bad, batch)

    def test_probability_sum_confidence_and_reported_choice_validated(self):
        state, *_ = current_state()
        batch = R.build_batch(state)
        bad = response_for(batch); bad["answers"]["victim"]["probabilities"]["KEEP"] = 0.5
        with self.assertRaises(ValueError): R.validate_choice_response(bad, batch)
        bad = response_for(batch); bad["answers"]["victim"]["confidence"] = True
        with self.assertRaises(ValueError): R.validate_choice_response(bad, batch)
        bad = response_for(batch); bad["answers"]["victim"]["choice"] = list(batch["questions"]["victim"]["criteria"])[-1]
        with self.assertRaises(ValueError): R.validate_choice_response(bad, batch)

    def test_global_keep_ignores_all_valid_conditional_answers(self):
        state, *_ = current_state()
        batch = R.build_batch(state)
        choices = {qid: (list(q["criteria"])[-1] if qid != "victim" else "KEEP") for qid, q in batch["questions"].items()}
        new, action, stop = R.apply_global_action(state, choices)
        self.assertEqual(stop, "global_keep")
        self.assertEqual(new["selected_ids"], state["selected_ids"])
        self.assertEqual(action["choice"], "KEEP")

    def test_only_selected_victims_conditional_answer_is_consumed(self):
        state, *_ = current_state()
        batch = R.build_batch(state)
        victim = state["removable_ids"][0]
        choices = {qid: list(q["criteria"])[-1] for qid, q in batch["questions"].items()}
        choices["victim"] = f"remove_{victim}"
        target = state["eligible_ids"][0]
        choices[f"replacement_for_{victim}"] = f"replace_{target}"
        new, action, stop = R.apply_global_action(state, choices)
        self.assertEqual(stop, "accepted_replacement")
        self.assertEqual(action, {"victim": victim, "replacement": target})
        self.assertIn(target, new["protected_ids"])
        self.assertIn(victim, new["evicted_ids"])
        self.assertNotIn(victim, new["eligible_ids"])
        self.assertNotIn(target, new["removable_ids"])

    def test_named_branch_keep_stops_without_changing_set(self):
        state, *_ = current_state()
        victim = state["removable_ids"][0]
        choices = {"victim": f"remove_{victim}", f"replacement_for_{victim}": "KEEP"}
        new, _, stop = R.apply_global_action(state, choices)
        self.assertEqual(stop, "branch_keep")
        self.assertEqual(new["selected_ids"], state["selected_ids"])

    def test_one_to_four_replacements_protect_and_never_bounce(self):
        state, spec, scores, facets = current_state(32)
        receipts = []
        for _ in range(4):
            batch = R.build_batch(state)
            self.assertIsNotNone(batch)
            victim = state["removable_ids"][0]
            candidate = state["eligible_ids"][0]
            choices = {qid: "KEEP" for qid in batch["questions"]}
            choices["victim"] = f"remove_{victim}"
            choices[f"replacement_for_{victim}"] = f"replace_{candidate}"
            new, _, stop = R.apply_global_action(state, choices)
            self.assertEqual(stop, "accepted_replacement")
            self.assertNotIn(victim, new["eligible_ids"])
            self.assertNotIn(candidate, new["removable_ids"])
            state = new
        self.assertEqual(len(state["evicted_ids"]), 4)
        self.assertEqual(len(state["protected_ids"]), 4)
        self.assertEqual(state["step"], 5)
        self.assertEqual(R.build_batch(state), None)

    def test_score_floor_and_deterministic_no_action(self):
        state, *_ = current_state(11, 1)
        state["scores"][state["eligible_ids"][0]] = 4.999
        self.assertNotIn(state["eligible_ids"][0], R.state_for_round(state)["eligible_ids"])
        self.assertIsNone(R.build_batch(state))

    def test_invalid_choice_falls_back_to_full_w0(self):
        spec, scores = synthetic_spec(20)
        facets = [{"id": "f0", "description": "facet"}]
        d = {"scores": scores}
        dorder = R._stable_score_order(spec, d)
        w0 = list(reversed(dorder))
        batch = R.build_batch(R.initial_state(spec, d, facets))
        malformed = {"parsed": response_for(batch)}
        malformed["parsed"]["answers"]["victim"]["probabilities"]["KEEP"] = 0.2
        order, trace, fallback = R.relative_selection(spec, d, [malformed], w0, facets)
        self.assertTrue(fallback)
        self.assertEqual(order, w0)
        self.assertTrue(trace["fallback"])

    def test_state_and_request_caps_reject_before_transport(self):

    def test_w0_first40_scores_preserve_inherited_unscored_tail(self):

        receipt = verified["baseline"]["receipts"]["cardiac-w0"]

        self.assertEqual(order[40:], [cid for cid in spec["candidate_ids"] if cid not in set(scored)])

    def test_offline_published_gate_allows_neutral_phase_without_neutral_receipt(self):
        with tempfile.TemporaryDirectory(prefix="exp045-offline-gate-") as td:
            root = pathlib.Path(td)
            names = {"replay.py.txt", "fixture_tests.py.txt", "manual-choice-checks.py.txt", "publication-gates-checks.py.txt", "transport-fixtures.py.txt", "transport-qualification.json", "prepared-plan.json", "neutral-body.json", "neutral-inventory.json", "source-pins.json", "registration-integrity.json"}
            for name in names:
                src = HERE / name
                if name == "prepared-plan.json":
                    (root / name).write_bytes(R.pretty(R.make_prepared_plan(HERE)))
                elif name == "neutral-body.json":
                    (root / name).write_bytes(R.make_neutral_fixture()["body"])
                elif name == "neutral-inventory.json":
                    neutral = R.make_neutral_fixture()
                    (root / name).write_bytes(R.pretty({k: v for k, v in neutral.items() if k not in {"body", "state"}}))
                elif src.is_file():
                    shutil.copyfile(src, root / name)
                else:
                    (root / name).write_bytes(b"synthetic qualification fixture\n")
            files = {name: R.sha((root / name).read_bytes()) for name in names}
            qual = {"schema": "exp045-offline-qualification/1", "status": "passed", "files": files, "fixture_tests_passed": 13, "manual_checks_passed": 8, "synthetic_operations": 21, "primary_deltas": {"A": 8, "B": 8}, "pool_inventory": 13, "neutral_quality_failed": True, "provider_calls": 0, "search_calls": 0}
            (root / "offline-qualification.json").write_bytes(R.pretty(qual))
            with mock.patch.object(R, "_verify_publication_revision", return_value="a" * 64):
                gate = R.publication_gates(root, True, offline_revision="a" * 40, require_neutral=False)
            self.assertTrue(gate["offline"])
            self.assertFalse(gate["neutral"])

    def test_fake_21_operation_pipeline_reaches_both_reference_analysis_and_fails_neutral_gate(self):
        verified = R.validate_baseline()
        refs = verified["refs"]
        fake_ops = []
        for op in R.operation_inventory(verified):
            dspec = verified["specmap"][op["d_request_id"]]
            drec = verified["baseline"]["receipts"][op["d_request_id"]]
            state = R.initial_state(dspec, drec, op["facets"])
            batch = R.build_batch(state)
            parsed = response_for(batch)  # A complete neutral answer that keeps the current set.
            chosen, _ = R.validate_choice_response(parsed, batch)
            next_state, action, stop = R.apply_global_action(state, chosen)
            self.assertEqual(stop, "global_keep")
            order, _, fallback = R.relative_selection(dspec, drec, [{"parsed": parsed}], R._w0_order(verified["specmap"][op["w0_request_id"]], verified["baseline"]["receipts"][op["w0_request_id"]]), op["facets"])
            self.assertFalse(fallback)
            fake_ops.append({"operation_id": op["id"], "status": "completed", "order": order, "steps": 1})
        fake_results = {"schema": "exp045-operation-results/1", "status": "complete", "operations": fake_ops}
        synthetic, facet_ids, navigation = R.build_rankings_from_results(verified, fake_results)
        report = R.analyze_rankings(synthetic, refs, facet_ids)
        self.assertIn("development_screen_passed", report)
        self.assertFalse(report["development_screen_passed"])
        primary = report["primary"]
        expected_q = {f"q{i}" for i in range(1, 9)}
        for ref in ("A", "B"):
            self.assertEqual(set(primary[ref]["paired_deltas"]), expected_q)
        self.assertEqual(set(synthetic), set(R.SOFTWARE) | set(R.EXTENDED))
        self.assertEqual(len(synthetic), 13)
        self.assertEqual(len(R.operation_inventory(verified)), 21)
        self.assertEqual(len(navigation), 5)
