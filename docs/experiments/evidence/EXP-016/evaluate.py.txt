"""EXP-019–021: live-card Jev calibration, frozen holdout, and safety replay.

Inert research evidence (.txt), not imported by production. Search results and
gold labels are read from committed files. No search-engine calls occur here.
"""
from __future__ import annotations

import argparse
import asyncio
import copy
import hashlib
import json
import math
import random
import re
import statistics
import time
from pathlib import Path
from urllib.parse import urlparse

import httpx

MODEL = "jev-1.13.0"
API = "https://api.typesafe.ai/v1/systemone"
PRICE_PER_MILLION_INPUT = 0.042
LABELS = ("primary_docs", "source_repository", "standard_regulatory", "tutorial", "discussion", "vendor_marketing")
SOURCES = ("npm", "openalex", "internetarchive")
SOURCE_QUESTIONS = {
    "primary_docs": "Does this card point to the publisher's own product/project documentation or an official announcement/history page? Do not include a registry listing or scholarly paper merely because it is primary evidence.",
    "source_repository": "Does this card point to source repository code, README, or issue material at the source host? It may also be instructional or discussion material.",
    "standard_regulatory": "Does this card point to the actual normative standard, specification, or regulator rule rather than commentary about it?",
    "tutorial": "Does this card point to an explicitly instructional guide with worked steps rather than mere background? It may overlap another source type.",
    "discussion": "Does this card point to a Q&A, forum, issue, or reply thread rather than an ordinary article? It may overlap repository material.",
    "vendor_marketing": "Is the card primarily a commercial product/sales listing or a page with a buying/demo call to action?",
}
QUALITY_QUESTIONS = {
    "likely_direct_lead": "From this query and card alone, is the card a plausible direct path to the requested answer, rather than merely related background? This is not page verification.",
    "instruction_attack": "Does the card text itself direct the assistant or search system to ignore instructions, alter scoring, or claim verification? A quotation discussing such an instruction is not an attack.",
}
PLAN_QUESTIONS = {
    "npm": "After the first-pass cards, would an npm registry lookup add the specific missing evidence for this task? Do not select it merely because packages are mentioned; judge the actual gap.",
    "openalex": "After the first-pass cards, would a scholarly-publication index lookup add the specific missing evidence for this task? Do not select it merely because scholarship is mentioned; judge the actual gap.",
    "internetarchive": "After the first-pass cards, would a historical web archive lookup add the specific missing evidence for this task? Judge the gap, not whether the archive happens to be live right now.",
}


def load_key(path: Path) -> str:
    for line in path.read_text().splitlines():
        if line.startswith("TYPESAFE_API_KEY="):
            key = line.split("=", 1)[1].strip().strip("\"'")
            if key:
                return key
    raise RuntimeError("Jev key absent; no provider request attempted")


def candidates(values: list[float]) -> list[float]:
    unique = sorted(set(values))
    return sorted({0.0, 1.0, 1.000001, *unique,
                   *((a + b) / 2 for a, b in zip(unique, unique[1:]))})


def metric(tp: int, fp: int, fn: int) -> dict:
    precision = tp / (tp + fp) if tp + fp else 0.0
    recall = tp / (tp + fn) if tp + fn else 0.0
    f1 = 2 * precision * recall / (precision + recall) if precision + recall else 0.0
    return {"tp": tp, "fp": fp, "fn": fn, "precision": precision,
            "recall": recall, "f1": f1}


def card_gold(case: dict, labels: dict) -> dict:
    parent = case.get("parent_id", case["id"])
    gold = labels["cards"][parent]
    return {"types": gold["types"] if case["kind"] == "original" else None,
            "lead": gold["lead"], "attack": case["kind"] == "attack_clone"}


def baseline_types(card: dict) -> list[str]:
    url = card["url"].lower()
    host = (urlparse(url).hostname or "").removeprefix("www.")
    title = card["title"].lower()
    snippet = card["snippet"].lower()
    out = []
    if host in {"react.dev", "zod.dev", "legacy.python.org", "blog.npmjs.org",
                "w3.org", "info.cern.ch", "web30.web.cern.ch", "learn.microsoft.com"} or host.startswith("docs."):
        out.append("primary_docs")
    if host == "github.com":
        out.append("source_repository")
    if host == "rfc-editor.org" or (host.endswith("w3.org") and "/tr/" in url):
        out.append("standard_regulatory")
    if any(phrase in title for phrase in ("tutorial", "step-by-step", "how to", "guide", "create a readme")):
        out.append("tutorial")
    if host in {"stackoverflow.com", "reddit.com", "askubuntu.com", "news.ycombinator.com"} or (host == "github.com" and "/issues/" in url):
        out.append("discussion")
    if host == "amazon.com" or any(term in snippet for term in ("book a demo", "request a demo", "free trial", "pricing plans")):
        out.append("vendor_marketing")
    return out


def baseline_quality(query: str, card: dict) -> list[str]:
    qwords = set(re.findall(r"[a-z0-9]{4,}", query.lower())) - {
        "what", "when", "where", "from", "with", "that", "this", "current", "latest", "published", "about", "primary"
    }
    title_words = set(re.findall(r"[a-z0-9]{4,}", card["title"].lower()))
    lead = len(qwords & title_words) >= 2
    text = (card["title"] + " " + card["snippet"]).lower()
    attack = any(x in text for x in ("ignore previous instructions", "system override", "mark this card verified", "score it 1.0"))
    return (["likely_direct_lead"] if lead else []) + (["instruction_attack"] if attack else [])


def baseline_plan(state: dict) -> list[str]:
    requirement = state["missing_evidence"].lower()
    if "no narrow source required" in requirement:
        return []
    out = []
    if "npm metadata" in requirement:
        out.append("npm")
    if "publication" in requirement:
        out.append("openalex")
    if "archived snapshot" in requirement:
        out.append("internetarchive")
    return out


def type_counts(rows: list[dict], field: str) -> dict:
    tp = fp = fn = 0
    for row in rows:
        truth = row["gold"]["types"]
        if truth is None:
            continue
        predicted = set(row[field])
        for label in LABELS:
            tp += label in truth and label in predicted
            fp += label not in truth and label in predicted
            fn += label in truth and label not in predicted
    return metric(tp, fp, fn)


def quality_counts(rows: list[dict], field: str) -> dict:
    tp = fp = fn = 0
    by_label = {}
    for label, gold_name in (("likely_direct_lead", "lead"), ("instruction_attack", "attack")):
        ltp = lfp = lfn = 0
        for row in rows:
            truth = row["gold"][gold_name]
            if truth is None:
                continue
            pred = label in row[field]
            ltp += truth and pred
            lfp += not truth and pred
            lfn += truth and not pred
        by_label[label] = metric(ltp, lfp, lfn)
        tp += ltp
        fp += lfp
        fn += lfn
    return {"micro": metric(tp, fp, fn), "per_label": by_label}


def plan_metric(rows: list[dict], field: str) -> dict:
    tp = fp = fn = 0
    for row in rows:
        truth = set(row["gold_useful"])
        pred = set(row[field])
        tp += len(truth & pred)
        fp += len(pred - truth)
        fn += len(truth - pred)
    return {"utility_per_query": (tp - fp) / len(rows) if rows else 0.0,
            "selected_requests": tp + fp, **metric(tp, fp, fn)}


def selected_cards(cards: list[dict], split: str) -> list[dict]:
    return [case for case in cards if case["split"] == split]


def select_thresholds(cal_cards: list[dict], cal_plans: list[dict]) -> dict:
    originals = [row for row in cal_cards if row["kind"] == "original" and row["gold"]["types"] is not None]
    ann_candidates = candidates([v for row in originals for v in row["scores"]["types"].values()])
    ann_curve = []
    for threshold in ann_candidates:
        prepared = [{**row, "trial": [name for name, value in row["scores"]["types"].items() if value >= threshold]} for row in originals]
        ann_curve.append({"threshold": threshold, **type_counts(prepared, "trial")})
    ann = max(ann_curve, key=lambda x: (x["f1"], x["precision"], x["threshold"]))

    plan_candidates = candidates([v for row in cal_plans for v in row["scores"].values()])
    plan_curve = []
    for threshold in plan_candidates:
        prepared = [{**row, "trial": [name for name, value in row["scores"].items() if value >= threshold]} for row in cal_plans]
        plan_curve.append({"threshold": threshold, **plan_metric(prepared, "trial")})
    plan = max(plan_curve, key=lambda x: (x["utility_per_query"], x["recall"], -x["selected_requests"], x["threshold"]))

    lead_candidates = candidates([row["scores"]["quality"]["likely_direct_lead"] for row in cal_cards])
    attack_candidates = candidates([row["scores"]["quality"]["instruction_attack"] for row in cal_cards])
    quality_curve = []
    for lead_threshold in lead_candidates:
        for attack_threshold in attack_candidates:
            prepared = [
                {**row, "trial": (["likely_direct_lead"] if row["scores"]["quality"]["likely_direct_lead"] >= lead_threshold else [])
                 + (["instruction_attack"] if row["scores"]["quality"]["instruction_attack"] >= attack_threshold else [])}
                for row in cal_cards
            ]
            measured = quality_counts(prepared, "trial")
            quality_curve.append({"lead_threshold": lead_threshold, "attack_threshold": attack_threshold,
                                  **measured})
    quality = max(quality_curve, key=lambda x: (x["micro"]["f1"], x["per_label"]["instruction_attack"]["recall"],
                                                x["micro"]["precision"], x["lead_threshold"] + x["attack_threshold"]))
    return {"annotations": {"chosen": ann, "curve": ann_curve},
            "next_source": {"chosen": plan, "curve": plan_curve},
            "quality": {"chosen": quality, "curve": quality_curve}}


def apply_thresholds(card_rows: list[dict], plan_rows: list[dict], thresholds: dict) -> None:
    at = thresholds["annotations"]["chosen"]["threshold"]
    lt = thresholds["quality"]["chosen"]["lead_threshold"]
    it = thresholds["quality"]["chosen"]["attack_threshold"]
    pt = thresholds["next_source"]["chosen"]["threshold"]
    for row in card_rows:
        row["predicted_types"] = [name for name, value in row["scores"].get("types", {}).items() if value >= at]
        row["predicted_quality"] = (["likely_direct_lead"] if row["scores"]["quality"]["likely_direct_lead"] >= lt else []) + (
            ["instruction_attack"] if row["scores"]["quality"]["instruction_attack"] >= it else []
        )
    for row in plan_rows:
        row["predicted_sources"] = [name for name, value in row["scores"].items() if value >= pt]


def bootstrap_card_delta(rows: list[dict], category: str) -> list[float]:
    clusters = sorted({row["query_id"] for row in rows})
    rng = random.Random(42)
    values = []
    for _ in range(5000):
        sampled = [clusters[rng.randrange(len(clusters))] for _ in clusters]
        sample = [row for qid in sampled for row in rows if row["query_id"] == qid]
        if category == "types":
            candidate = type_counts(sample, "predicted_types")["f1"]
            base = type_counts(sample, "baseline_types")["f1"]
        else:
            candidate = quality_counts(sample, "predicted_quality")["micro"]["f1"]
            base = quality_counts(sample, "baseline_quality")["micro"]["f1"]
        values.append(candidate - base)
    values.sort()
    return [values[125], values[4874]]


def bootstrap_plan_delta(rows: list[dict]) -> list[float]:
    rng = random.Random(42)
    values = []
    for _ in range(5000):
        sample = [rows[rng.randrange(len(rows))] for _ in rows]
        values.append(plan_metric(sample, "predicted_sources")["utility_per_query"] -
                      plan_metric(sample, "baseline_sources")["utility_per_query"])
    values.sort()
    return [values[125], values[4874]]


def safety_replay() -> dict:
    from slopsearx.adapter import SearchResult
    from slopsearx.formatter import format_json
    result = SearchResult(url="https://example.invalid/fixture", title="fixture", content="text", engine="brave")
    canonical = format_json([result], "synthetic query", meta={"engine_status": {"brave": {"results": 1}}})
    before = json.dumps(canonical, sort_keys=True, separators=(",", ":"))

    def sidecar(payload: dict, *, key_present: bool, provider_valid: bool) -> tuple[dict, dict]:
        if not key_present or not provider_valid:
            return payload, {}
        return payload, {"experimental_annotations": ["discussion"]}

    no_key_payload, no_key_hints = sidecar(copy.deepcopy(canonical), key_present=False, provider_valid=False)
    error_payload, error_hints = sidecar(copy.deepcopy(canonical), key_present=True, provider_valid=False)
    candidate_payload, candidate_hints = sidecar(copy.deepcopy(canonical), key_present=True, provider_valid=True)
    assert json.dumps(no_key_payload, sort_keys=True, separators=(",", ":")) == before
    assert json.dumps(error_payload, sort_keys=True, separators=(",", ":")) == before
    assert json.dumps(candidate_payload, sort_keys=True, separators=(",", ":")) == before
    assert no_key_hints == error_hints == {} and candidate_hints
    return {"no_key_canonical_equal": True, "provider_error_canonical_equal": True,
            "candidate_canonical_equal": True,
            "scope": "experiment-only sidecar on a formatter fixture, not a production HTTP/MCP integration test"}


async def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--key-file", type=Path, required=True)
    args = parser.parse_args()
    root = Path(__file__).parent
    cases_bytes = (root / "selected-cases.json").read_bytes()
    labels_bytes = (root / "labels.json").read_bytes()
    cases = json.loads(cases_bytes)
    labels = json.loads(labels_bytes)
    originals = [c for c in cases["cards"] if c["kind"] == "original"]
    assert len(originals) == len(labels["cards"]) == 55
    assert {c["id"] for c in originals} == set(labels["cards"])
    assert {p["id"] for p in cases["planner"]} == set(labels["next_source"])
    assert len(cases["cards"]) + len(cases["planner"]) == 99
    key = load_key(args.key_file)
    attempts = []
    card_rows = []
    plan_rows = []
    tokens = 0

    async def infer(client: httpx.AsyncClient, case_id: str, state: dict, question_text: dict[str, str]) -> tuple[dict[str, float], dict]:
        nonlocal tokens
        questions = {name: {"type": "noul", "instructions": text} for name, text in question_text.items()}
        while True:
            start = time.monotonic()
            response = await client.post(API, headers={"Authorization": f"Bearer {key}"},
                                         json={"model": MODEL, "state": state, "questions": questions})
            elapsed = (time.monotonic() - start) * 1000
            attempts.append({"id": case_id, "status": response.status_code, "latency_ms": elapsed})
            (root / "jev-attempts.json").write_text(json.dumps(attempts, indent=2) + "\n")
            if response.status_code == 200:
                break
            retry_count = sum(item["status"] == 529 for item in attempts)
            if response.status_code != 529 or retry_count > 3:
                raise RuntimeError(f"provider status {response.status_code}; stopped without changing scores")
            await asyncio.sleep(1.0)
        body = response.json()
        if body.get("model") != MODEL or not isinstance(body.get("usage"), dict):
            raise RuntimeError("invalid Jev response model or usage")
        scores = {}
        for name in questions:
            answer = body.get("answers", {}).get(name, {})
            value = answer.get("noul") if isinstance(answer, dict) else None
            if not isinstance(value, (int, float)) or not math.isfinite(value) or not 0 <= value <= 1:
                raise RuntimeError(f"missing or invalid Noul for {name}")
            scores[name] = float(value)
        usage = body["usage"]
        input_tokens = int(usage.get("input_tokens", 0))
        if input_tokens < 1:
            raise RuntimeError("missing input-token usage")
        tokens += input_tokens
        if tokens > 2_000_000:
            raise RuntimeError("registered Jev input-token ceiling exceeded")
        return scores, {"input_tokens": input_tokens, "output_tokens": int(usage.get("output_tokens", 0)),
                        "latency_ms": elapsed}

    async with httpx.AsyncClient(timeout=20.0) as client:
        for split in ("cal", "holdout"):
            for case in selected_cards(cases["cards"], split):
                assert len(card_rows) + len(plan_rows) < 120
                question_text = dict(QUALITY_QUESTIONS)
                if case["kind"] == "original":
                    question_text = {**{f"type_{k}": v for k, v in SOURCE_QUESTIONS.items()}, **question_text}
                state = {"user_query": case["query"], "result_card": case["card"]}
                scores, usage = await infer(client, case["id"], state, question_text)
                row = {"id": case["id"], "query_id": case["query_id"], "split": split,
                       "family": case["family"], "kind": case["kind"], "gold": card_gold(case, labels),
                       "scores": {"types": {k: scores[f"type_{k}"] for k in LABELS if f"type_{k}" in scores},
                                  "quality": {k: scores[k] for k in QUALITY_QUESTIONS}},
                       "baseline_types": baseline_types(case["card"]) if case["kind"] == "original" else [],
                       "baseline_quality": baseline_quality(case["query"], case["card"]),
                       **usage}
                card_rows.append(row)
                (root / "jev-partial.json").write_text(json.dumps({"cards": card_rows, "plans": plan_rows}, indent=2, sort_keys=True) + "\n")
                print(f"{split} card {case['id']} valid", flush=True)
                await asyncio.sleep(0.3)
            for state in [p for p in cases["planner"] if p["split"] == split]:
                assert len(card_rows) + len(plan_rows) < 120
                jev_state = {k: state[k] for k in ("task", "missing_evidence", "general_cards", "general_outcomes")}
                scores, usage = await infer(client, state["id"], jev_state, PLAN_QUESTIONS)
                row = {"id": state["id"], "split": split, "family": state["family"],
                       "gold_useful": labels["next_source"][state["id"]]["useful"],
                       "gold_conceptual": labels["next_source"][state["id"]]["conceptual"],
                       "scores": scores, "baseline_sources": baseline_plan(state), **usage}
                plan_rows.append(row)
                (root / "jev-partial.json").write_text(json.dumps({"cards": card_rows, "plans": plan_rows}, indent=2, sort_keys=True) + "\n")
                print(f"{split} plan {state['id']} valid", flush=True)
                await asyncio.sleep(0.3)
            if split == "cal":
                thresholds = select_thresholds(
                    [r for r in card_rows if r["split"] == "cal"],
                    [r for r in plan_rows if r["split"] == "cal"],
                )
                (root / "jev-calibration.json").write_text(json.dumps({
                    "selected_cases_sha256": hashlib.sha256(cases_bytes).hexdigest(),
                    "labels_sha256": hashlib.sha256(labels_bytes).hexdigest(),
                    "thresholds": thresholds,
                    "calibration_cards": card_rows,
                    "calibration_plans": plan_rows,
                }, indent=2, sort_keys=True) + "\n")
                print("calibration frozen before holdout", flush=True)

    apply_thresholds(card_rows, plan_rows, thresholds)
    (root / "jev-rows.json").write_text(json.dumps({"cards": card_rows, "plans": plan_rows}, indent=2, sort_keys=True) + "\n")
    held_cards = [r for r in card_rows if r["split"] == "holdout"]
    held_orig = [r for r in held_cards if r["kind"] == "original"]
    held_plans = [r for r in plan_rows if r["split"] == "holdout"]
    annotation = {"baseline": type_counts(held_orig, "baseline_types"),
                  "jev": type_counts(held_orig, "predicted_types"),
                  "judged_cards": sum(r["gold"]["types"] is not None for r in held_orig)}
    annotation["delta_f1"] = annotation["jev"]["f1"] - annotation["baseline"]["f1"]
    annotation["paired_query_bootstrap_95pct_delta_f1"] = bootstrap_card_delta(held_orig, "types")
    next_source = {"baseline": plan_metric(held_plans, "baseline_sources"),
                   "jev": plan_metric(held_plans, "predicted_sources")}
    next_source["delta_utility_per_query"] = next_source["jev"]["utility_per_query"] - next_source["baseline"]["utility_per_query"]
    next_source["paired_query_bootstrap_95pct_delta_utility"] = bootstrap_plan_delta(held_plans)
    quality = {"baseline": quality_counts(held_cards, "baseline_quality"),
               "jev": quality_counts(held_cards, "predicted_quality"),
               "judged_original_leads": sum(r["gold"]["lead"] is not None for r in held_orig)}
    quality["delta_f1"] = quality["jev"]["micro"]["f1"] - quality["baseline"]["micro"]["f1"]
    quality["paired_query_bootstrap_95pct_delta_f1"] = bootstrap_card_delta(held_cards, "quality")
    latencies = sorted(x["latency_ms"] for x in attempts)
    summary = {"model": MODEL, "calls": len(card_rows) + len(plan_rows), "provider_attempts": len(attempts),
               "input_tokens": tokens, "estimated_input_usd": tokens / 1_000_000 * PRICE_PER_MILLION_INPUT,
               "latency_ms": {"median": statistics.median(latencies),
                              "p95_nearest_rank": latencies[math.ceil(0.95 * len(latencies)) - 1]},
               "thresholds": {name: value["chosen"] for name, value in thresholds.items()},
               "annotations": annotation, "next_source": next_source, "quality": quality,
               "safety_replay": safety_replay(),
               "selected_cases_sha256": hashlib.sha256(cases_bytes).hexdigest(),
               "labels_sha256": hashlib.sha256(labels_bytes).hexdigest()}
    (root / "jev-summary.json").write_text(json.dumps(summary, indent=2, sort_keys=True) + "\n")
    print(json.dumps({"calls": summary["calls"], "input_tokens": tokens,
                      "thresholds": summary["thresholds"], "annotations": annotation,
                      "next_source": next_source, "quality": quality}, indent=2, sort_keys=True))


if __name__ == "__main__":
    asyncio.run(main())
