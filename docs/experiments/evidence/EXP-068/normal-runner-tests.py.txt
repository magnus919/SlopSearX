def load(name,path):
    loader=importlib.machinery.SourceFileLoader(name,str(path));spec=importlib.util.spec_from_loader(name,loader);mod=importlib.util.module_from_spec(spec);loader.exec_module(mod);return mod

def card(i):return {"id":"x%02d"%i,"url":"https://example.test/%02d"%i,"title":"Example source %02d"%i,"snippet":"A bounded public metadata excerpt %02d."%i}
def case():return {"id":"q1","query":"ordinary query","purpose":"Find strong public reading leads for this task.","facets":[{"id":"f1","description":"implementation"},{"id":"f2","description":"validation"}]}
def response(request,*,usage=True,score=2,metadata=True):
    answers={}
    for qid,q in request["questions"].items():
        answers[qid]={"type":"score","score":score,**({"confidence":.8,"ignored":"metadata"} if metadata else {})}
    data={"model":P.MODEL,"answers":answers}
    if usage:data["usage"]={"input_tokens":11,"output_tokens":3}
    return json.dumps(data,allow_nan=False,separators=(",",":")).encode()

class PrepareTests(unittest.TestCase):
    def test_relative_request_uses_full_canonical_pool_and_trusted_global_id_questions(self):
        cards=[card(i) for i in range(80)];built=P.bodies(case(),cards);req=built["relative"]
        self.assertEqual(list(req["questions"]),built["ids"]);self.assertEqual(len(req["state"]["candidates"]),80)
        self.assertEqual(req["state"]["purpose"],case()["purpose"]);self.assertEqual(req["state"]["requested_facets"],case()["facets"])
        for cid,q in req["questions"].items():
            self.assertEqual(q,{"type":"score","instructions":P.PROTOCOL["instructions"].format(id=cid),"criteria":list(P.PROTOCOL["criteria"])})
        self.assertEqual(len(P.PROTOCOL["criteria"]),5)

    def test_http_request_boundary_uses_exact_registered_scope_first_instruction(self):
        built=P.bodies(case(),[card(0),card(1)]);req=built["relative"]
        cid=next(iter(req["questions"]))
        self.assertIn("First establish task scope",P.PROTOCOL["instructions"])
        self.assertEqual(req["questions"][cid]["instructions"],P.PROTOCOL["instructions"].format(id=cid))
        self.assertEqual(P.validate_request(P.wire(req)),req)
        changed=json.loads(P.wire(req));changed["questions"][cid]["instructions"]=changed["questions"][cid]["instructions"].replace("First establish task scope","Establish task scope")
        with self.assertRaises(ValueError):P.validate_request(P.wire(changed))

    def test_relative_body_is_rotation_invariant_and_w0_preserves_native_full_tail(self):
        cards=[card(i) for i in range(59)];base=P.bodies(case(),cards);rot=P.bodies(case(),cards,"rotate")
        self.assertEqual(base["relative"],rot["relative"])
        self.assertEqual(rot["provenance_order_ids"],base["provenance_order_ids"][20:]+base["provenance_order_ids"][:20])
        self.assertEqual(base["w0"]["state"]["query"],case()["query"]);self.assertEqual(len(base["w0"]["questions"]),40)
        self.assertEqual(base["tail_ids"],["x%02d"%i for i in range(40,59)])

    def test_relative_score_order_descending_with_canonical_ties_only(self):
        ids=["z","a","m"];scores={"z":4,"a":4,"m":1}
        self.assertEqual(P.relative_order(scores,ids),["z","a","m"])
        built=P.bodies(case(),[card(0),card(1),card(2)])
        self.assertEqual(P.relative_order({cid:2 for cid in built["ids"]},built["ids"]),built["ids"])
        with self.assertRaises(ValueError):P.relative_order(scores,["z","z","m"])

    def test_response_schema_bounds_metadata_and_usage(self):
        built=P.bodies(case(),[card(0),card(1)]);body=built["relative"]
        self.assertEqual(P.validate_relative_response(response(body),body,built["ids"])["usage"],{"input_tokens":11,"output_tokens":3})
        w0=built["w0"];w0_ids=list(w0["questions"])
        self.assertEqual(P.validate_score_response(response(w0,score=9),w0,w0_ids)["scores"][w0_ids[0]],9.0)
        with self.assertRaises(ValueError):P.validate_score_response(response(w0,score=9.1),w0,w0_ids)
        for score in (-.1,4.1,True):
            raw=response(body,score=score)
            with self.subTest(score=score),self.assertRaises(ValueError):P.validate_relative_response(raw,body,built["ids"])
        with self.assertRaises(ValueError):P.validate_relative_response(response(body,score=4.1).replace(b'"score":4.1',b'"score":NaN'),body,built["ids"])
        duplicate=b'{"model":"jev-1.13.0","model":"jev-1.13.0","answers":{}}'
        with self.assertRaisesRegex(ValueError,"duplicate-json-key"):P.validate_relative_response(duplicate,body,built["ids"])
        with self.assertRaises(ValueError):P.validate_relative_response(b" "*2_000_001,body,built["ids"])
        self.assertEqual(P.usage_from_raw(response(body,score=8)),{"input_tokens":11,"output_tokens":3})
        self.assertIsNone(P.validate_relative_response(response(body,usage=False),body,built["ids"])["usage"])

    def test_request_validator_rejects_misbinding_criteria_omission_and_bad_w0_fields(self):
        built=P.bodies(case(),[card(0),card(1)]);request=json.loads(P.wire(built["relative"]))
        cid=next(iter(request["questions"]));request["questions"][cid]["instructions"]="compare wrong card"
        with self.assertRaises(ValueError):P.validate_request(P.wire(request))
        request=json.loads(P.wire(built["relative"]));request["questions"][cid]["criteria"][0]="changed criterion"
        with self.assertRaises(ValueError):P.validate_request(P.wire(request))
        request=json.loads(P.wire(built["relative"]));request["questions"].pop(cid)
        with self.assertRaises(ValueError):P.validate_request(P.wire(request))
        request=json.loads(P.wire(built["relative"]));request["state"]["candidates"][0]["id"]="wrong-global-id"
        with self.assertRaises(ValueError):P.validate_request(P.wire(request))
        for field,value in (("url",""),("url","u"*513),("title","t"*257),("snippet","s"*1201),("id","i"*81)):
            request=json.loads(P.wire(built["w0"]));request["state"]["candidates"][0][field]=value
            with self.subTest(field=field),self.assertRaises(ValueError):P.validate_request(P.wire(request))
        with self.assertRaises(ValueError):P.bodies(case(),[card(i) for i in range(81)])

    def test_invalid_relative_score_with_known_usage_fails_once_and_falls_back_to_fresh_w0(self):

        class FakeOwned:
            calls=0
            def __init__(self,*_):self.last_raw=None
            def __call__(self,body):
                FakeOwned.calls+=1;request=P.strict_json(body)
                if FakeOwned.calls==1:self.last_raw=response(request,score=2);return self.last_raw,1.0
                if FakeOwned.calls==2:self.last_raw=response(request,score=5);return self.last_raw,1.0
                if FakeOwned.calls==3:self.last_raw=response(request,score=5);return self.last_raw,1.0
                raise AssertionError("retry or later dispatch")
        with tempfile.TemporaryDirectory() as tmp,patch.object(N,"Owned",FakeOwned):
            result=N.run(acq,queries,Path(tmp)/"failed",True)
            self.assertEqual(result["status"],"incomplete");self.assertEqual(FakeOwned.calls,3)
            first=result["operations"][0];self.assertEqual(first["arms"]["w0"]["status"],"valid")
            self.assertEqual(first["arms"]["relative"]["status"],"failed")
            self.assertEqual(first["arms"]["relative"]["order_ids"],first["arms"]["w0"]["order_ids"])

            self.assertEqual(result["budget"]["input_tokens"],33);self.assertFalse(result["budget"]["unknown_usage"])
            receipt=json.loads((Path(tmp)/"failed"/"receipts"/"call-02.json").read_bytes())
