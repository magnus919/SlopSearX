"""Owned exact-byte TypeSafe HTTP wrapper; explicit offline fixtures only."""
import argparse
import io
import hashlib
import json
import signal
from contextlib import contextmanager
import os
import sys
import time
import urllib.error
import urllib.request

ENDPOINT = 'https://api.typesafe.ai/v1/systemone'
LIMIT = 2000000

class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


def validate_request(body):
    if type(body) is not bytes or len(body) > 384000:
        raise ValueError('request-bound')
    data = json.loads(body)
    if type(data) is not dict or set(data) != {'model', 'state', 'questions'} or data['model'] != 'jev-1.13.0':
        raise ValueError('request-api-shape')
    questions = data['questions']
    if type(questions) is not dict or not 1 <= len(questions) <= 80:
        raise ValueError('request-question-inventory')
    for question in questions.values():
        if type(question) is not dict or set(question) != {'type', 'instructions', 'criteria'} or question['type'] != 'noul':
            raise ValueError('request-noul-contract')
        if type(question['instructions']) is not str or not question['instructions']:
            raise ValueError('request-instructions')
        criteria = question['criteria']
        if type(criteria) is not dict or set(criteria) != {'true', 'false'} or any(type(v) is not str for v in criteria.values()):
            raise ValueError('request-criteria')
    return data


@contextmanager
def remote_deadline():
    old_handler = signal.getsignal(signal.SIGALRM)
    old_timer = signal.getitimer(signal.ITIMER_REAL)
    started = time.monotonic()
    def expired(signum, frame):
        raise TimeoutError('remote-hard-deadline')
    signal.signal(signal.SIGALRM, expired)
    signal.setitimer(signal.ITIMER_REAL, min(1,old_timer[0]) if old_timer[0] else 1)
    try:
        yield
    finally:
        signal.setitimer(signal.ITIMER_REAL,0)
        signal.signal(signal.SIGALRM,old_handler)
        if old_timer[0]:signal.setitimer(signal.ITIMER_REAL,max(.000001,old_timer[0]-(time.monotonic()-started)),old_timer[1])


def exchange(body, token, opener, clock=time.monotonic, timing=None):
    validate_request(body)
    req = urllib.request.Request(ENDPOINT, data=body, method='POST',
        headers={'Authorization': 'Bearer ' + token, 'Content-Type': 'application/json'})
    started = clock()
    with remote_deadline(), opener.open(req, timeout=1) as response:
        if response.status != 200:
            raise ValueError('http-status')
        buffer = bytearray()
        while True:
            if clock() - started >= 1:
                raise TimeoutError('remote-deadline')
            chunk = response.read(min(65536, LIMIT + 1 - len(buffer)))
            if not chunk:
                break
            buffer.extend(chunk)
            if len(buffer) > LIMIT:
                raise ValueError('response-bound')
        elapsed = clock() - started
        if elapsed >= 1:
            raise TimeoutError('remote-deadline')
        if timing is not None:
            timing['remote_http_ms'] = elapsed * 1000
    return bytes(buffer)


class FixtureResponse(io.BytesIO):
    status = 200

class FixtureOpener:
    def __init__(self, kind): self.kind = kind
    def open(self, request, timeout):
        assert request.full_url == ENDPOINT and request.method == 'POST' and timeout == 1
        if self.kind == 'http-error':
            raise urllib.error.HTTPError(ENDPOINT, 503, 'fixture', {}, None)
        data = validate_request(request.data)
        raw = json.dumps({'model': 'jev-1.13.0', 'answers': {
            qid: {'type': 'noul', 'noul': .5} for qid in data['questions']},
            'usage': {'input_tokens': 1, 'output_tokens': 1}}).encode()
        return FixtureResponse(raw)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--offline-fixture', choices=['neutral', 'http-error'])
    args = parser.parse_args()
    body = sys.stdin.buffer.read(384001)
    if args.offline_fixture:
        token, opener = 'offline-fixture', FixtureOpener(args.offline_fixture)
    else:
        token = os.environ.get('TYPESAFE_API_KEY', '')
        if not token:
            print('missing-credential', file=sys.stderr)
            return 2
        opener = urllib.request.build_opener(NoRedirect())
    try:
        timing = {}
        raw = exchange(body, token, opener, timing=timing)
    except Exception as error:
        # Never serialize request, headers, endpoint, exception message or secrets.
        print('owned-http-' + type(error).__name__, file=sys.stderr)
        return 2
    sys.stdout.buffer.write(raw)
    print(json.dumps({'remote_http_ms': timing['remote_http_ms'], 'request_sha256': hashlib.sha256(body).hexdigest(), 'response_sha256': hashlib.sha256(raw).hexdigest()}), file=sys.stderr)
    return 0

if __name__ == '__main__': sys.exit(main())
