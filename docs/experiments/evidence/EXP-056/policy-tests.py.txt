def pool(n=20, facets_n=2):
    ids = [f"c{i}" for i in range(n)]
    cards = [{"id": cid, "title": f"Synthetic title {cid}", "url": f"https://fixture.invalid/{cid}", "snippet": f"Synthetic visible snippet for {cid}."} for cid in ids]
    scores = {cid: float(9 - i / 20) if i < min(10, n) else 6.0 - i / 100 for i, cid in enumerate(ids)}
    facets = [{"id": f"facet-{i}", "description": f"Visible facet description {i}"} for i in range(facets_n)]
    nouls = {cid: {i: 0.1 for i in range(facets_n)} for cid in ids}
    return ids, cards, scores, facets, nouls


def raises(call, label):
    try:
        call()
    except (TypeError, ValueError):
        return
    raise AssertionError(f"expected rejection: {label}")


def raises_reason(call, phrase, label):
    try:
        call()
    except ValueError as exc:
        assert phrase in str(exc), f"{label}: unexpected rejection reason {exc!r}"
        return
    raise AssertionError(f"expected {phrase!r} rejection: {label}")


def batch_for(n=20, facets_n=2, nominated=None, protected=(), evicted=(), round_number=1):
    ids, cards, scores, facets, nouls = pool(n, facets_n)
    if nominated:
        for facet_i, candidate_i in enumerate(nominated):
            for cid in ids:
                nouls[cid][facet_i] = 0.2
            if candidate_i is not None:
                nouls[candidate_i][facet_i] = 0.9
    request, batch = build_round(cards, "Find useful source leads", facets, scores, nouls,
                                 ids[:min(10, n)], list(protected), list(evicted), round_number)
    return ids, cards, scores, facets, nouls, request, batch


def values(batch, addition, loss):
    return {qid: (addition if binding["kind"] == "addition" else loss) for qid, binding in batch["bindings"].items()}

# Exercise the exact state-byte boundary with default json.dumps spacing. Every source
# field remains inside its individual UTF-8 byte cap, and source bytes are unchanged.
def exact_state_request(cards_for_state, target, purpose_value="Purpose"):
    request0, _ = build_round(cards_for_state, purpose_value, facets80, scores80, nouls80, ids80[:10], [], [], 1)
    current = len(__import__("json").dumps(request0["state"], ensure_ascii=False).encode("utf-8"))
    assert current <= target
    remaining = target - current
    for card in cards_for_state:
        for field, limit in (("title", 256), ("url", 512), ("snippet", 1200)):
            room = limit - len(card[field].encode("utf-8"))
            take = min(room, remaining)
            card[field] += "x" * take
            remaining -= take
            if remaining == 0:
                break
        if remaining == 0:
            break
    assert remaining == 0, "synthetic source fields lacked byte capacity to reach boundary"
    exact_request, _ = build_round(cards_for_state, purpose_value, facets80, scores80, nouls80, ids80[:10], [], [], 1)
    exact_state = len(__import__("json").dumps(exact_request["state"], ensure_ascii=False).encode("utf-8"))
    assert exact_state == target
    return exact_request
