"""Exact HTTP contract with fake response streams, never network requests."""
import importlib.machinery
import importlib.util
import io
from pathlib import Path
import unittest

p = Path(__file__).with_name('http-wrapper.py.txt')
l = importlib.machinery.SourceFileLoader('httpowned', str(p))
s = importlib.util.spec_from_loader(l.name, l)
m = importlib.util.module_from_spec(s)
l.exec_module(m)

class Opener:
    def __init__(self, raw, status=200): self.raw, self.status, self.calls = raw, status, []
    def open(self, request, timeout):
        self.calls.append((request, timeout))
        response = io.BytesIO(self.raw)
        response.status = self.status
        return response

class Tests(unittest.TestCase):
    def test_exact_body_response_and_fixed_destination(self):
        raw = b'{"unmodified": 0.5001}\n'
        opener = Opener(raw)
        body = b'{ "questions": {} }\n'
        self.assertEqual(m.exchange(body, 'fixture-token', opener), raw)
        req, timeout = opener.calls[0]
        self.assertEqual(req.full_url, m.ENDPOINT)
        self.assertEqual(req.data, body)
        self.assertEqual(req.method, 'POST')
        self.assertEqual(timeout, 1)
        self.assertEqual(req.get_header('Authorization'), 'Bearer fixture-token')
    def test_status_failure_never_retries(self):
        opener = Opener(b'error', 503)
        with self.assertRaises(ValueError): m.exchange(b'{}', 'fixture', opener)
        self.assertEqual(len(opener.calls), 1)
    def test_exact_response_cap_and_one_more(self):
        self.assertEqual(len(m.exchange(b'{}', 'fixture', Opener(b'x'*2000000))), 2000000)
        with self.assertRaises(ValueError): m.exchange(b'{}', 'fixture', Opener(b'x'*2000001))
    def test_request_overflow_prevents_call(self):
        opener = Opener(b'{}')
        with self.assertRaises(ValueError): m.exchange(b'x'*384001, 'fixture', opener)
        self.assertEqual(opener.calls, [])
    def test_remote_clock_deadline(self):
        ticks = iter([0, 1])
        with self.assertRaises(TimeoutError): m.exchange(b'{}', 'fixture', Opener(b'{}'), clock=lambda: next(ticks))
    def test_redirect_handler_does_not_forward_auth(self):
        self.assertIsNone(m.NoRedirect().redirect_request(None, None, 302, '', {}, 'https://fixture.invalid/'))

if __name__ == '__main__': unittest.main()
