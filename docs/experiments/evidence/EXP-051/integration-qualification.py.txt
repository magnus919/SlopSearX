"""Independent EXP-051 lossless source/metadata integration preflight.

No reviewer labels, network, or final packet files are produced.
"""
import argparse
import hashlib
import json
import os
import runpy
from pathlib import Path

BASE = Path(__file__).resolve().parent
ROOT = BASE.parents[3]

def check(projection_path):
    builder_path = BASE / "delivery-builder.py.txt"
    module = runpy.run_path(str(builder_path))
    raw = Path(projection_path).read_bytes()
    inventory = json.loads((BASE / "input-inventory.json").read_bytes())
    assert hashlib.sha256(raw).hexdigest() == inventory["registered_private_projection_sha256"]
    projection = json.loads(raw)
    index = {}
    for row in projection["pools"]:
        key = tuple(row["card"][field] for field in ("title", "url", "snippet"))
        assert key not in index
        index[key] = row["metadata"]
    packets, chunks, manifest = module["prepare_delivery"](Path(projection_path), ROOT)
    reconstructed = {f"P{i:02d}": [] for i in range(1, 7)}
    metadata_checked = 0
    for name, pool_ref, ordinal, total, body, count in chunks:
        assert len(body) <= 24000
        chunk = json.loads(body)
        assert chunk["pool_ref"] == pool_ref and chunk["chunk_index"] == ordinal and chunk["chunk_total"] == total
        assert chunk["end_marker"] == f"EXP051-{pool_ref}-CHUNK-{ordinal:02d}-OF-{total:02d}-END"
        assert count == len(chunk["cards"])
        source = next(p for p in packets if p["pool"]["pool_ref"] == pool_ref)
        for field in ("purpose", "facets", "selected_ids"):
            assert chunk[field] == source["pool"][field]
        assert chunk["cases"] == source["cases"]
        for row in chunk["cards"]:
            card = row["card"]
            key = tuple(card[field] for field in ("title", "url", "snippet"))
            expected = json.loads(json.dumps(index[key]))
            abstract = expected.get("fields", {}).get("abstract")
            if abstract is not None:
                del abstract["original_sha256"]
            assert row["metadata"] == expected
            assert set(row) == {"card", "metadata"}
            reconstructed[pool_ref].append(card)
            metadata_checked += 1
        entry = next(e for e in manifest["chunks"] if e["filename"] == name)
        assert entry["sha256"] == hashlib.sha256(body).hexdigest()
        assert entry["bytes"] == len(body)
    for source in packets:
        assert reconstructed[source["pool"]["pool_ref"]] == source["pool"]["candidates"]
    assert metadata_checked == 232
    return {"schema": "exp051-independent-packet-preflight/1", "status": "passed",
            "builder_sha256": hashlib.sha256(builder_path.read_bytes()).hexdigest(),
            "checks": ["exact_projection_input_hash", "unique_original_visible_card_joins",
                       "all_232_metadata_rows_exact_except_digest_omission", "all_original_cards_and_order",
                       "all_repeated_context_and_cases", "all_chunk_markers_and_byte_limits",
                       "all_manifest_chunk_hashes"],
            "metadata_rows_checked": metadata_checked, "pool_count": 6, "case_count": 12,
            "chunk_count": len(chunks), "max_chunk_bytes": max(len(c[4]) for c in chunks),
            "packets_written": 0, "reviewer_invocations": 0, "provider_calls": 0, "search_calls": 0}

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--projection", default=os.environ.get("EXP051_PROJECTION"))
    args = parser.parse_args()
    if not args.projection:
        parser.error("provide --projection or EXP051_PROJECTION")
    print(json.dumps(check(Path(args.projection)), indent=2))
