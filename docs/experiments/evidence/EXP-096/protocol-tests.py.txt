import hashlib
import json
import unittest
from pathlib import Path

HERE = Path(__file__).resolve().parent
PREDECESSOR_PROTOCOL = HERE.parent / "EXP-094" / "protocol.json"
CURRENT = HERE / "protocol.json"


class ProtocolTests(unittest.TestCase):
    def test_answer_grouped_representation_is_explicit_and_source_wire_is_frozen(self):
        protocol = json.loads(CURRENT.read_text())
        representation = protocol["answer_assessment_packets"]["representation"]
        self.assertEqual(representation["schema"], "exp096-grouped-evidence-catalog/1")
        self.assertEqual(representation["scope"], "answer_assessor_model_input_only")
        self.assertTrue(representation["source_inputs_unchanged"])
        self.assertTrue(representation["answer_rubric_unchanged"])
        self.assertFalse(representation["opaque_compression_or_text_chunking"])
        self.assertIn("evidence_sources[source_id].passages[evidence_code]", representation["lookup"])
        self.assertEqual(representation["claim_identity"]["authoritative_inventory"],
                         "facet_claims supplies frozen claim IDs and exact text by facet")
        self.assertEqual(representation["claim_identity"]["answer_content"],
                         "answer_facets retains raw answer content")
        failure = protocol["answer_assessment_packets"]["preparation_failure"]
        self.assertIn("one-shot fail-closed", failure)
        self.assertIn("partial artifacts are retained", failure)
        self.assertIn("must not be cleaned, retried", failure)
        self.assertEqual(hashlib.sha256((HERE / "grouped-evidence-catalog.py.txt").read_bytes()).hexdigest(),
                         protocol["answer_assessment_packets"]["representation"]["codec_sha256"])

    def test_exp095_size_failure_is_terminal_and_has_no_qualification_credit(self):
        protocol = json.loads(CURRENT.read_text())
        receipt = json.loads((HERE / "exp095-terminal-preflight.json").read_text())
        self.assertEqual(receipt["status"], "blocked-before-answer-assessor-submission")
        self.assertEqual(receipt["input_bytes"], 505412)
        self.assertEqual(receipt["viewer_delivery_bytes_including_reserve"], 551306)
        self.assertEqual(receipt["unchanged_total_delivery_cap_bytes"], 384000)
        self.assertEqual(receipt["provider_calls"], 0)
        self.assertFalse(receipt["sparse_fixture_qualification_credit"])
        self.assertIn("new representation", receipt["prospective_successor"])
        self.assertIn("supplied to the grader", protocol["answer_assessment_packets"]["delivery"])

    def test_exp096_runtime_authority_and_answer_order_provenance_are_explicit(self):
        protocol = json.loads(CURRENT.read_text())
        self.assertIn("new EXP-096 admission", protocol["recovery_runtime"]["health"])
        self.assertIn("new EXP-096 admission", protocol["recovery_runtime"]["health_request"])
        self.assertIn("EXP-096 lossless grouped evidence table", protocol["recovery_runtime"]["purpose"])
        self.assertIn("guarded selector artifact", protocol["preflight_requirements"]["answerer_inputs"])
        self.assertIn("base-variant top-10 orders", protocol["preflight_requirements"]["answerer_inputs"])

    def test_fixed_candidate_and_all_frozen_numerical_gates(self):
        new = json.loads(CURRENT.read_text())
        old = json.loads(PREDECESSOR_PROTOCOL.read_text())
        for field in ("candidate", "quality", "task_use", "consumer", "jev_invocation"):
            self.assertEqual(new[field], old[field], field)
        quality = new["quality"]
        self.assertEqual(quality["independent_references"], ["A", "B"])
        self.assertEqual(quality["mean_ndcg10_gain_minimum"], 0.02)
        self.assertEqual(quality["bootstrap_draws"], 10000)
        self.assertEqual(quality["bootstrap_seed"], 7701)
        self.assertEqual(quality["per_case_delta_minimum"], -0.03)
        self.assertEqual(quality["top10_overlap_minimum"], 0.8)
        self.assertEqual(quality["incumbent_identity_displacement"], "diagnostic_only")
        self.assertEqual(quality["exact_navigation_targets_per_stage"], 5)
        self.assertTrue(quality["all_targets_acquired_and_top1_required"])

    def test_only_source_and_answer_reference_tokens_are_reencoded(self):
        protocol = json.loads(CURRENT.read_text())
        source = protocol["source_citation_aliases"]
        self.assertEqual(source["format"], "D<task digit>E<four decimal digits>")
        self.assertIn("before dispatch", source["codebook"])
        self.assertIn("exact inverse decoding", source["validation"])
        self.assertIn("before dispatch", source["codebook"])
        self.assertEqual(source["unchanged"], [
            "source_id", "task_id", "claim/facet/task/packet/arm identities",
            "source text and context", "all non-reference values",
            "array/list order and multiplicity", "catalog order"])
        self.assertEqual(source["codec_sha256"], hashlib.sha256(
            (HERE / "citation-codec.py.txt").read_bytes()).hexdigest())
        predecessor = HERE.parent / "EXP-094"
        for key, filename in (("source_prompt_sha256", "source-assessor-prompt.txt"),
                              ("answerer_prompt_sha256", "answerer-prompt.txt"),
                              ("answer_assessor_prompt_sha256", "answer-assessor-prompt.txt")):
            current = (HERE / filename).read_bytes()
            self.assertEqual(source[key], hashlib.sha256(current).hexdigest())
            self.assertEqual(current, (predecessor / filename).read_bytes())

    def test_fresh_slots_and_cumulative_budget_are_explicit(self):
        protocol = json.loads(CURRENT.read_text())
        models = protocol["models"]
        self.assertEqual(models["source_assessment_calls_expected"], 43)
        self.assertEqual(models["answer_assessment_calls_expected"], 32)
        self.assertEqual(models["historical_grader_submissions_counted"], 28)
        self.assertEqual(models["maximum_logical_grader_submissions_per_stage"], 112)
        self.assertIn("103 of 112", models["cumulative_development_grader_call_accounting"])
        self.assertEqual(protocol["confirmation_scope"]["not_run"], True)
        self.assertIn("entirely new tasks and pools", protocol["confirmation_scope"]["future_requirement"])
        self.assertEqual(protocol["acquisition"]["carry_forward_card_assignments"], 16)
        self.assertEqual(protocol["acquisition"]["carry_forward_card_records"], 884)
        self.assertFalse(protocol["acquisition"]["failed_source_submission_reused"])

    def test_exp094_is_terminal_and_no_previous_grader_response_is_reused(self):
        protocol = json.loads(CURRENT.read_text())
        predecessor = protocol["predecessor"]
        self.assertEqual(predecessor["id"], "EXP-094")
        self.assertEqual(predecessor["terminal_status"], "interrupted/deadline-expired")
        self.assertFalse(predecessor["mutate_or_reclassify"])
        self.assertEqual(predecessor["valid_source_assignments_carried"], 5)
        self.assertTrue(predecessor["rejected_spawn_created_no_task"])
        self.assertIn("EXP-094 expired", protocol["purpose"])
        self.assertEqual(protocol["external_source_pins"]["final_text_reader"]["sha256"],
                         "__EXP095_PRIVATE_FINAL_READER_SHA256__")

    def test_only_fresh_independent_assessments_request_sol(self):
        protocol = json.loads(CURRENT.read_text())
        models = protocol["models"]
        self.assertEqual(models["assessment_model"]["model_id"], "gpt-6.1-sol")
        self.assertEqual(set(models["assessment_model"]["scope"]),
                         {"source_assessment", "answer_assessment"})
        self.assertEqual(models["answerer_model"], "free")
        self.assertEqual(models["health_model"], "free")
        self.assertEqual(models["source_assessment_calls_expected"], 43)
        self.assertEqual(models["answer_assessment_calls_expected"], 32)
        self.assertEqual(models["historical_grader_submissions_counted"] +
                         models["fresh_source_and_answer_assessments"], 103)
        self.assertEqual(models["maximum_logical_grader_submissions_per_stage"], 112)
        self.assertEqual(models["maximum_total_logical_model_submissions_per_stage"], 130)
        self.assertIn("requested model in each dispatch receipt",
                      models["assessment_model"]["assignment"])

    def test_source_manifest_binds_assessment_model_without_changing_slot_rows(self):
        preparer = (HERE / "prepare-source-assessments.py.txt").read_text()
        self.assertIn("requested_assessment_model_id", preparer)
        self.assertIn('"gpt-6.1-sol"', preparer)
        carrier = (HERE / "final-json-carrier.py.txt").read_text()
        self.assertIn('"protocol_sha256", "qualified_source_revision"', carrier)
        self.assertNotIn('"requested_assessment_model_id"', carrier)

    def test_no_card_calls_or_output_repairs(self):
        protocol = json.loads(CURRENT.read_text())
        self.assertNotIn("card-reference", protocol["phase_plan"]["ordered_phases"])
        self.assertFalse(protocol["phase_plan"]["card_reference_phase_permitted"])
        self.assertIn("no retries", protocol["stop"])
        self.assertEqual(protocol["acquisition"]["new_searches"], 0)
        self.assertEqual(protocol["acquisition"]["new_source_fetches"], 0)


if __name__ == "__main__":
    unittest.main()
