class CheckpointError(ValueError):
    pass

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def canonical(value):
    return (json.dumps(value, sort_keys=True, ensure_ascii=False,
        separators=(",", ":"), allow_nan=False) + "\n").encode("utf-8")

def strict_json(raw):
    def pairs(rows):
        value = {}
        for key, item in rows:
            if key in value:
                raise CheckpointError("duplicate-json-key")
            value[key] = item
        return value
    def invalid(_):
        raise CheckpointError("nonfinite-json-value")
    return json.loads(raw, object_pairs_hook=pairs, parse_constant=invalid)

def safe_file(root, relative):
    path = Path(relative)
    if (not isinstance(relative, str) or not relative or path.is_absolute()
            or any(part in (".", "..") for part in path.parts)):
        raise CheckpointError("unsafe-checkpoint-path")
    cursor = root
    for part in path.parts:
        cursor = cursor / part
        if cursor.is_symlink():
            raise CheckpointError("checkpoint-symlink")
    if not cursor.is_file() or not cursor.resolve().is_relative_to(root):
        raise CheckpointError("checkpoint-file-missing")
    return cursor

def verify_inventory(root, pins_raw):
    if sha(pins_raw) != PINS_SHA256:
        raise CheckpointError("checkpoint-manifest-pin")
    pins = strict_json(pins_raw)
    if (pins.get("schema") != "exp095-source-checkpoint-pins/1"
            or pins.get("origin") != "EXP-094"
            or pins.get("accepted_assignments") != 5
            or pins.get("all_originally_valid_assignments_required") is not True
            or pins.get("quality_analyzed") is not False
            or pins.get("grade_repaired") is not False
            or pins.get("prior_time_credit_seconds") != 0):
        raise CheckpointError("checkpoint-scope")
    rows = pins.get("files")
    if not isinstance(rows, list) or len(rows) != pins.get("file_count"):
        raise CheckpointError("checkpoint-inventory-count")
    names = set()
    total = 0
    for row in rows:
        if type(row) is not dict or set(row) != {"path", "bytes", "sha256"}:
            raise CheckpointError("checkpoint-inventory-row")
        path = safe_file(root, row["path"])
        if row["path"] in names:
            raise CheckpointError("checkpoint-inventory-duplicate")
        names.add(row["path"])
        raw = path.read_bytes()
        if len(raw) != row["bytes"] or sha(raw) != row["sha256"]:
            raise CheckpointError("checkpoint-file-pin")
        total += len(raw)
    observed = {p.relative_to(root).as_posix() for p in root.rglob("*") if p.is_file()}
    if observed != names or total != pins.get("total_bytes"):
        raise CheckpointError("checkpoint-inventory-coverage")
    return pins

def load_pinned(name, path, expected):
    if path.is_symlink() or sha(path.read_bytes()) != expected:
        raise CheckpointError("checkpoint-verifier-source-pin")
    module = types.ModuleType(name)
    module.__file__ = str(path)
    sys.modules[name] = module
    exec(compile(path.read_bytes(), str(path), "exec"), module.__dict__)
    return module

def verify_source_checkpoint(checkpoint_root, pins_path=None, *, source_input_manifest=None):
    root = Path(checkpoint_root).resolve()
    if Path(checkpoint_root).is_symlink() or not root.is_dir():
        raise CheckpointError("checkpoint-root")
    pins_file = Path(pins_path) if pins_path is not None else HERE / "source-checkpoint-pins.json"
    if pins_file.is_symlink():
        raise CheckpointError("checkpoint-manifest-symlink")
    pins = verify_inventory(root, pins_file.read_bytes())
    old_bytecode = sys.dont_write_bytecode
    sys.dont_write_bytecode = True
    try:
        return _replay(root, pins, source_input_manifest)
    finally:
        sys.dont_write_bytecode = old_bytecode

            or q.get("external_source_pins", {}).get("final_text_reader", {}).get("sha256")

        raise CheckpointError("checkpoint-origin-qualification")
    reader = load_pinned("exp095_historical_reader", root / "historical-reader.py", pins["original_reader_sha256"])

    # exact historical parent/path/depth checks, parser and tools remain intact.

    manifest = strict_json(safe_file(root, "assessment-inputs/source-slot-manifest.json").read_bytes())
    jobs = manifest["active_source_slots"]
    if len(jobs) != 48 or manifest.get("requested_assessment_model_id") != "gpt-6.1-sol":
