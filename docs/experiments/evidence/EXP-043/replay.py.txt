def project(cards):
    return [{k:c[k] for k in ("id","title","url","snippet")} for c in cards]

def validate_facets(fs):
    if type(fs)is not list or not 1<=len(fs)<=MAX_FACETS: raise ValueError("facet count must be 1..4")
    seen=set()
    for facet in fs:
        if type(facet)is not dict or set(facet)!={"id","description"}: raise ValueError("facet must contain only id and description")
        fid,desc=facet["id"],facet["description"]
        if not isinstance(fid,str) or len(fid.encode("utf-8"))>64 or not re.fullmatch(r"[a-z][a-z0-9_]{0,63}",fid) or fid in seen: raise ValueError("facet ID invalid or duplicate")
        if not isinstance(desc,str) or not desc.strip() or len(desc.encode("utf-8"))>256: raise ValueError("facet description invalid or too long")
        seen.add(fid)
    return fs

def candidate_body(kind,task,cards,fs,contract):
    cards=project(cards); questions={}
    if kind=="D":
        for c in cards:
            questions[c["id"]]={"type":"score","instructions":contract["instructions"].replace("{id}",c["id"]),"criteria":list(contract["criteria"])}
        state={"query":task,"candidates":cards}
    elif kind=="F":
        validate_facets(fs)
        for i,c in enumerate(cards):
            for j,_ in enumerate(fs):
                qid=f"facet-{i:03d}-{j:02d}"
                questions[qid]={"type":"noul","instructions":F_TEMPLATE.format(candidate_id=c["id"],facet_index=j)}
        state={"query":task,"candidates":cards,"requested_facets":[{"id":f["id"],"description":f["description"]} for f in fs]}
    else: raise ValueError("candidate body kind must be D or F")
    return {"model":MODEL,"state":state,"questions":questions}

def request_record(spec):
    body=spec["body"]
    return {k:v for k,v in spec.items() if k!="body"}|{"request_bytes":len(body),"request_sha256":sha(body),"state_bytes":len(wire(strict_json(body)["state"]))}

def request_eligibility(spec):
    n=len(spec["candidate_ids"]); reasons=[]
    if n>MAX_CARDS: reasons.append("candidate_count")
    if len(spec["question_ids"])>MAX_QUESTIONS: reasons.append("question_count")
    if spec["request_bytes"]>spec["request_cap"]: reasons.append("request_bytes")
    if spec["state_bytes"]>CAP_STATE: reasons.append("state_bytes")
    if len(spec["candidate_ids"])!=len(set(spec["candidate_ids"])): reasons.append("duplicate_candidates")
    if spec["kind"] in {"D","F"} and n<1: reasons.append("empty_candidate_pool")
    return {"eligible":not reasons,"reasons":reasons,"fallback_order_ids":None if not reasons else list(spec["candidate_ids"])}

def _make_candidate_spec(rid,pool,kind,role,task,cards,fs,contract,pair_id=None,repeat_of=None,target=None):

    for c in cards: H.validate_card(c)
    facets=fs or []; qids=ids if kind=="D" else [f"facet-{i:03d}-{j:02d}" for i in range(len(ids)) for j in range(len(facets))]
    if len(task.encode("utf-8"))>4096: raise ValueError(f"query too large: {rid}")
    cap=CAP_DF
    spec={"id":rid,"pool_id":pool,"kind":kind,"role":role,"task":task,"candidate_ids":ids,"question_ids":qids,"score_ids":ids if kind=="D" else [],"facets":facets,"pair_id":pair_id,"repeat_of":repeat_of,"target_id":target,"request_cap":cap,"body":body}
    spec.update(request_record(spec))
    if not request_eligibility(spec)["eligible"]: raise ValueError(f"candidate request exceeds registered limits: {rid}: {request_eligibility(spec)['reasons']}")
    return spec

def validate_refs(root,data,grouped,required):
    pool_ref,hashes=BASE.validate_refs(root,data,grouped,required)
    # The pinned helper returns pool -> reference -> card. Transpose explicitly and verify every representative.
    refs={"A":{},"B":{}}
    expected_pools=set(SOFTWARE)|set(NATURAL)
    if set(pool_ref)!=expected_pools: raise ValueError("reference pool inventory mismatch")
    for pool in sorted(expected_pools):
        if set(pool_ref[pool])!={"A","B"}: raise ValueError(f"reference pair missing: {pool}")
        expected={c["id"] for c in grouped[pool]}
        for ref in ("A","B"):
            rows=pool_ref[pool][ref]
            if set(rows)!=expected: raise ValueError(f"representative annotation membership mismatch: {ref}/{pool}")
            refs[ref][pool]=rows
    if any(set(refs[r])!=expected_pools for r in refs): raise ValueError("transposed reference shape mismatch")
    return refs,hashes

def validate_answer(data,spec):
    if type(data)is not dict or set(data)!={"model","answers","usage","_elapsed_ms"}: return {},{},["envelope"]
    errors=[]
    if data["model"]!=MODEL: errors.append("model")
    usage=data["usage"]
    if type(usage)is not dict or set(usage)!={"input_tokens","output_tokens"} or any(type(v)is not int or v<0 for v in usage.values()): errors.append("usage")
    elapsed=data["_elapsed_ms"]
    if isinstance(elapsed,bool) or not isinstance(elapsed,(int,float)) or not math.isfinite(elapsed) or elapsed<0 or elapsed>MAX_REMOTE_MS: errors.append("remote_http_ms")
    answers=data["answers"]
    if type(answers)is not dict or set(answers)!=set(spec["question_ids"]): return {},{},errors+["answer_membership"]
    scores={}; nouls={cid:{} for cid in spec["candidate_ids"]}
    if spec["kind"] in {"W0","D"}:
        for cid in spec["score_ids"]:
            a=answers[cid]
            if type(a)is not dict or not {"type","score"}<=set(a) or set(a)-{"type","score","confidence","legend","probabilities"} or a.get("type")!="score": errors.append("score_type:"+cid); continue
            val=a.get("score")
            if isinstance(val,bool) or not isinstance(val,(int,float)) or not math.isfinite(val) or not 0<=val<=9: errors.append("score_value:"+cid); continue
            _,es=H.validate_response({"model":MODEL,"answers":{cid:a}},{"candidate_ids":[cid]})
            if es: errors.extend(es); continue
            scores[cid]=float(val)
    elif spec["kind"]=="F":
        for i,cid in enumerate(spec["candidate_ids"]):
            for j,_ in enumerate(spec["facets"]):
                qid=f"facet-{i:03d}-{j:02d}"; a=answers[qid]
                if type(a)is not dict or set(a)!={"type","noul"} or a.get("type")!="noul": errors.append("noul_type:"+qid); continue
                val=a.get("noul")
                if isinstance(val,bool) or not isinstance(val,(int,float)) or not math.isfinite(val) or not 0<=val<=1: errors.append("noul_value:"+qid); continue
                nouls[cid][j]=float(val)
    else: errors.append("kind")
    if spec["kind"] in {"W0","D"} and len(scores)!=len(spec["score_ids"]): errors.append("score_missing")
    if spec["kind"]=="F" and any(len(row)!=len(spec["facets"]) for row in nouls.values()): errors.append("facet_missing")
    usagein=usage.get("input_tokens") if type(usage)is dict else None
    if usagein is not None and usagein>MAX_CALL_INPUT: errors.append("per_call_input_tokens")
    return (scores,nouls,errors)

def parse_transport(raw):
    obj=H.strict_json_loads(raw.decode("utf-8"))
    if type(obj)is dict and "error" in obj:
        required={"error","status","_elapsed_ms"}; optional={"provider_error_bytes","provider_error_sha256","provider_error_code"}
        if not required<=set(obj) or set(obj)-required-optional: raise ValueError("unexpected provider error envelope")
        if not isinstance(obj["error"],str) or not obj["error"] or type(obj["status"])is not int or not 400<=obj["status"]<=599: raise ValueError("invalid sanitized provider error")
        elapsed=obj["_elapsed_ms"]
        if isinstance(elapsed,bool) or not isinstance(elapsed,(int,float)) or not math.isfinite(elapsed) or not 0<=elapsed<=MAX_REMOTE_MS: raise ValueError("invalid provider error elapsed time")
        if set(obj)&optional:
            if not optional<=set(obj) or type(obj["provider_error_bytes"])is not int or not 0<=obj["provider_error_bytes"]<=MAX_RESPONSE+1 or not isinstance(obj["provider_error_sha256"],str) or not re.fullmatch(r"[0-9a-f]{64}",obj["provider_error_sha256"]) or obj["provider_error_code"] not in PROVIDER_ERROR_CODES: raise ValueError("invalid sanitized provider diagnostics")
        return obj
    return H.parse_transport_result(raw)

def read_receipts(root,manifest,require_all=False):
    receipts={}; errors=[]; byid={r["id"]:r for r in manifest["requests"]}; folder=root/"receipts"
    if not folder.exists(): return {},["missing:"+rid for rid in byid]
    paths=list(folder.iterdir()); allowed_names={f"{rid}.json" for rid in byid}|{f"{rid}.response.bin" for rid in byid}
    if any(p.name not in allowed_names for p in paths): errors.append("unknown_receipt_artifact")
    for rid,req in byid.items():
        rp=folder/(rid+".json"); bp=folder/(rid+".response.bin")
        if not rp.is_file() or not bp.is_file(): errors.append("missing:"+rid); continue
        try:
            rec=load_json(rp)
            if bp.stat().st_size>MAX_RESPONSE: raise ValueError("response byte cap")
            raw=bp.read_bytes(); parsed=parse_transport(raw)
        except Exception: errors.append("malformed:"+rid); continue
        allowed={"schema","id","request_sha256","model","answers","usage","remote_http_ms","transport_wall_ms","transport_exit_code","response_bytes","response_sha256","fallback_source","fallback_order_ids","provider_http_status","provider_error_bytes","provider_error_sha256","provider_error_code","status","validation_errors","error"}
        if type(rec)is not dict: errors.append("receipt_type:"+rid); continue
        if "error" in parsed: errors.append("provider_error_receipt:"+rid); continue
        data={"model":rec.get("model"),"answers":rec.get("answers"),"usage":rec.get("usage"),"_elapsed_ms":rec.get("remote_http_ms")}
        scores,nouls,errs=validate_answer(data,req)
        raw_scores,raw_nouls,raw_errs=validate_answer(parsed,req)
        raw_matches=not raw_errs and parsed==data
        if type(rec)is not dict or set(rec)!=allowed or rec.get("schema")!="exp043-receipt/1" or rec.get("id")!=rid or rec.get("request_sha256")!=req["request_sha256"] or rec.get("status")!="valid" or rec.get("validation_errors")!=[] or rec.get("error") is not None:
            errors.append("receipt_identity:"+rid); continue
        wall=rec.get("transport_wall_ms"); exitcode=rec.get("transport_exit_code")
        if errs or not raw_matches or (isinstance(wall,bool) or not isinstance(wall,(int,float)) or not math.isfinite(wall) or wall<0 or wall>MAX_WALL*1000) or type(exitcode)is not int or exitcode!=0 or type(rec.get("response_bytes"))is not int or rec["response_bytes"]!=len(raw) or rec["response_bytes"]>MAX_RESPONSE or rec.get("response_sha256")!=sha(raw):
            errors.append("invalid_receipt:"+rid); continue
        rec["scores"]=scores; rec["nouls"]=nouls; receipts[rid]=rec
    if require_all and errors: return receipts,errors
    return receipts,errors

def read_history(root,manifest,refhash):
    path=root/"attempts.jsonl"
    if not path.exists(): return set(),set(),None
    events=[strict_json(line) for line in path.read_bytes().splitlines() if line]
    if any(type(event)is not dict for event in events): raise ValueError("attempt ledger entries must be objects")
    starts=[event for event in events if event.get("event")=="run_started"]
    if len(starts)!=1 or starts[0].get("reference_hashes")!=refhash or starts[0].get("prepared_manifest_sha256")!=sha((root/"prepared-manifest.json").read_bytes()): raise ValueError("run history not bound to frozen refs/manifest")
    if set(starts[0])!={"event","prepared_manifest_sha256","reference_hashes","at_utc"}: raise ValueError("malformed run-start event")
    H.parse_run_start_time(starts[0].get("at_utc")); begun=set(); done=set(); terminal=False; byid={r["id"]:r for r in manifest["requests"]}
    for event in events:
        kind=event.get("event")
        if kind not in {"run_started","attempt_started","attempt_finished","runner_stopped"}: raise ValueError("unknown attempt-ledger event")
        if kind=="run_started": continue
        if kind=="attempt_started":
            rid=event.get("request_id"); allowed={"event","attempt","request_id","request_sha256","reference_hashes","preflight_input_token_estimate","preflight_output_token_estimate","preflight_is_heuristic","at_utc"}
            if set(event)!=allowed or event.get("reference_hashes")!=refhash or terminal or rid not in byid or rid in begun or event.get("attempt")!=1 or event.get("request_sha256")!=byid[rid]["request_sha256"]: raise ValueError("invalid or duplicate started attempt")
            if begun-done: raise ValueError("attempts must be serial")
            begun.add(rid)
        elif kind=="attempt_finished":
            rid=event.get("request_id"); allowed={"event","attempt","request_id","request_sha256","status","error","input_tokens","output_tokens","response_sha256","response_bytes","at_utc"}
            if set(event)!=allowed or rid not in begun or rid in done or event.get("attempt")!=1 or event.get("request_sha256")!=byid[rid]["request_sha256"] or event.get("status") not in {"valid","invalid"} or (event.get("status")=="valid")!=(event.get("error") is None): raise ValueError("invalid attempt-finished event")
            for key in ("input_tokens","output_tokens"):
                val=event.get(key)
                if val is not None and (type(val)is not int or val<0): raise ValueError("invalid ledger token count")
            size=event.get("response_bytes"); digest=event.get("response_sha256")
            size_cap=MAX_RESPONSE+1 if event["status"]=="invalid" and event.get("error")=="response_overflow" else MAX_RESPONSE
            if type(size)is not int or size<0 or size>size_cap or not isinstance(digest,str) or not re.fullmatch(r"[0-9a-f]{64}",digest): raise ValueError("invalid ledger response identity")
            receipt_path=root/"receipts"/(rid+".json"); response_path=root/"receipts"/(rid+".response.bin")
            if not receipt_path.is_file() or not response_path.is_file(): raise ValueError("finished attempt lacks durable receipt/blob")
            receipt=load_json(receipt_path); raw=response_path.read_bytes()
            if type(receipt)is not dict or receipt.get("request_sha256")!=event["request_sha256"] or receipt.get("status")!=event["status"] or receipt.get("error")!=event["error"] or receipt.get("response_bytes")!=size or receipt.get("response_sha256")!=digest or len(raw)!=size or sha(raw)!=digest: raise ValueError("ledger/receipt/raw response identity mismatch")
            usage=receipt.get("usage")
            if event["status"]=="valid":
                if type(usage)is not dict or type(event.get("input_tokens"))is not int or type(event.get("output_tokens"))is not int or event["input_tokens"]!=usage.get("input_tokens") or event["output_tokens"]!=usage.get("output_tokens"): raise ValueError("valid attempt usage mismatch")
            elif event.get("input_tokens") is not None or event.get("output_tokens") is not None:
                if type(usage)is not dict or event.get("input_tokens")!=usage.get("input_tokens") or event.get("output_tokens")!=usage.get("output_tokens"): raise ValueError("failed attempt known usage mismatch")
            done.add(rid)
            if event["status"]!="valid": terminal=True
        else:
            shapes=({"event","reason","at_utc"},{"event","request_id","reason","at_utc"},{"event","request_id","reason","fallback_order_ids","at_utc"},{"event","request_id","reason","input_estimate","output_estimate","at_utc"})
            if set(event) not in shapes or not isinstance(event.get("reason"),str) or not event["reason"]: raise ValueError("malformed terminal event")
            terminal=True
    if terminal: raise ValueError("terminal stop is durable; never resume/retry")
    if begun-done: raise ValueError("uncertain started attempt; never retry")
    return begun,done,starts[0]["at_utc"]

def score_order(spec,rec):
    ids=spec["candidate_ids"]; scores=rec["scores"]
    if spec["kind"]=="W0":
        scored=BASE.stable_scores(scores,spec["score_ids"]); return scored+[cid for cid in ids if cid not in set(scored)]
    return BASE.stable_scores(scores,ids)

def combined_order(dspec,drec,fspec,frec,incumbent_order):
    if not drec or not frec or drec.get("status")!="valid" or frec.get("status")!="valid": return list(incumbent_order),"w0_pair_fallback"
    if dspec["candidate_ids"]!=fspec["candidate_ids"] or dspec["task"]!=fspec["task"] or dspec["facets"] or not fspec["facets"]: raise ValueError("D/F candidate order, purpose or facet boundary mismatch")
    if set(strict_json(dspec["body"])["state"])!={"query","candidates"} or set(strict_json(fspec["body"])["state"])!={"query","candidates","requested_facets"}: raise ValueError("D/F states do not isolate facet context")
    order,_=BASE.selector(drec["scores"],frec["nouls"],fspec["facets"],dspec["candidate_ids"])
    return order,"candidate_pair"

def request_record_for_manifest(spec): return {k:v for k,v in spec.items() if k!="body"}

def ranking(spec,receipt):
    if spec["kind"] in {"W0","D"}: return score_order(spec,receipt)
    raise ValueError("F is a facet-only response and has no standalone rank order")

def transpose_references(pool_ref):
    expected=set(SOFTWARE)|set(NATURAL)
    if set(pool_ref)!=expected: raise ValueError("reference pool set mismatch")
    out={"A":{},"B":{}}
    for pool,byref in pool_ref.items():
        if set(byref)!={"A","B"}: raise ValueError("reference labels missing")
        for ref,rows in byref.items():
            if pool in out[ref]: raise ValueError("duplicate reference/pool after transpose")
            out[ref][pool]=rows
    if any(set(out[r])!=expected for r in out): raise ValueError("transpose lost reference pool")
    return out

def analyze(root):
    root=root_path(root); check=verify(root,True)
    if not check["ok"] or not check["ready_to_run"]: raise ValueError("verification/ref qualification failed")
    manifest=load_json(root/"prepared-manifest.json"); receipts,errors=read_receipts(root,manifest,True)
    if errors or len(receipts)!=CALL_COUNT: raise ValueError(f"all 65 valid receipts required: {errors}")
    begun,done,start=read_history(root,manifest,check["reference_hashes"]); allids={r["id"] for r in manifest["requests"]}
    if begun!=allids or done!=allids: raise ValueError("one completed attempt per request required")
    data=data_sources(root); grouped,_=BASE.qualify_groups_memory(root,data); specs={s["id"]:s for s in expected_specs(root,data,grouped)}
    ranks={}; facetids={p:[x["id"] for x in data[2][p]] for p in (*SOFTWARE,*NATURAL)}
    def make_e(pool,variant,rot=False):

        ds,fr=specs[did],receipts[fid]; dr=receipts[did]
        order,source=combined_order(ds,dr,specs[fid],fr,score_order(specs[f"{pool}-w0"],receipts[f"{pool}-w0"]))
        if source!="candidate_pair": raise ValueError(f"invalid candidate pair in completed analysis: {pool}/{variant}")

    for pool in SOFTWARE:
        w0=score_order(specs[f"{pool}-w0"],receipts[f"{pool}-w0"]); d=score_order(specs[f"{pool}-d"],receipts[f"{pool}-d"]); e=make_e(pool,"main")
        ranks[pool]={"w0":w0,"d_star":d,"e":e}
    for pool in EXTENDED:
        w0=score_order(specs[f"{pool}-w0"],receipts[f"{pool}-w0"]); d=score_order(specs[f"{pool}-d"],receipts[f"{pool}-d"]); e=make_e(pool,"main")
        ranks[pool]={"w0":w0,"d_star":d,"e":e,"repeat":make_e(pool,"repeat"),"rotated":make_e(pool,"rotated20")}
    pool_ref,_=BASE.validate_refs(root,data,grouped,True); refs=transpose_references(pool_ref)
    report=BASE.ANALYSIS.analyze(ranks,refs,facetids,H.ndcg_at_10,H.paired_bootstrap)
    nav=[]; navpass=True
    for q,cid in NAV_IDS:
        w=score_order(specs[f"nav-{q}-{cid}-w0"],receipts[f"nav-{q}-{cid}-w0"]); d=score_order(specs[f"nav-{q}-{cid}-d"],receipts[f"nav-{q}-{cid}-d"])
        ok=cid in w[:3] and cid in d[:3]; navpass=navpass and ok; nav.append({"query_id":q,"target_id":cid,"w0_top3":w[:3],"d_top3":d[:3],"target_rank_w0":w.index(cid)+1,"target_rank_d":d.index(cid)+1,"f_dispatched":False,"passed":ok})
    pairdur={}
    for s in manifest["requests"]:
        if s["kind"]=="F":
            d_id=s["pair_id"].replace("-main","-d").replace("-repeat","-d-repeat").replace("-rotated20","-d-rotated20")
            pairdur[s["pair_id"]]=receipts[d_id]["remote_http_ms"]+receipts[s["id"]]["remote_http_ms"]
    tin=sum(r["usage"]["input_tokens"] for r in receipts.values()); tout=sum(r["usage"]["output_tokens"] for r in receipts.values())
    remote=[r["remote_http_ms"] for r in receipts.values()]
    operational=len(receipts)==CALL_COUNT and tin<=MAX_TOTAL_INPUT and tout<=MAX_TOTAL_OUTPUT and all(r["usage"]["input_tokens"]<=MAX_CALL_INPUT for r in receipts.values()) and max(remote)<=MAX_REMOTE_MS and max(pairdur.values())<=MAX_PAIR_MS
    passed=report["development_screen_passed"] and navpass and operational
    result={"schema":"exp043-report/1","decision":"supports_optional_mode_for_fresh_confirmation_only" if passed else "unsupported_for_selection","development_analysis":report,"navigation":{"controls":nav,"passed":navpass,"e_bypassed":True},"operational":{"valid_calls":len(receipts),"expected_calls":CALL_COUNT,"arm_counts":{"W0":18,"D":26,"F":21},"input_tokens":tin,"output_tokens":tout,"max_per_call_input_tokens":max(r["usage"]["input_tokens"] for r in receipts.values()),"max_remote_http_ms":max(remote),"pair_remote_ms":pairdur,"pair_gate_passed":all(v<=MAX_PAIR_MS for v in pairdur.values()),"passed":operational},"grouping_sha256":manifest["grouping_sha256"],"references":check["reference_hashes"],"prepared_manifest_sha256":sha((root/"prepared-manifest.json").read_bytes()),"limitations":["saved frozen_projection corpus; not engine provenance","constructed80 has 79 post-group representatives and is operational only","reused assistant references are not independent human gold","pair remote time excludes proxy/platform overhead"]}
