from pathlib import Path

HERE=Path(__file__).resolve().parent

def load(name,path):
    l=importlib.machinery.SourceFileLoader(name,str(path)); s=importlib.util.spec_from_loader(name,l)
    m=importlib.util.module_from_spec(s); sys.modules[name]=m; l.exec_module(m); return m

def sha(raw): return hashlib.sha256(raw).hexdigest()
def canonical(value): return (json.dumps(value,ensure_ascii=False,sort_keys=True,separators=(",",":"),allow_nan=False)+"\n").encode()

class AdmissionTests(unittest.TestCase):
    def setUp(self):
        self.temp=tempfile.TemporaryDirectory(prefix="exp086-admission-")
        self.root=Path(self.temp.name)/"repo"; self.exp=self.root/"docs/experiments/evidence/EXP-086"
        self.exp.mkdir(parents=True)
        self.adm=load("exp086_admission_"+str(id(self)),HERE/"admission.py.txt")
        self.bridge_tests=load("exp086_bridge_tests_"+str(id(self)),HERE/"input-bridge-tests.py.txt")
        self._make_fixture()
        self.adm.REPO=self.root; self.adm.HERE=self.exp
        self.adm.REG_PATH="docs/experiments/evidence/EXP-086/registration-integrity.json"
        self.adm.QUAL_PATH="docs/experiments/evidence/EXP-086/qualification-integrity.json"
        self.adm.MANIFEST_PATH=self.exp/"stage-development-manifest.json"
        self.adm.LEASE_ROOT=Path(self.temp.name)/"leases-exp086"

    def tearDown(self): self.temp.cleanup()
    def git(self,*args): return subprocess.run(["git",*args],cwd=self.root,check=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE).stdout
    def commit(self,msg):
        self.git("add","-A"); self.git("-c","user.name=EXP-086 tests","-c","user.email=exp086@example.invalid","commit","-m",msg)
        return self.git("rev-parse","HEAD").decode().strip()

    def test_exact_committed_source_and_private_input_bundle_verify(self):
        proof=self.adm.verify_closure(self.source,self.qraw,self.bundle)
        self.assertEqual(proof["qualified_source_revision"],self.qualified_rev)
        self.assertEqual(proof["study_id"],"EXP-086")
        self.assertEqual(proof["input_integrity"]["file_count"],len(self.pins["files"]))

    def test_one_shot_permits_bind_stage_root_and_reject_acquisition_phase(self):
        root=Path(self.temp.name)/"stage"; root.mkdir(mode=0o700)
        clock=self.adm.create_stage_clock(root,self.qualified_rev)
        cap=self.adm.admit_live(root,self.source,self.qraw,self.bundle)
        stage=self.adm.consume_capability(cap)
        phase=self.adm.acquire_phase(stage,"health",root,clock[1])
        self.adm.require_phase(phase,clock[1],root)
        preparation=self.adm.acquire_phase(stage,"assessment-preparation",root,clock[1])
        self.adm.require_phase(preparation,clock[1],root)
        child=root/"capture"; child.mkdir()
        with self.assertRaisesRegex(self.adm.AdmissionError,"phase-permit-required"):
            self.adm.require_phase(phase,clock[1],child)
        with self.assertRaisesRegex(self.adm.AdmissionError,"stage-permit-required"):
            self.adm.acquire_phase(stage,"capture",root,clock[1])
        self.assertTrue(str(self.adm.LEASE_ROOT).endswith("leases-exp086"))

    def test_prior_exp078_clock_bytes_do_not_satisfy_new_manifest_binding(self):
        root=Path(self.temp.name)/"stage"; root.mkdir(mode=0o700)
        clock=self.adm.create_stage_clock(root,self.qualified_rev)
        old=dict(clock[0],stage_manifest_sha256="f"*64)
        raw=canonical(old)
        manifest_raw=(self.exp/"stage-development-manifest.json").read_bytes()
        manifest=json.loads(manifest_raw)
        with self.assertRaisesRegex(self.adm.AdmissionError,"stage-clock-binding"):
            self.adm._active_clock(raw,manifest,sha(manifest_raw),clock[0]["protocol_sha256"],
                                   clock[0]["cohorts_sha256"],self.qualified_rev)

    def test_other_study_identity_is_rejected(self):
        bad=dict(self.source,study_id="EXP-082")
        with self.assertRaisesRegex(self.adm.AdmissionError,"qualification-binding"):
            self.adm.verify_closure(bad,self.qraw,self.bundle)

if __name__=="__main__": unittest.main()
